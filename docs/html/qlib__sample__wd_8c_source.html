36QLIB_STATUS_T QLIB_SAMPLE_WatchDogConfig(QLIB_CONTEXT_T* qlibContext,
37 bool enable_b,
38 bool secure_b,
39 QLIB_AWDT_TH_T threshold,
40 uint32_t section_u32)
41{
42 QLIB_WATCHDOG_CONF_T watchdogCFG;
43 QLIB_WATCHDOG_CONF_T prevWatchdogCFG;
44 QLIB_STATUS_T status = QLIB_STATUS__OK;
45
46 watchdogCFG.enable = enable_b;
47 watchdogCFG.authenticated = secure_b;
48 watchdogCFG.sectionID = section_u32;
49 watchdogCFG.threshold = threshold;
50
51 /*-------------------------------------------------------------------------------------------------------
52 The rest of WD features are not among the parameters of this sample and are board related.
53 This is why we use constant values.
54 -------------------------------------------------------------------------------------------------------*/
55 watchdogCFG.lfOscEn = (bool)true;
56 watchdogCFG.swResetEn = (bool)false;
57 watchdogCFG.lock = (bool)false;
58 watchdogCFG.oscRateHz = 0u;
59 watchdogCFG.fallbackEn = (bool)false;
60
61 /*-------------------------------------------------------------------------------------------------------
62 In case of secure watchdog we need to open session to the section WD which it is associated with.
63 -------------------------------------------------------------------------------------------------------*/
64 QLIB_STATUS_RET_CHECK(QLIB_Watchdog_ConfigGet(qlibContext, &prevWatchdogCFG));
65 if ((bool)true == prevWatchdogCFG.authenticated)
66 {
67 QLIB_STATUS_RET_CHECK(QLIB_OpenSession(qlibContext, prevWatchdogCFG.sectionID, QLIB_SESSION_ACCESS_FULL));
68 }
69
70 /*-------------------------------------------------------------------------------------------------------
71 Configure the watchdog with the parameters defined above.
72 -------------------------------------------------------------------------------------------------------*/
73 QLIB_STATUS_RET_CHECK_GOTO(QLIB_Watchdog_ConfigSet(qlibContext, &watchdogCFG), status, close_session);
74
75 /*-------------------------------------------------------------------------------------------------------
76 Now WD is configured and the counter on flash starts to count to 2^^threshold, incrementing every
77 second ( not accurate ). Once it gets there reset pin is set, so you need to touch (pet) the
78 watchdog to zero the timer.
79 -------------------------------------------------------------------------------------------------------*/
80
81close_session:
82 if ((bool)true == prevWatchdogCFG.authenticated)
83 {
84 (void)QLIB_CloseSession(qlibContext, prevWatchdogCFG.sectionID);
85 }
86
87 return status;
88}

90QLIB_STATUS_T QLIB_SAMPLE_WatchDogTouch(QLIB_CONTEXT_T* qlibContext, bool secure_b, uint32_t section_u32)
91{
92 QLIB_STATUS_T status = QLIB_STATUS__OK;
93
94 if (secure_b == (bool)true)
95 {
96 /*---------------------------------------------------------------------------------------------------
97 In case of secure watchdog we need to open session to the section WD which it is associated with.
98 ---------------------------------------------------------------------------------------------------*/
99 QLIB_STATUS_RET_CHECK(QLIB_OpenSession(qlibContext, section_u32, QLIB_SESSION_ACCESS_FULL));
100 }
101
102 /*-------------------------------------------------------------------------------------------------------
103 Touch the watchdog to zero the flash counter.
104 -------------------------------------------------------------------------------------------------------*/
105 QLIB_STATUS_RET_CHECK_GOTO(QLIB_Watchdog_Touch(qlibContext), status, close_session);
106
107 /*-------------------------------------------------------------------------------------------------------
108 Now the counter on flash is zeroed. This prevents from the counter to reach the threshold and
109 reset the chip.
110 -------------------------------------------------------------------------------------------------------*/
111close_session:
112 if ((bool)true == secure_b)
113 {
114 (void)QLIB_CloseSession(qlibContext, section_u32);
115 }
116
117 return status;
118}

135

140 -------------------------------------------------------------------------------------------------------*/
141close_session:

143 {
144 (void)QLIB_CloseSession(qlibContext, section_u32);
145 }

199QLIB_STATUS_T QLIB_SAMPLE_WatchDog_manuf_flow(QLIB_CONTEXT_T* qlibContext)
200{
201 QLIB_HW_VER_T hwVer;
202 QLIB_ID_T id;
203 uint32_t sect_u32;
204 uint8_t die_u8 = 0u;
205 QLIB_FLASH_CONFIG_T flashCfg = {0u};
206 QLIB_DIE_CONFIG_T dieCfg = {0u};
207 QLIB_SECTIONS_CONF_TABLE_T cfgLine[QLIB_MAX_DIES_SUPPORTED];
208 QLIB_EXTENDED_SECTION_CONF_T* sectCfg;
209 uint8_t diesNum_u8 = QLIB_GET_NUM_OF_DIES(qlibContext);
210
211 // See what chip is connected and its unique ids. We might need this info to obtain keys for this
212 // chips according to formula or via DB
213 QLIB_STATUS_RET_CHECK(QLIB_GetHWVersion(qlibContext, &hwVer));
214 QLIB_STATUS_RET_CHECK(QLIB_GetId(qlibContext, &id));
215
216 // We do read / modify /write configuration since we want to change only what is needed
217 // read configuration
218 QLIB_STATUS_RET_CHECK(QLIB_GetDeviceConfigMultiDie(qlibContext, &flashCfg, &dieCfg, cfgLine, diesNum_u8));
219
220 // modify configuration
221
222 // set pre provision master key if needed
223 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetDeviceKey(qlibContext,
224 id.std.uniqueID,
225 QLIB_KID__DEVICE_KEY_PRE_PROVISIONING,
226 dieCfg.preProvisionedMasterKey));
227
228 // set new master key
229 QLIB_STATUS_RET_CHECK(
230 QLIB_SAMPLE_GetDeviceKey(qlibContext, id.std.uniqueID, QLIB_KID__DEVICE_MASTER, dieCfg.deviceMasterKey));
231
232 /*-------------------------------------------------------------------------------------------------------
233 This is the default WD configuration stored in section configuration loaded on power up.
234 Be careful with it, since if you configure and enable default WD threshold which is very short, flash may signal reset
235 even before board initialization is over.
236 -------------------------------------------------------------------------------------------------------*/
237 sect_u32 = 1u;
238 flashCfg.watchdogDefault.sectionID = sect_u32;
239 flashCfg.watchdogDefault.authenticated = (bool)false;
240 flashCfg.watchdogDefault.enable = (bool)true;
241 flashCfg.watchdogDefault.fallbackEn = (bool)false;
242 flashCfg.watchdogDefault.lfOscEn = (bool)true;
243 flashCfg.watchdogDefault.lock = (bool)false;
244 flashCfg.watchdogDefault.oscRateHz = 0u;
245 flashCfg.watchdogDefault.swResetEn = (bool)false;
246 flashCfg.watchdogDefault.threshold = QLIB_AWDT_TH_12_DAYS;
247
248 // adding a section for the watchdog
249#if QLIB_MAX_DIES_SUPPORTED > 1u
250 QLIB_STATUS_RET_CHECK(QLIB_GetActiveDie(qlibContext, &die_u8));
251#endif // QLIB_MAX_DIES_SUPPORTED > 1u
252 sectCfg = cfgLine[die_u8].sectionConfigTable;
253
254 sectCfg[sect_u32].baseAddr = QLIB_SAMPLE_RESERVED_BOOT_SECTION_SIZE;
255 sectCfg[sect_u32].size = SIZE_256KB_;
256
257 sectCfg[sect_u32].sectionConfig.policy.plainAccessReadEnable = 1u;
258 sectCfg[sect_u32].sectionConfig.policy.plainAccessWriteEnable = 0u;
259 sectCfg[sect_u32].sectionConfig.policy.authPlainAccess = 1u;
260 sectCfg[sect_u32].sectionConfig.policy.checksumIntegrity = 0u;
261 sectCfg[sect_u32].sectionConfig.policy.digestIntegrity = 0u;
262 sectCfg[sect_u32].sectionConfig.policy.digestIntegrityOnAccess = 0u;
263 sectCfg[sect_u32].sectionConfig.policy.rollbackProt = 0u;
264 sectCfg[sect_u32].sectionConfig.policy.slog = 0u;
265 sectCfg[sect_u32].sectionConfig.policy.writeProt = 0u;
266
267 // Full access key configuration.This is the key to use later for the section configuration
268 // changes or write access
269 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSectionAccessKey(qlibContext,
270 id.std.uniqueID,
271 QLIB_KID__FULL_ACCESS_SECTION,
272 sect_u32,
273 die_u8,
274 sectCfg[sect_u32].fullAccessKey));
275
276 // get restricted access key to configure into chip.
277 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSectionAccessKey(qlibContext,
278 id.std.uniqueID,
279 QLIB_KID__RESTRICTED_ACCESS_SECTION,
280 sect_u32,
281 die_u8,
282 sectCfg[sect_u32].restrictedKey));
283
284 // write configuration
285 QLIB_STATUS_RET_CHECK(QLIB_ConfigDeviceMultiDie(qlibContext, &flashCfg, &dieCfg, cfgLine, diesNum_u8));
286
287 /*-------------------------------------------------------------------------------------------------------
288 Flash loads the above watchdog configuration only at power up, therefore a hard reset is required for
289 the configuration to apply
290 -------------------------------------------------------------------------------------------------------*/
291
292#if QLIB_MAX_DIES_SUPPORTED > 1u
293 // write configuration sets die to 0. return to orig die
294 QLIB_STATUS_RET_CHECK(QLIB_SetActiveDie(qlibContext, die_u8));
295#endif // QLIB_MAX_DIES_SUPPORTED > 1u
296
297 return QLIB_STATUS__OK;
298}

300QLIB_STATUS_T QLIB_SAMPLE_WatchDog_field_flow(QLIB_CONTEXT_T* qlibContext)
301{
302 QLIB_STATUS_T status = QLIB_STATUS__OK;
303 uint32_t section_u32 = 1u;
304 CONST_KEY_P_T key;
305 QLIB_ID_T id;
306 uint8_t die_u8 = 0u;
307 uint32_t time_u32;
308 bool expired_b;
309
310 /*-------------------------------------------------------------------------------------------------------
311 obtain access key and load the key to library
312 If secured WD is used it is associated with the section number and we must have correct full access key to that section_u32.
313 -------------------------------------------------------------------------------------------------------*/
314 QLIB_STATUS_RET_CHECK(QLIB_GetId(qlibContext, &id));
315#if QLIB_MAX_DIES_SUPPORTED > 1u
316 QLIB_STATUS_RET_CHECK(QLIB_GetActiveDie(qlibContext, &die_u8));
317#endif // QLIB_MAX_DIES_SUPPORTED > 1u
318 QLIB_STATUS_RET_CHECK(
319 QLIB_SAMPLE_GetSectionKeyPtr(qlibContext, id.std.uniqueID, QLIB_KID__FULL_ACCESS_SECTION, section_u32, die_u8, &key));
320 QLIB_STATUS_RET_CHECK(QLIB_LoadKey(qlibContext, section_u32, key, (bool)true));
321
322 /*-------------------------------------------------------------------------------------------------------
323 We configure WD to signal every 4 minutes (approximately) . Now its counter starts to count up, and if not
324 "touched", then after 4 minutes the reset bit will be raised.
325 -------------------------------------------------------------------------------------------------------*/
326 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_WatchDogConfig(qlibContext,
327 (bool)true,
328 (bool)true,
329 QLIB_AWDT_TH_4_MINUTES,
330 section_u32),
331 status,
332 remove_key);
333
334 /*-------------------------------------------------------------------------------------------------------
335 Check whether the WD counter is expired
336 -------------------------------------------------------------------------------------------------------*/
337 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_WatchDogGet(qlibContext, (bool)true, section_u32, &time_u32, &expired_b),
338 status,
339 remove_key);
340
341 /*-------------------------------------------------------------------------------------------------------
342 Touching the WD will zero the counter and WD will bark only after 4 minutes again...
343 Configure this in timer loop to continuously pet the WD.
344 -------------------------------------------------------------------------------------------------------*/
345 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_WatchDogTouch(qlibContext, (bool)true, section_u32), status, remove_key);
346
347 /*-------------------------------------------------------------------------------------------------------
348 Disable WD
349 -------------------------------------------------------------------------------------------------------*/
350 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_WatchDogConfig(qlibContext,
351 (bool)false,
352 (bool)true,
353 QLIB_AWDT_TH_4_MINUTES,
354 section_u32),
355 status,
356 remove_key);
357
358 /*-------------------------------------------------------------------------------------------------------
359 Clean the key reference from QLIB. optional
360 -------------------------------------------------------------------------------------------------------*/
361 QLIB_STATUS_RET_CHECK(QLIB_RemoveKey(qlibContext, section_u32, (bool)true));
362
363 return QLIB_STATUS__OK;
364
365remove_key:
366 (void)QLIB_RemoveKey(qlibContext, section_u32, (bool)true);
367 return status;
368}
