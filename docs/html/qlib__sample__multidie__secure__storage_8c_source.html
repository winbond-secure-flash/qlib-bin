37QLIB_STATUS_T QLIB_SAMPLE_MultiDieSecureStorage(void* userData_pv, QLIB_SAMPLE_SetFetchCmd_CB setFetchCmd)
38{
39 QLIB_CONTEXT_T qlibContext;
40 QLIB_STATUS_T status = QLIB_STATUS__OK;
41 QLIB_BUS_FORMAT_T busFormat = QLIB_BUS_FORMAT(QLIB_BUS_MODE_1_1_1, (bool)false);
42
43 /*-------------------------------------------------------------------------------------------------------
44 Init QLIB - needed when running QLIB either on a device or on a remote server.
45 -------------------------------------------------------------------------------------------------------*/
46 QLIB_STATUS_RET_CHECK(QLIB_InitLib(&qlibContext));
47
48 /*-------------------------------------------------------------------------------------------------------
49 Set the user data into Qlib's context
50 -------------------------------------------------------------------------------------------------------*/
51 QLIB_SetUserData(&qlibContext, userData_pv);
52
53 /*-------------------------------------------------------------------------------------------------------
54 Take the ownership of flash communication channel (it belongs to local MCU or to remote server, exclusively)
55 -------------------------------------------------------------------------------------------------------*/
56 QLIB_STATUS_RET_CHECK(QLIB_Connect(&qlibContext));
57
58 /*-------------------------------------------------------------------------------------------------------
59 Init Flash Device - not needed when using QLIB on a remote server
60 -------------------------------------------------------------------------------------------------------*/
61 QLIB_STATUS_RET_CHECK_GOTO(QLIB_InitDevice(&qlibContext, busFormat), status, disconnect);
62
63 /*-------------------------------------------------------------------------------------------------------
64 Run sample code - configuration for sample
65 -------------------------------------------------------------------------------------------------------*/
66 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_MultiDieSecureStorage_manuf_flow(&qlibContext), status, disconnect);
67
68 /*-------------------------------------------------------------------------------------------------------
69 Run sample flow
70 -------------------------------------------------------------------------------------------------------*/
71 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_MultiDieSecureStorage_field_flow(&qlibContext, setFetchCmd), status, disconnect);
72
73 /*-------------------------------------------------------------------------------------------------------
74 Release the ownership of flash communication channel (it belongs to local MCU or to remote server, exclusively)
75 -------------------------------------------------------------------------------------------------------*/
76 QLIB_STATUS_RET_CHECK(QLIB_Disconnect(&qlibContext));
77
78 return QLIB_STATUS__OK;
79
80disconnect:
81 (void)QLIB_Disconnect(&qlibContext);
82 return status;
83}

99QLIB_STATUS_T QLIB_SAMPLE_MultiDieSecureStorage_field_flow(QLIB_CONTEXT_T* qlibContext, QLIB_SAMPLE_SetFetchCmd_CB setFetchCmd)
100{
101 uint32_t dieId_u32;
102 QLIB_STATUS_T status = QLIB_STATUS__OK;
103 QLIB_STD_ADDR_MODE_T addrMode = QLIB_STD_ADDR_MODE__3_BYTE; // Initial value for current sample
104 QLIB_BUS_FORMAT_T busFormat = QLIB_BUS_FORMAT(QLIB_BUS_MODE_1_1_1, (bool)false); // Initial value for current sample
105 uint32_t dummyCycles_u32 = FLASH_CR_DUMMY_DEFAULT; // Initial value for current sample
106
107 QLIB_ASSERT_RET((dummyCycles_u32 == FLASH_CR_DUMMY_DEFAULT) ||
108 (QLIB_FeatureSupported(qlibContext, W77Q_FEATURE_FAST_READ_DUMMY_CONFIG) != 0u),
109 QLIB_STATUS__INVALID_PARAMETER);
110
111 /*-------------------------------------------------------------------------------------------------------
112 Set SPI interface
113 In case XIP is used, the SPI interface parameters should be set in QLIB, flash and in core simultaneously,
114 to allow both proper QLIB operation and command fetch operations from core.
115 Bus mode and dummy cycles are unchanged
116 -------------------------------------------------------------------------------------------------------*/
117 if (addrMode != QLIB_STD_ADDR_MODE__4_BYTE)
118 {
119 QLIB_ASSERT_RET((QLIB_FeatureSupported(qlibContext, W77Q_FEATURE_4_BYTES_ADDRESS_MODE) != 0u),
120 QLIB_STATUS__INVALID_PARAMETER);
121 QLIB_STATUS_RET_CHECK(
122 QLIB_SAMPLE_SetSPI(qlibContext, busFormat, dummyCycles_u32, QLIB_STD_ADDR_MODE__4_BYTE, setFetchCmd));
123 }
124
125 for (dieId_u32 = 0u; dieId_u32 < QLIB_GET_NUM_OF_DIES(qlibContext); dieId_u32++)
126 {
127#if QLIB_MAX_DIES_SUPPORTED > 1u
128 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SetActiveDie(qlibContext, dieId_u32), status, recover_spi);
129#endif // QLIB_MAX_DIES_SUPPORTED > 1u
130
131 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_SecureStorage_field_flow(qlibContext), status, recover_spi);
132 }
133
134 /*-------------------------------------------------------------------------------------------------------
135 Restore original SPI parameters (3-byte address mode) to allow proper XIP operation for subsequent code
136 -------------------------------------------------------------------------------------------------------*/
137 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_SetSPI(qlibContext, busFormat, dummyCycles_u32, addrMode, setFetchCmd));
138
139 return QLIB_STATUS__OK;
140
141recover_spi:
142 (void)QLIB_SAMPLE_SetSPI(qlibContext, busFormat, dummyCycles_u32, addrMode, setFetchCmd);
143 return status;
144}
