41QLIB_STATUS_T QLIB_SAMPLE_SOC_flow(QLIB_CONTEXT_T* qlibContext, uint32_t* oemSectionId_pu32, KEY_T sectionProvisionKey)
42{
43 QLIB_ID_T id;
44 QLIB_FLASH_CONFIG_T flashCfg = {0u};
45 QLIB_DIE_CONFIG_T dieCfg = {0u};
46 QLIB_SECTIONS_CONF_TABLE_T cfgLine[QLIB_MAX_DIES_SUPPORTED];
47 QLIB_EXTENDED_SECTION_CONF_T* sectCfg_ps;
48 uint8_t die_u8 = 0u;
49 uint8_t diesNum_u8 = QLIB_GET_NUM_OF_DIES(qlibContext);
50
51 QLIB_ASSERT_RET(qlibContext != NULL, QLIB_STATUS__INVALID_PARAMETER);
52 QLIB_ASSERT_RET(sectionProvisionKey != NULL, QLIB_STATUS__INVALID_PARAMETER);
53 QLIB_ASSERT_RET(oemSectionId_pu32 != NULL, QLIB_STATUS__INVALID_PARAMETER);
54
55 /*-------------------------------------------------------------------------------------------------------
56 Get flash IDs - used for key derivation
57 -------------------------------------------------------------------------------------------------------*/
58 QLIB_STATUS_RET_CHECK(QLIB_GetId(qlibContext, &id));
59
60 /*-------------------------------------------------------------------------------------------------------
61 Read current device configuration (read-modify-write)
62 -------------------------------------------------------------------------------------------------------*/
63 QLIB_STATUS_RET_CHECK(QLIB_GetDeviceConfigMultiDie(qlibContext, &flashCfg, &dieCfg, cfgLine, diesNum_u8));
64
65 /*-------------------------------------------------------------------------------------------------------
66 Set the pre-provisioned (current) master key so the device accepts the new configuration
67 -------------------------------------------------------------------------------------------------------*/
68 QLIB_STATUS_RET_CHECK(
69 QLIB_SAMPLE_GetDeviceKey(qlibContext, id.sec.wid, QLIB_KID__DEVICE_KEY_PRE_PROVISIONING, dieCfg.preProvisionedMasterKey));
70
71 /*-------------------------------------------------------------------------------------------------------
72 Set the new device master key
73 -------------------------------------------------------------------------------------------------------*/
74 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetDeviceKey(qlibContext, id.sec.wid, QLIB_KID__DEVICE_MASTER, dieCfg.deviceMasterKey));
75
76#if QLIB_MAX_DIES_SUPPORTED > 1u
77 QLIB_STATUS_RET_CHECK(QLIB_SetActiveDie(qlibContext, die_u8));
78#endif // QLIB_MAX_DIES_SUPPORTED > 1u
79
80 sectCfg_ps = cfgLine[die_u8].sectionConfigTable;
81
82 /*-------------------------------------------------------------------------------------------------------
83 Configure section 0: set base address, size, policy and access keys.
84 Section 0 is the SoC-owned firmware section (rollback protected, plain-read allowed).
85 -------------------------------------------------------------------------------------------------------*/
86 sectCfg_ps[QLIB_SAMPLE_SOC_SECTION_ID].baseAddr = 0u;
87 sectCfg_ps[QLIB_SAMPLE_SOC_SECTION_ID].size = QLIB_SAMPLE_SOC_SECTION_SIZE;
88
89 sectCfg_ps[QLIB_SAMPLE_SOC_SECTION_ID].sectionConfig.policy.plainAccessReadEnable = 1u;
90 sectCfg_ps[QLIB_SAMPLE_SOC_SECTION_ID].sectionConfig.policy.plainAccessWriteEnable = 0u;
91 sectCfg_ps[QLIB_SAMPLE_SOC_SECTION_ID].sectionConfig.policy.authPlainAccess = 0u;
92 sectCfg_ps[QLIB_SAMPLE_SOC_SECTION_ID].sectionConfig.policy.checksumIntegrity = 0u;
93 sectCfg_ps[QLIB_SAMPLE_SOC_SECTION_ID].sectionConfig.policy.digestIntegrity = 0u;
94 sectCfg_ps[QLIB_SAMPLE_SOC_SECTION_ID].sectionConfig.policy.digestIntegrityOnAccess = 0u;
95 sectCfg_ps[QLIB_SAMPLE_SOC_SECTION_ID].sectionConfig.policy.rollbackProt = 1u;
96 sectCfg_ps[QLIB_SAMPLE_SOC_SECTION_ID].sectionConfig.policy.slog = 0u;
97 sectCfg_ps[QLIB_SAMPLE_SOC_SECTION_ID].sectionConfig.policy.writeProt = 0u;
98
99 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSectionAccessKey(qlibContext,
100 id.sec.wid,
101 QLIB_KID__FULL_ACCESS_SECTION,
102 QLIB_SAMPLE_SOC_SECTION_ID,
103 die_u8,
104 sectCfg_ps[QLIB_SAMPLE_SOC_SECTION_ID].fullAccessKey));
105
106 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSectionAccessKey(qlibContext,
107 id.sec.wid,
108 QLIB_KID__RESTRICTED_ACCESS_SECTION,
109 QLIB_SAMPLE_SOC_SECTION_ID,
110 die_u8,
111 sectCfg_ps[QLIB_SAMPLE_SOC_SECTION_ID].restrictedKey));
112
113 /*-------------------------------------------------------------------------------------------------------
114 Configure OEM section : set base address and size only.
115 Keys and policy for OEM section are provisioned by the OEM using the provisioning key.
116 -------------------------------------------------------------------------------------------------------*/
117 sectCfg_ps[QLIB_SAMPLE_OEM_SECTION_ID].baseAddr = QLIB_SAMPLE_SOC_SECTION_SIZE;
118 sectCfg_ps[QLIB_SAMPLE_OEM_SECTION_ID].size = QLIB_SAMPLE_OEM_SECTION_SIZE;
119
120 /*-------------------------------------------------------------------------------------------------------
121 Set address size to suit the plain section size
122 -------------------------------------------------------------------------------------------------------*/
123 flashCfg.stdAddrSize.addrLen_u8 = 22u;
124
125 if (QLIB_FeatureSupported(qlibContext, W77Q_FEATURE_DEVCFG_LOCK))
126 {
127 /*---------------------------------------------------------------------------------------------------
128 Lock the section configuration
129 ---------------------------------------------------------------------------------------------------*/
130 dieCfg.devcfgLock = (bool)true;
131 }
132
133 /*-------------------------------------------------------------------------------------------------------
134 Write the complete device configuration to flash
135 -------------------------------------------------------------------------------------------------------*/
136 QLIB_STATUS_RET_CHECK(QLIB_ConfigDeviceMultiDie(qlibContext, &flashCfg, &dieCfg, cfgLine, diesNum_u8));
137
138 /*-------------------------------------------------------------------------------------------------------
139 Derive the OEM section provisioning key from Kd and return it to the caller for secure transfer to OEM.
140 Also output the OEM section ID so the caller knows which section to provision.
141 -------------------------------------------------------------------------------------------------------*/
142 QLIB_STATUS_RET_CHECK(QLIB_GetSectionProvisionKey(QLIB_SAMPLE_OEM_SECTION_ID, dieCfg.deviceMasterKey, sectionProvisionKey));
143
144 *oemSectionId_pu32 = QLIB_SAMPLE_OEM_SECTION_ID;
145
146 return QLIB_STATUS__OK;
147}

149QLIB_STATUS_T QLIB_SAMPLE_OEM_flow(QLIB_CONTEXT_T* qlibContext, uint32_t sectionId_u32, const KEY_T sectionProvisionKey)
150{
151 QLIB_ID_T id;
152 QLIB_DIE_CONFIG_T dieCfg;
153 QLIB_FLASH_CONFIG_T flashCfg;
154 QLIB_SECTION_CONF_T sectionConf = {0u};
155 QLIB_STATUS_T status = QLIB_STATUS__OK;
156 KEY_T sectionFullAccessKey;
157 KEY_T sectionRestrictedKey;
158#ifndef EXCLUDE_LMS
159 LMS_PUB_KEY_T sectionLmsKey;
160 const uint8_t* pLmsKey_pu8 = QLIB_FeatureSupported(qlibContext, W77Q_FEATURE_LMS) == 1u ? sectionLmsKey : NULL;
161#else
162 const uint8_t* pLmsKey_pu8 = NULL;
163#endif // EXCLUDE_LMS
164 uint8_t die_u8 = 0u;
165
166 QLIB_ASSERT_RET(qlibContext != NULL, QLIB_STATUS__INVALID_PARAMETER);
167 QLIB_ASSERT_RET(sectionProvisionKey != NULL, QLIB_STATUS__INVALID_PARAMETER);
168
169 /*-------------------------------------------------------------------------------------------------------
170 Get flash IDs for key derivation
171 -------------------------------------------------------------------------------------------------------*/
172 QLIB_STATUS_RET_CHECK(QLIB_GetId(qlibContext, &id));
173
174 /*-------------------------------------------------------------------------------------------------------
175 Derive the OEM section access keys (full and restricted)
176 -------------------------------------------------------------------------------------------------------*/
177 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSectionAccessKey(qlibContext,
178 id.sec.wid,
179 QLIB_KID__FULL_ACCESS_SECTION,
180 sectionId_u32,
181 die_u8,
182 sectionFullAccessKey));
183
184 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSectionAccessKey(qlibContext,
185 id.sec.wid,
186 QLIB_KID__RESTRICTED_ACCESS_SECTION,
187 sectionId_u32,
188 die_u8,
189 sectionRestrictedKey));
190
191#ifndef EXCLUDE_LMS
192 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSectionLmsKey(qlibContext, id.sec.wid, sectionId_u32, die_u8, sectionLmsKey));
193#endif // EXCLUDE_LMS
194
195 /*-------------------------------------------------------------------------------------------------------
196 Provision OEM section keys using the provisioning key received from SoC
197 -------------------------------------------------------------------------------------------------------*/
198 QLIB_STATUS_RET_CHECK(QLIB_SetSectionKeys(qlibContext,
199 sectionId_u32,
200 sectionProvisionKey,
201 sectionFullAccessKey,
202 sectionRestrictedKey,
203 pLmsKey_pu8));
204
205 /*-------------------------------------------------------------------------------------------------------
206 Load the full access key and open a FULL session to configure the section policy
207 -------------------------------------------------------------------------------------------------------*/
208 QLIB_STATUS_RET_CHECK(QLIB_LoadKey(qlibContext, sectionId_u32, sectionFullAccessKey, (bool)true));
209
210 QLIB_STATUS_RET_CHECK_GOTO(QLIB_OpenSession(qlibContext, sectionId_u32, QLIB_SESSION_ACCESS_CONFIG_ONLY), status, remove_key);
211
212 /*-------------------------------------------------------------------------------------------------------
213 Set the OEM section policy
214 -------------------------------------------------------------------------------------------------------*/
215 sectionConf.policy.plainAccessReadEnable = 0u;
216 sectionConf.policy.plainAccessWriteEnable = 0u;
217 sectionConf.policy.authPlainAccess = 0u;
218 sectionConf.policy.checksumIntegrity = 0u;
219 sectionConf.policy.digestIntegrity = 0u;
220 sectionConf.policy.digestIntegrityOnAccess = 0u;
221 sectionConf.policy.rollbackProt = 0u;
222 sectionConf.policy.slog = 0u;
223 sectionConf.policy.writeProt = 0u;
224
225 /*-------------------------------------------------------------------------------------------------------
226 Keep the session open after configuration so the lock check ahead is done with authenticated command
227 -------------------------------------------------------------------------------------------------------*/
228 sectionConf.postActions.reload = 1u;
229
230 QLIB_STATUS_RET_CHECK_GOTO(QLIB_ConfigDeviceSection(qlibContext, sectionId_u32, &sectionConf), status, close_session);
231
232 if (QLIB_FeatureSupported(qlibContext, W77Q_FEATURE_DEVCFG_LOCK))
233 {
234 /*---------------------------------------------------------------------------------------------------
235 Verify DEVCFG.CFG_LOCK is locked so the SoC cannot modify section mapping.
236 GMC is read with authentication since a session is open.
237 ---------------------------------------------------------------------------------------------------*/
238 QLIB_STATUS_RET_CHECK_GOTO(QLIB_GetDeviceConfigMultiDie(qlibContext,
239 &flashCfg,
240 &dieCfg,
241 NULL,
242 QLIB_GET_NUM_OF_DIES(qlibContext)),
243 status,
244 close_session);
245
246 QLIB_ASSERT_WITH_ERROR_GOTO(dieCfg.devcfgLock == (bool)true,
247 QLIB_STATUS__SYSTEM_IN_INCORRECT_STATE,
248 status,
249 close_session);
250 }
251 QLIB_STATUS_RET_CHECK_GOTO(QLIB_CloseSession(qlibContext, sectionId_u32), status, remove_key);
252
253 QLIB_STATUS_RET_CHECK(QLIB_RemoveKey(qlibContext, sectionId_u32, (bool)true));
254
255 return QLIB_STATUS__OK;
256
257close_session:
258 (void)QLIB_CloseSession(qlibContext, sectionId_u32);
259remove_key:
260 (void)QLIB_RemoveKey(qlibContext, sectionId_u32, (bool)true);
261
262 return status;
263}

265QLIB_STATUS_T QLIB_SAMPLE_SoC_OEM_SectionProvision(void* userData_pv)
266{
267 QLIB_CONTEXT_T qlibContext;
268 QLIB_STATUS_T status = QLIB_STATUS__OK;
269 QLIB_BUS_FORMAT_T busFormat = QLIB_BUS_FORMAT(QLIB_BUS_MODE_1_1_1, (bool)false);
270 KEY_T sectionProvisionKey;
271 uint32_t oemSectionId_u32 = 0u;
272
273 /*-------------------------------------------------------------------------------------------------------
274 Init QLIB - needed when running QLIB either on a device or on a remote server.
275 -------------------------------------------------------------------------------------------------------*/
276 QLIB_STATUS_RET_CHECK(QLIB_InitLib(&qlibContext));
277
278 /*-------------------------------------------------------------------------------------------------------
279 Set the user data into Qlib's context
280 -------------------------------------------------------------------------------------------------------*/
281 QLIB_SetUserData(&qlibContext, userData_pv);
282
283 /*-------------------------------------------------------------------------------------------------------
284 Take the ownership of flash communication channel (it belongs to local MCU or to remote server, exclusively)
285 -------------------------------------------------------------------------------------------------------*/
286 QLIB_STATUS_RET_CHECK(QLIB_Connect(&qlibContext));
287
288 /*-------------------------------------------------------------------------------------------------------
289 Init Flash Device - not needed when using QLIB on a remote server
290 -------------------------------------------------------------------------------------------------------*/
291 QLIB_STATUS_RET_CHECK_GOTO(QLIB_InitDevice(&qlibContext, busFormat), status, disconnect);
292
293 /*-------------------------------------------------------------------------------------------------------
294 SoC flow: configure SoC and OEM sections, set master key, derive section provisioning key and section ID
295 -------------------------------------------------------------------------------------------------------*/
296 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_SOC_flow(&qlibContext, &oemSectionId_u32, sectionProvisionKey), status, disconnect);
297
298 /*-------------------------------------------------------------------------------------------------------
299 OEM flow: receive provisioning key and section ID, set section keys and policy, verify CFG_LOCK
300 -------------------------------------------------------------------------------------------------------*/
301 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_OEM_flow(&qlibContext, oemSectionId_u32, sectionProvisionKey), status, disconnect);
302
303disconnect:
304 (void)QLIB_Disconnect(&qlibContext);
305 return status;
306}
