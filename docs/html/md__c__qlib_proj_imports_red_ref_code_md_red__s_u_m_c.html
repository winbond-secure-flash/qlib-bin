// KeyId is used through LMS cycle as a LMS key pair identifier
LMS_KEY_ID_T keyId_g;
QLIB_STATUS_T hsm_lms_generate_keypair(LMS_KEY_ID_T keyId)
{
ALG_LMS_T lmsAlg;
LMS_ALG_STATUS lmsRet;
lmsRet = lmsInit(&lmsAlg);
ASSERT_MSG_RET(ALG_SUCCESS == lmsRet, QLIB_STATUS__COMMAND_FAIL, "Failed lmsInit with %s\n", LMS_STATUS_TO_STR(lmsRet));
// this is done once for each KID and takes ~10 min on average PC
LMS_GEN_KEY_PAIR_OP_T genKeyParams = {keyId, LMOTS_SHA256_N24_W4_PARAM_SET_ID, LMS_SHA256_M24_H20_PARAM_SET_ID};
lmsRet = lmsGenerateKeyPair(&lmsAlg, &genKeyParams);
ASSERT_MSG_RET(ALG_SUCCESS == lmsRet, QLIB_STATUS__COMMAND_FAIL, "Failed lmsGenerateKeyPair with %s\n", LMS_STATUS_TO_STR(lmsRet));
lmsRet = lmsFinit(&lmsAlg);
ASSERT_MSG_RET(ALG_SUCCESS == lmsRet, QLIB_STATUS__COMMAND_FAIL, "Failed lmsFinit with %s\n", LMS_STATUS_TO_STR(lmsRet));
return QLIB_STATUS__OK;
}
QLIB_STATUS_T hsm_lms_public_get(LMS_KEY_ID_T keyId, LMS_PUB_KEY_T publicKey )
{
ALG_LMS_T lmsAlg;
LMS_ALG_STATUS lmsRet;
lmsRet = lmsInit(&lmsAlg);
ASSERT_MSG_RET(ALG_SUCCESS == lmsRet, QLIB_STATUS__COMMAND_FAIL, "Failed lmsInit with %s\n", LMS_STATUS_TO_STR(lmsRet));
// Retrieve public lms key from the HSM
LMS_GET_PUB_KEY_OP_T getPubKeyParams = {keyId, publicKey, sizeof(LMS_PUB_KEY_T), 0};
lmsRet = lmsGetPublicKey(&lmsAlg, &getPubKeyParams);
ASSERT_MSG_RET(ALG_SUCCESS == lmsRet, QLIB_STATUS__COMMAND_FAIL, "Failed lmsGetPublicKey with %s\n", LMS_STATUS_TO_STR(lmsRet));
// ensure it is of right length
ASSERT_MSG_RET(getPubKeyParams.publicKeyBytesReal == LMS_PUBLIC_KEY_SIZE(24), QLIB_STATUS__COMMAND_FAIL,
"lmsGetPublicKey: read %d bytes instead of %I64d\n", getPubKeyParams.publicKeyBytesReal, LMS_PUBLIC_KEY_SIZE(24));
lmsRet = lmsFinit(&lmsAlg);
ASSERT_MSG_RET(ALG_SUCCESS == lmsRet, QLIB_STATUS__COMMAND_FAIL, "Failed lmsFinit with %s\n", LMS_STATUS_TO_STR(lmsRet));
return QLIB_STATUS__OK;
}
QLIB_STATUS_T hsm_lms_sign_message(LMS_KEY_ID_T keyId, QLIB_LMS_MSG_DATA_STRUCT_T lmsMsg, QLIB_LMS_SIG_BUFFER_T signature )
{
ALG_LMS_T lmsAlg;
LMS_ALG_STATUS lmsRet;
lmsRet = lmsInit(&lmsAlg);
ASSERT_MSG_RET(ALG_SUCCESS == lmsRet, QLIB_STATUS__COMMAND_FAIL, "Failed lmsInit with %s\n", LMS_STATUS_TO_STR(lmsRet));
// sign it
LMS_SIGN_MESSAGE_OP_T signParams = { keyId, lmsMsg, sizeof(QLIB_LMS_MSG_DATA_STRUCT_T), signature, sizeof(QLIB_LMS_SIG_BUFFER_T), 0};
// create signature
lmsRet = lmsSignMessage(&lmsAlg, &signParams);
ASSERT_MSG_RET(ALG_SUCCESS == lmsRet, QLIB_STATUS__COMMAND_FAIL, "Failed lmsSignMessage with %s\n", LMS_STATUS_TO_STR(lmsRet));
ASSERT_MSG_RET( signParams.signatureBytesReal ==
QLIB_LMS_SIGNATURE_SIZE(QLIB_LMS_PARAM_M, QLIB_LMS_PARAM_N, QLIB_LMS_PARAM_P, QLIB_LMS_TREE_HEIGHT),
QLIB_STATUS__COMMAND_FAIL,
"lmsSignMessage: read %d bytes instead of %I64ld\n",
signParams.signatureBytesReal,
QLIB_LMS_SIGNATURE_SIZE(QLIB_LMS_PARAM_M, QLIB_LMS_PARAM_N, QLIB_LMS_PARAM_P, QLIB_LMS_TREE_HEIGHT));
lmsRet = lmsFinit(&lmsAlg);
ASSERT_MSG_RET(ALG_SUCCESS == lmsRet, QLIB_STATUS__COMMAND_FAIL, "Failed lmsFinit with %s\n", LMS_STATUS_TO_STR(lmsRet));
return QLIB_STATUS__OK;
}
QLIB_STATUS_T hsm_lms_verify(LMS_PUB_KEY_T pub, QLIB_LMS_MSG_DATA_STRUCT_T msg, QLIB_LMS_SIG_BUFFER_T sign)
{
ALG_LMS_T lmsAlg;
LMS_ALG_STATUS lmsRet;
lmsRet = lmsInit(&lmsAlg);
ASSERT_MSG_RET(ALG_SUCCESS == lmsRet, QLIB_STATUS__COMMAND_FAIL, "Failed lmsInit with %s\n", LMS_STATUS_TO_STR(lmsRet));
LMS_VERIFY_SIGNATURE_OP_T verifyParams = { pub, sizeof(LMS_PUB_KEY_T),
msg, sizeof(QLIB_LMS_MSG_DATA_STRUCT_T),
sign, sizeof(QLIB_LMS_SIG_BUFFER_T) };
lmsRet = lmsVerifySignature(&lmsAlg, &verifyParams);
ASSERT_MSG_RET(ALG_SUCCESS == lmsRet, QLIB_STATUS__COMMAND_FAIL, "Failed lmsVerifySignature with %s\n", LMS_STATUS_TO_STR(lmsRet));
lmsRet = lmsFinit(&lmsAlg);
ASSERT_MSG_RET(ALG_SUCCESS == lmsRet, QLIB_STATUS__COMMAND_FAIL, "Failed lmsFinit with %s\n", LMS_STATUS_TO_STR(lmsRet));
return QLIB_STATUS__OK;
}

QLIB_STATUS_T red_SUM_section_config( QLIB_CONTEXT_T* ctx,
QLIB_ID_T* id,
uint32_t dieNum,
uint32_t sec,
QLIB_SECTION_CONF_T* cfgNew,
uint8_t* digestExt_pu8,
bool isSymm)
{
// to configure sectionm we do read, modify, write configuration
if (isSymm)
{
KEY_T keyF;
// obtain full, access key, load its pointer to library, open session
QLIB_STATUS_RET_CHECK(fwGetSectionKey(
ctx, id, QLIB_KID__FULL_ACCESS_SECTION, sec, dieNum, keyF, sizeof (KEY_T)));
// at this point we want to swap between sections
QLIB_STATUS_RET_CHECK(QLIB_LoadKey(ctx, sec, keyF, true));
QLIB_STATUS_RET_CHECK(QLIB_OpenSession(ctx, sec, QLIB_SESSION_ACCESS_FULL));
QLIB_STATUS_RET_CHECK(QLIB_ConfigDeviceSection(ctx, sec, cfgNew));
if (cfgNew->postActions.reload) // session is kept open by the chip
{
QLIB_STATUS_RET_CHECK(QLIB_CloseSession(ctx, sec));
}
QLIB_STATUS_RET_CHECK(QLIB_RemoveKey(ctx, sec, true));
}
#ifndef EXCLUDE_LMS
else
{
bool isProvisioned;
// for asymmetric lms cypher LMS public key should be configured
QLIB_STATUS_RET_CHECK(QLIB_IsKeyProvisioned(ctx, QLIB_KID__SECTION_LMS, sec, &isProvisioned));
QLIB_ASSERT_RET(isProvisioned, QLIB_STATUS__COMMAND_FAIL);
// lms public is inside so we have to encode command sign it and send to chip
QLIB_LMS_MSG_DATA_STRUCT_T lmsMsg;
// encode lms message for new section configuration
QLIB_STATUS_RET_CHECK(QLIB_UTILS_LMS_CreateSetScrMsg(sec, cfgNew, digestExt_pu8, lmsMsg));
// calculate signature
QLIB_LMS_SIG_BUFFER_T signature;
QLIB_STATUS_RET_CHECK(hsm_lms_sign_message(keyId_g, lmsMsg, signature));
// sending to chip command and its signature
QLIB_STATUS_RET_CHECK(QLIB_SendLMSCommand(ctx, lmsMsg, keyId_g, signature, sec));
}
#endif
return QLIB_STATUS__OK;
}
QLIB_STATUS_T red_SUM_manuf_flow(QLIB_CONTEXT_T* ctx, bool isSymm)
{
QLIB_HW_VER_T hwVer;
QLIB_ID_T id;
uint32_t sec = 0, die = 0;
QLIB_FLASH_CONFIG_T flashCfg;
QLIB_DIE_CONFIG_T dieCfg;
QLIB_SECTIONS_CONF_TABLE_T cfgLine[QLIB_MAX_DIES_SUPPORTED];
uint8_t numOfDies = QLIB_GET_NUM_OF_DIES(ctx);
// See what chip is connected and its unique ids. We might need this info to obtain keys for this
// chips according to formula or via DB see fwGetSectionKey() function
QLIB_STATUS_RET_CHECK(QLIB_GetHWVersion(ctx, &hwVer));
QLIB_STATUS_RET_CHECK(QLIB_GetId(ctx, &id));
// We do read / modify /write configuration since we want to change only what is needed
// read configuration
QLIB_STATUS_RET_CHECK(QLIB_GetDeviceConfigMultiDie( ctx, &flashCfg, &dieCfg, cfgLine, numOfDies));
// modify configuration
// set pre provision master key if needed
QLIB_STATUS_RET_CHECK(fwGetMasterKey(ctx, &id, QLIB_KID__DEVICE_KEY_PRE_PROVISIONING, dieCfg.preProvisionedMasterKey));
// set new master key
QLIB_STATUS_RET_CHECK(fwGetMasterKey(ctx, &id, QLIB_KID__DEVICE_MASTER, dieCfg.deviceMasterKey));
// setting layout and policy of the section 0
// this section plain read enabled and plain write enabled
QLIB_EXTENDED_SECTION_CONF_T * sectCfg = cfgLine[die].sectionConfigTable;
sectCfg[sec].baseAddr = 0;
sectCfg[sec].size = RB_SECT_SIZE;
// note that policy will be config
sectCfg[sec].sectionConfig.policy.rollbackProt = 1;
sectCfg[sec].sectionConfig.policy.plainAccessReadEnable = 1;
sectCfg[sec].sectionConfig.policy.plainAccessWriteEnable = 1;
if (isSymm)
{
// for symmetric cypher configuration we only need to configure full access key. The policy
// will be configured with this key as well
QLIB_STATUS_RET_CHECK(fwGetSectionKey(
ctx, &id, QLIB_KID__FULL_ACCESS_SECTION, sec, die, sectCfg[sec].fullAccessKey, sizeof(KEY_T)));
}
#ifndef EXCLUDE_LMS
else
{
// for asymmetric lms cypher we need to configure LMS public key. Note that we will need to configure
// section policy separately since qlib can not provide lms signature for policy configuration.
bool isProvisioned;
QLIB_STATUS_RET_CHECK(QLIB_IsKeyProvisioned(ctx, QLIB_KID__SECTION_LMS, sec, &isProvisioned));
// lms public key can be already inside, we assume that this function was already ran before
if (false == isProvisioned)
{
// lms public key is obtained from hsm
LMS_PUB_KEY_T publicKey;
// we use kid that was generated once when we created LMS key pair
QLIB_STATUS_RET_CHECK(hsm_lms_public_get(keyId_g, publicKey));
// copy to chip configuration data structure
(void)memcpy(sectCfg[sec].lmsKey, publicKey, sizeof(LMS_PUB_KEY_T));
}
}
#endif
// write configuration
QLIB_STATUS_RET_CHECK(QLIB_ConfigDeviceMultiDie(ctx, &flashCfg, &dieCfg, cfgLine, numOfDies));
// note that in case we work with ASYM cipher only we need to configure rollback policy with ASYM as well
if (false == isSymm)
{
QLIB_STATUS_RET_CHECK(red_SUM_section_config(ctx, &id, die, sec, &sectCfg[sec].sectionConfig, NULL, false));
}
return QLIB_STATUS__OK;
}
QLIB_STATUS_T red_SUM_field_flow( QLIB_CONTEXT_T* ctx, bool isSymm)
{
QLIB_HW_VER_T hwVer;
QLIB_ID_T id;
uint32_t sec = 0, die = 0;
uint8_t numOfDies = QLIB_GET_NUM_OF_DIES(ctx);
uint8_t fw_curr[FW_SIZE], fw_next[FW_SIZE];
// See what chip is connected and its unique ids. We might need this info to obtain keys for this
// chips according to formula or via DB see fwGetSectionKey() function
QLIB_STATUS_RET_CHECK(QLIB_GetHWVersion(ctx, &hwVer));
QLIB_STATUS_RET_CHECK(QLIB_GetId(ctx, &id));
// current fw is in active part of the boot section, aka 0 -> size/2, and it is read only
QLIB_STATUS_RET_CHECK(QLIB_Read(ctx, fw_curr, sec, 0, FW_SIZE, false, false));
// ensure write to active part of the rollback section will fail
QLIB_ASSERT_RET(QLIB_STATUS__OK != QLIB_Write(ctx, fw_curr, sec, 0, FW_SIZE, false),
QLIB_STATUS__TEST_FAIL);
// next fw is located in inactive part of the boot section, aka size/2 -> size, and it is plain rw
QLIB_STATUS_RET_CHECK(QLIB_Write(ctx, fw_next, sec, RB_SECT_SIZE / 2, FW_SIZE, false));
// swaping the halves of the sections is actually new configuration. Let's prepare it.
QLIB_SECTION_CONF_T cfgNew = {0};
QLIB_STATUS_RET_CHECK(QLIB_GetDeviceSectionConfig(ctx, sec, NULL, NULL, &cfgNew));
// at this example we do not do reset since this will trigger device reboot.
cfgNew.postActions.swap = 1;
cfgNew.postActions.reload = 1;
cfgNew.postActions.reset = 0;
// we increment section version on each update, later to utilize rollback protection
cfgNew.version++;
// now we issue command that will atomically switch rollback section parts
QLIB_STATUS_RET_CHECK(red_SUM_section_config(ctx, &id, die, sec, &cfgNew, NULL, isSymm));
// now let's check if the sections swapped and current fw is updated
QLIB_STATUS_RET_CHECK(QLIB_Read(ctx, fw_curr, sec, 0, FW_SIZE, false, false));
QLIB_ASSERT_RET(0 == memcmp(fw_curr, fw_next, FW_SIZE), QLIB_STATUS__TEST_FAIL);
// now let's see that previous version will not be accepted despite correct crypto
// this is rollback protection feature, aka even if hacker controls correct update flow
// flash will not accept if version is less then current one
cfgNew.version--;
QLIB_ASSERT_RET(QLIB_STATUS__OK != red_SUM_section_config(ctx, &id, die, sec, &cfgNew,NULL, isSymm),
QLIB_STATUS__TEST_FAIL);
return QLIB_STATUS__OK;
}
