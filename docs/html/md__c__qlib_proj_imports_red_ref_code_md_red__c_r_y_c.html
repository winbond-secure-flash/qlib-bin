// function which checks INC functionality of the MC, which requires reading and writing
QLIB_STATUS_T w77q_mc_check_inc(W77Q_RPMC_T *cnt, KEY_T key, uint32_t *val)
{
uint32_t counterVal = 0;
uint32_t counterValNext = 0;
// using the counter
ASSERT_MSG_RET(DEFS_STATUS_OK == W77Q_RPMC_Init(cnt, key), QLIB_STATUS__COMMAND_FAIL, "FMC_MC_Init fail \n");
ASSERT_MSG_RET(DEFS_STATUS_OK == W77Q_RPMC_Get(cnt, &counterVal, key), QLIB_STATUS__COMMAND_FAIL, "FMC_MC_Get fail \n");
ASSERT_MSG_RET(DEFS_STATUS_OK == W77Q_RPMC_Inc(cnt, key), QLIB_STATUS__COMMAND_FAIL, "FMC_MC_Inc fail \n");
ASSERT_MSG_RET(DEFS_STATUS_OK == W77Q_RPMC_Get(cnt, &counterValNext, key), QLIB_STATUS__COMMAND_FAIL, "FMC_MC_Get fail \n");
ASSERT_MSG_RET(counterVal + 1 == counterValNext, QLIB_STATUS__COMMAND_FAIL, "MC counters' check fail \n");
if (val)
{
*val = counterValNext;
}
return QLIB_STATUS__OK;
}
// function which checks GET functionality of the MC, which requires reading only
QLIB_STATUS_T w77q_mc_check_get(W77Q_RPMC_T *cnt, KEY_T key, uint32_t *val)
{
uint32_t counterVal = 0;
// using the counter
ASSERT_MSG_RET(DEFS_STATUS_OK == W77Q_RPMC_Init(cnt, key), QLIB_STATUS__COMMAND_FAIL, "FMC_MC_Init fail \n");
ASSERT_MSG_RET(DEFS_STATUS_OK == W77Q_RPMC_Get(cnt, &counterVal, key), QLIB_STATUS__COMMAND_FAIL, "FMC_MC_Get fail \n");
if (val)
{
*val = counterVal;
}
return QLIB_STATUS__OK;
}

// flash configuration function which configures 4 sections with different security policies and counter types
QLIB_STATUS_T w77q_config_flash(QLIB_CONTEXT_T* ctx)
{
QLIB_ID_T id;
uint32_t sect = 0, die = 0;
QLIB_FLASH_CONFIG_T flashCfg = {0};
QLIB_DIE_CONFIG_T dieCfg = {0};
QLIB_SECTIONS_CONF_TABLE_T cfgLine[QLIB_MAX_DIES_SUPPORTED];
uint8_t diesNum = QLIB_GET_NUM_OF_DIES(ctx);
#ifndef APP_EXCLUDE_GET_ID
// See what chip is connected and its unique ids. We might need this info to obtain keys for this
// chips according to formula or via DB see fwGetSectionKey() function
QLIB_STATUS_RET_CHECK(QLIB_GetId(ctx, &id));
#endif
// We do read / modify /write configuration since we want to change only what is needed
// read configuration
QLIB_STATUS_RET_CHECK(QLIB_GetDeviceConfigMultiDie( ctx, &flashCfg, &dieCfg, cfgLine, diesNum));
QLIB_EXTENDED_SECTION_CONF_T * sectCfg = cfgLine[die].sectionConfigTable;
// modify configuration
// set pre provision master key if needed
QLIB_STATUS_RET_CHECK(fwGetMasterKey(ctx, &id, QLIB_KID__DEVICE_KEY_PRE_PROVISIONING, dieCfg.preProvisionedMasterKey));
// set new master key
QLIB_STATUS_RET_CHECK(fwGetMasterKey(ctx, &id, QLIB_KID__DEVICE_MASTER, dieCfg.deviceMasterKey));
// this section plain read / write, aka counter over this section will be not protected
sect = 0;
sectCfg[sect].baseAddr = 0;
sectCfg[sect].size = SIZE_1MB_;
sectCfg[sect].sectionConfig.policy.plainAccessReadEnable = 1;
sectCfg[sect].sectionConfig.policy.plainAccessWriteEnable = 1;
// full access key set for the section configuration changes or write access
QLIB_STATUS_RET_CHECK( fwGetSectionKey(
ctx, &id, QLIB_KID__FULL_ACCESS_SECTION, sect, die, sectCfg[sect].fullAccessKey, sizeof(KEY_T)));
// this section plain read enabled / plain write disabled, aka protected for write
sect = 1;
sectCfg[sect].baseAddr = SIZE_1MB_;
sectCfg[sect].size = SIZE_1MB_;
sectCfg[sect].sectionConfig.policy.plainAccessReadEnable = 1;
sectCfg[sect].sectionConfig.policy.plainAccessWriteEnable = 0;
// full access key set for the section configuration changes or write access
QLIB_STATUS_RET_CHECK( fwGetSectionKey(
ctx, &id, QLIB_KID__FULL_ACCESS_SECTION, sect, die, sectCfg[sect].fullAccessKey, sizeof(KEY_T)));
// this section plain read / write disabled, aka protected for read/write
sect = 2;
sectCfg[sect].baseAddr = SIZE_2MB_;
sectCfg[sect].size = SIZE_1MB_;
sectCfg[sect].sectionConfig.policy.plainAccessReadEnable = 0;
sectCfg[sect].sectionConfig.policy.plainAccessWriteEnable = 0;
// full access key set for the section configuration changes or write access
QLIB_STATUS_RET_CHECK( fwGetSectionKey(
ctx, &id, QLIB_KID__FULL_ACCESS_SECTION, sect, die, sectCfg[sect].fullAccessKey, sizeof(KEY_T)));
// this section plain read / write disabled, aka protected for read/write
sect = 3;
sectCfg[sect].baseAddr = SIZE_2MB_ + SIZE_1MB_;
sectCfg[sect].size = SIZE_1MB_;
sectCfg[sect].sectionConfig.policy.plainAccessReadEnable = 0;
sectCfg[sect].sectionConfig.policy.plainAccessWriteEnable = 0;
// full access key set for the section configuration changes or write access
QLIB_STATUS_RET_CHECK( fwGetSectionKey(
ctx, &id, QLIB_KID__FULL_ACCESS_SECTION, sect, die, sectCfg[sect].fullAccessKey, sizeof(KEY_T)));
// restricted access key set for read access
QLIB_STATUS_RET_CHECK( fwGetSectionKey(
ctx, &id, QLIB_KID__RESTRICTED_ACCESS_SECTION, sect, die, sectCfg[sect].restrictedKey, sizeof(KEY_T)));
// write configuration
QLIB_STATUS_RET_CHECK(QLIB_ConfigDeviceMultiDie(ctx, &flashCfg, &dieCfg, cfgLine, diesNum));
return QLIB_STATUS__OK;
}
// full manufactire flow for MC over w77q
QLIB_STATUS_T red_CRY_manuf_flow(QLIB_CONTEXT_T* ctx)
{
QLIB_STATUS_T qlib_ret;
KEY_T keyF;
QLIB_ID_T id;
DEFS_STATUS fmc_ret;
uint32_t section = 0;
#ifndef APP_EXCLUDE_GET_ID
// See what chip is connected and its unique ids. We might need this info to obtain keys for this
// chips according to formula or via DB see fwGetSectionKey() function
QLIB_STATUS_RET_CHECK(QLIB_GetId(ctx, &id));
#endif
// in this example we will build four counters with different key requirements. To do that we
// will configure 4 sections over flash with different keys accordingly.
// section# | 0 | 1 | 2 | 3 |
//----------|--------------------------|---------------------------|---------------------------|---------------------------|
// size | 1M | 1M | 1M | 1M |
//----------|--------------------------|---------------------------|---------------------------|---------------------------|
// policy | plain read = 1 | plain read = 1 | plain read = 0 | plain read = 0 |
// | plain write = 1 | plain write = 0 | plain write = 0 | plain write = 0 |
//----------|--------------------------|---------------------------|---------------------------|---------------------------|
// Counter | W77Q_RPMC_TYPE_RN_WN | W77Q_RPMC_TYPE_RN_WF | W77Q_RPMC_TYPE_RF_WF | W77Q_RPMC_TYPE_RR_WF |
qlib_ret = w77q_config_flash(ctx);
ASSERT_MSG_RET(qlib_ret == QLIB_STATUS__OK, qlib_ret, "w77q_config_flash failed with %s\n", STATUS_TO_STR(qlib_ret));
qlib_ret = QLIB_ResetFlash(ctx);
ASSERT_MSG_RET(qlib_ret == QLIB_STATUS__OK, qlib_ret, "QLIB_ResetFlash failed with %s\n", STATUS_TO_STR(qlib_ret));
uint32_t step = FMC_MC_FLASH_SIZE*17;
for ( uint32_t counter_offset = 0; counter_offset < SIZE_1MB_; counter_offset += step )
{
#ifndef APP_EXCLUDE_PLAIN_ACCESS
// ******************************************************************************
// mc0 W77Q_RPMC_TYPE_RN_WN
// ******************************************************************************
section = 0;
W77Q_RPMC_T mc0;
fmc_ret = W77Q_RPMC_Set_Params(&mc0, W77Q_RPMC_TYPE_RN_WN, section, counter_offset, ctx);
ASSERT_MSG_RET(DEFS_STATUS_OK == fmc_ret, QLIB_STATUS__COMMAND_FAIL, "W77Q_RPMC_Set_Params fail:%s \n", DEFS_STATUS_TO_STR(fmc_ret));
// formatting protected counter requires key
//QLIB_STATUS_RET_CHECK(fwGetSectionKey(ctx, &id, QLIB_KID__FULL_ACCESS_SECTION, section, 0, keyF, sizeof(KEY_T)));
fmc_ret = W77Q_RPMC_Format(&mc0, true, NULL);
ASSERT_MSG_RET(DEFS_STATUS_OK == fmc_ret, QLIB_STATUS__COMMAND_FAIL, "W77Q_RPMC_Format mc0 fail: %s\n", DEFS_STATUS_TO_STR(fmc_ret));
// ******************************************************************************
// mc1 W77Q_RPMC_TYPE_RN_WF
// ******************************************************************************
section = 1;
W77Q_RPMC_T mc1;
fmc_ret = W77Q_RPMC_Set_Params(&mc1, W77Q_RPMC_TYPE_RN_WF, section, counter_offset, ctx);
ASSERT_MSG_RET(DEFS_STATUS_OK == fmc_ret, QLIB_STATUS__COMMAND_FAIL, "W77Q_RPMC_Set_Params fail:%s \n", DEFS_STATUS_TO_STR(fmc_ret));
// formatting counter requires write permission
// since counter is secure write section, full access key is needed
QLIB_STATUS_RET_CHECK(fwGetSectionKey(ctx, &id, QLIB_KID__FULL_ACCESS_SECTION, section, 0, keyF, sizeof(KEY_T)));
fmc_ret = W77Q_RPMC_Format(&mc1, true, keyF);
ASSERT_MSG_RET(DEFS_STATUS_OK == fmc_ret, QLIB_STATUS__COMMAND_FAIL, "W77Q_RPMC_Format mc1 fail: %s\n", DEFS_STATUS_TO_STR(fmc_ret));
#endif //#ifndef EXCLUDE_PLAIN_ACCESS
// ******************************************************************************
// mc2 W77Q_RPMC_TYPE_RF_WF
// ******************************************************************************
section = 2;
W77Q_RPMC_T mc2;
fmc_ret = W77Q_RPMC_Set_Params(&mc2, W77Q_RPMC_TYPE_RF_WF, section, counter_offset, ctx);
ASSERT_MSG_RET(DEFS_STATUS_OK == fmc_ret, QLIB_STATUS__COMMAND_FAIL, "W77Q_RPMC_Set_Params fail:%s \n", DEFS_STATUS_TO_STR(fmc_ret));
// formatting counter requires write permission
// since counter is secure write section, full access key is needed
QLIB_STATUS_RET_CHECK(fwGetSectionKey(ctx, &id, QLIB_KID__FULL_ACCESS_SECTION, section, 0, keyF, sizeof(KEY_T)));
fmc_ret = W77Q_RPMC_Format(&mc2, true, keyF);
ASSERT_MSG_RET(DEFS_STATUS_OK == fmc_ret, QLIB_STATUS__COMMAND_FAIL, "W77Q_RPMC_Format mc2 fail: %s\n", DEFS_STATUS_TO_STR(fmc_ret));
// ******************************************************************************
// mc3 W77Q_RPMC_TYPE_RR_WF
// ******************************************************************************
section = 3;
W77Q_RPMC_T mc3;
fmc_ret = W77Q_RPMC_Set_Params(&mc3, W77Q_RPMC_TYPE_RR_WF, section, counter_offset, ctx);
ASSERT_MSG_RET(DEFS_STATUS_OK == fmc_ret, QLIB_STATUS__COMMAND_FAIL, "W77Q_RPMC_Set_Params fail:%s \n", DEFS_STATUS_TO_STR(fmc_ret));
// formatting counter requires write permission
// since counter is secure write section, full access key is needed
QLIB_STATUS_RET_CHECK(fwGetSectionKey(ctx, &id, QLIB_KID__FULL_ACCESS_SECTION, section, 0, keyF, sizeof(KEY_T)));
fmc_ret = W77Q_RPMC_Format(&mc3, true, keyF);
ASSERT_MSG_RET(DEFS_STATUS_OK == fmc_ret, QLIB_STATUS__COMMAND_FAIL, "W77Q_RPMC_Format mc3 fail: %s\n", DEFS_STATUS_TO_STR(fmc_ret));
}
PRINTF("Formatted %d counters each section successfully\n", SIZE_1MB_ / step);
return QLIB_STATUS__OK;
}

// full field flow for MC over w77q
QLIB_STATUS_T red_CRY_field_flow( QLIB_CONTEXT_T* ctx)
{
uint32_t cnt_a, cnt_b, section;
DEFS_STATUS fmc_ret;
QLIB_ID_T id;
KEY_T keyF;
#ifndef APP_EXCLUDE_GET_ID
// See what chip is connected and its unique ids. We might need this info to obtain keys for this
// chips according to formula or via DB see fwGetSectionKey() function
QLIB_STATUS_RET_CHECK(QLIB_GetId(ctx, &id));
#endif
//PRINTF("Testing Monotonic counter #1 ...\n");
uint32_t step = FMC_MC_FLASH_SIZE*17;
for ( uint32_t counter_offset = 0; counter_offset < SIZE_1MB_; counter_offset += step )
{
#ifndef APP_EXCLUDE_PLAIN_ACCESS
// ******************************************************************************
// mc0 is W77Q_RPMC_TYPE_RN_WN so actor works with no keys
// ******************************************************************************
section = 0;
W77Q_RPMC_T mc0;
fmc_ret = W77Q_RPMC_Set_Params(&mc0, W77Q_RPMC_TYPE_RN_WN, section, counter_offset, ctx);
ASSERT_MSG_RET(DEFS_STATUS_OK == fmc_ret, QLIB_STATUS__COMMAND_FAIL, "W77Q_RPMC_Set_Params fail:%s \n", DEFS_STATUS_TO_STR(fmc_ret));
ASSERT_MSG_RET(QLIB_STATUS__OK == w77q_mc_check_inc(&mc0, NULL, &cnt_a), QLIB_STATUS__COMMAND_FAIL,"w77q_mc_check_inc mc0 fail\n");
// ********************************************************************************************
// mc1 is W77Q_RPMC_TYPE_RN_WF so actor reads without a key but someone with a key writes to it
// ********************************************************************************************
section = 1;
W77Q_RPMC_T mc1;
fmc_ret = W77Q_RPMC_Set_Params(&mc1, W77Q_RPMC_TYPE_RN_WF, section, counter_offset, ctx);
ASSERT_MSG_RET(DEFS_STATUS_OK == fmc_ret, QLIB_STATUS__COMMAND_FAIL, "W77Q_RPMC_Set_Params fail:%s \n", DEFS_STATUS_TO_STR(fmc_ret));
// inc sequence does not work without the key
ASSERT_MSG_RET(QLIB_STATUS__OK != w77q_mc_check_inc(&mc1, NULL, &cnt_a), QLIB_STATUS__COMMAND_FAIL,"w77q_mc_check_inc mc1 fail\n");
// inc sequence works with full access key
QLIB_STATUS_RET_CHECK(fwGetSectionKey(ctx, &id, QLIB_KID__FULL_ACCESS_SECTION, section, 0, keyF, sizeof(KEY_T)));
ASSERT_MSG_RET(QLIB_STATUS__OK == w77q_mc_check_inc(&mc1, keyF, &cnt_a), QLIB_STATUS__COMMAND_FAIL,"w77q_mc_check_inc mc1 fail\n");
// get sequence works without the key and provides correct counter value
ASSERT_MSG_RET(QLIB_STATUS__OK == w77q_mc_check_get(&mc1, NULL, &cnt_b), QLIB_STATUS__COMMAND_FAIL,"w77q_mc_check_inc mc1 fail\n");
ASSERT_MSG_RET(cnt_a == cnt_b, QLIB_STATUS__COMMAND_FAIL, "MC counters' check fail, should be equal \n");
#endif //#ifndef EXCLUDE_PLAIN_ACCESS
// ******************************************************************************
// mc2 is W77Q_RPMC_TYPE_RF_WF so actor reads and writes with same full access key
// ******************************************************************************
section = 2;
W77Q_RPMC_T mc2;
fmc_ret = W77Q_RPMC_Set_Params(&mc2, W77Q_RPMC_TYPE_RF_WF, section, counter_offset, ctx);
ASSERT_MSG_RET(DEFS_STATUS_OK == fmc_ret, QLIB_STATUS__COMMAND_FAIL, "W77Q_RPMC_Set_Params fail:%s \n", DEFS_STATUS_TO_STR(fmc_ret));
// inc sequence does not work without the key
ASSERT_MSG_RET(QLIB_STATUS__OK != w77q_mc_check_inc(&mc2, NULL, &cnt_a), QLIB_STATUS__COMMAND_FAIL,"w77q_mc_check_inc mc2 fail\n");
// inc sequence works with full access key
QLIB_STATUS_RET_CHECK(fwGetSectionKey(ctx, &id, QLIB_KID__FULL_ACCESS_SECTION, section, 0, keyF, sizeof(KEY_T)));
ASSERT_MSG_RET(QLIB_STATUS__OK == w77q_mc_check_inc(&mc2, keyF, &cnt_a), QLIB_STATUS__COMMAND_FAIL,"w77q_mc_check_inc mc2 fail\n");
// get sequence does not work without the key
ASSERT_MSG_RET(QLIB_STATUS__OK != w77q_mc_check_get(&mc2, NULL, &cnt_b), QLIB_STATUS__COMMAND_FAIL,"w77q_mc_check_inc mc2 fail\n");
// get sequence works with full access key
ASSERT_MSG_RET(QLIB_STATUS__OK == w77q_mc_check_get(&mc2, keyF, &cnt_b), QLIB_STATUS__COMMAND_FAIL,"w77q_mc_check_inc mc2 fail\n");
ASSERT_MSG_RET(cnt_a == cnt_b, QLIB_STATUS__COMMAND_FAIL, "MC counters' check fail, should be equal \n");
// inc sequence does not works with wrong key access key ( from next section in this example )
QLIB_STATUS_RET_CHECK(fwGetSectionKey(ctx, &id, QLIB_KID__FULL_ACCESS_SECTION, section + 1, 0, keyF, sizeof(KEY_T)));
ASSERT_MSG_RET(QLIB_STATUS__OK != w77q_mc_check_inc(&mc2, keyF, &cnt_a), QLIB_STATUS__COMMAND_FAIL,"w77q_mc_check_inc mc2 fail\n");
// *********************************************************************************************************
// mc3 is W77Q_RPMC_TYPE_RR_WF so actor reads it with one key ( restricted ), someone else writes to it
// *********************************************************************************************************
section = 3;
W77Q_RPMC_T mc3;
fmc_ret = W77Q_RPMC_Set_Params(&mc3, W77Q_RPMC_TYPE_RR_WF, section, counter_offset, ctx);
ASSERT_MSG_RET(DEFS_STATUS_OK == fmc_ret, QLIB_STATUS__COMMAND_FAIL, "W77Q_RPMC_Set_Params fail:%s \n", DEFS_STATUS_TO_STR(fmc_ret));
// get sequence does not work without the key
ASSERT_MSG_RET(QLIB_STATUS__OK != w77q_mc_check_get(&mc3, NULL, &cnt_a), QLIB_STATUS__COMMAND_FAIL, "w77q_mc_check_get mc3 fail\n");
// get sequence works with restricted access key
QLIB_STATUS_RET_CHECK(fwGetSectionKey(ctx, &id, QLIB_KID__RESTRICTED_ACCESS_SECTION, section, 0, keyF, sizeof(KEY_T)));
ASSERT_MSG_RET(QLIB_STATUS__OK == w77q_mc_check_get(&mc3, keyF, &cnt_a), QLIB_STATUS__COMMAND_FAIL, "w77q_mc_check_get mc3 fail\n");
// inc sequence does not work with restricted
ASSERT_MSG_RET(QLIB_STATUS__OK != w77q_mc_check_inc(&mc3, keyF, &cnt_a), QLIB_STATUS__COMMAND_FAIL, "w77q_mc_check_inc mc3 fail\n");
// *********************************************************************************************************
// check if mc3 requires maintance, full access key is required
// *********************************************************************************************************
bool needMaint;
fmc_ret = W77Q_RPMC_IsMaintenanceNeeded(&mc3, &needMaint);
ASSERT_MSG_RET(DEFS_STATUS_OK == fmc_ret, QLIB_STATUS__COMMAND_FAIL, "W77Q_RPMC_IsMaintenanceNeeded fail:%s \n", DEFS_STATUS_TO_STR(fmc_ret));
if (needMaint)
{
QLIB_STATUS_RET_CHECK(fwGetSectionKey(ctx, &id, QLIB_KID__FULL_ACCESS_SECTION, section, 0, keyF, sizeof(KEY_T)));
fmc_ret = W77Q_RPMC_PerformMaintenance(&mc3, keyF);
ASSERT_MSG_RET(DEFS_STATUS_OK == fmc_ret, QLIB_STATUS__COMMAND_FAIL, "W77Q_RPMC_PerformMaintenance fail:%s \n", DEFS_STATUS_TO_STR(fmc_ret));
}
}
PRINTF("Tested negative/positive %d counters successfully\n", SIZE_1MB_ / step);
// now let's run couter maintanence if needed
return QLIB_STATUS__OK;
}
