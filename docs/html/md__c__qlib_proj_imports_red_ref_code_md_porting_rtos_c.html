// Context structure for RTOS/HAL
typedef struct {
HAL_SPI_HandleTypeDef* hspi; // Pointer to the HAL SPI handle
MutexHandle_t mutex; // Mutex to protect access to this SPI device
uint32_t timeout_ms; // Default transaction timeout
// Add other config if needed (e.g., CS# GPIO info if HAL doesn't manage it)
} RtosSpiContext;
// --- PLAT_SPI_WriteReadTransaction Implementation (RTOS/HAL) ---
int PLAT_SPI_WriteReadTransaction(const void* userData,
QLIB_BUS_MODE_T format,
uint32_t flags,
const uint8_t* dataOutStream,
uint32_t cmdSize,
uint32_t addressSize,
uint32_t dataOutSize,
uint32_t dummyCycles,
uint8_t* dataIn,
uint32_t dataInSize)
{
const RtosSpiContext* context = (const RtosSpiContext*)userData;
if (!context || !context->hspi || !context->mutex) {
return PLAT_SPI_INV_PARAM;
}
HAL_StatusTypeDef hal_status = HAL_OK;
int plat_status = PLAT_SPI_OK;
// --- 1. Acquire Mutex ---
if (xSemaphoreTake(context->mutex, pdMS_TO_TICKS(context->timeout_ms)) != 1) { // Use pdTRUE for FreeRTOS
return PLAT_SPI_BUSY; // Failed to acquire mutex within timeout
}
// --- 2. Configure SPI (if needed) ---
// Some HALs require configuration per transaction, others configure once at init.
// This is a placeholder showing potential reconfiguration.
// HAL_SPI_SetConfig(context->hspi, format); // Adjust based on actual HAL API
// --- 3. Perform Transaction using HAL ---
// Option A: HAL supports combined TransmitReceive (Preferred if suitable)
// This often requires preparing a single TX buffer and a single RX buffer.
// It assumes the HAL handles CS# assertion/deassertion correctly for the combined call.
// NOTE: This simple HAL_SPI_TransmitReceive might not directly support separate
// command, address, data, dummy phases within a single call. It usually performs
// full-duplex transfer of a given size. Adapting requires careful buffer setup.
// Option B: Separate Transmit and Receive calls (More flexible, requires HAL to keep CS# low)
// Assumes HAL_SPI_Transmit and HAL_SPI_Receive calls under the same mutex lock
// will keep CS# asserted IF the HAL is configured appropriately (often requires
// specific HAL init settings or flags). If not, manual CS# control is needed around these calls.
uint32_t totalWriteSize = cmdSize + addressSize + dataOutSize;
uint8_t dummy_byte = 0xFF; // Dummy byte for clocking
// Transmit Command + Address + Data
if (totalWriteSize > 0) {
// NOTE: HAL functions often take uint16_t for size. Check your HAL.
// May need to handle transfers larger than 65535 bytes in chunks if required.
hal_status = HAL_SPI_Transmit(context->hspi, (uint8_t*)dataOutStream, (uint16_t)totalWriteSize, context->timeout_ms);
if (hal_status != HAL_OK) {
plat_status = (hal_status == HAL_TIMEOUT) ? PLAT_SPI_TIMEOUT : PLAT_SPI_ERROR;
goto cleanup; // Go to mutex release
}
}
// Transmit Dummy Bytes (Clock out dummy cycles)
uint32_t num_dummy_bytes = (dummyCycles + 7) / 8;
for (uint32_t i = 0; i < num_dummy_bytes; ++i) {
hal_status = HAL_SPI_Transmit(context->hspi, &dummy_byte, 1, context->timeout_ms);
if (hal_status != HAL_OK) {
plat_status = (hal_status == HAL_TIMEOUT) ? PLAT_SPI_TIMEOUT : PLAT_SPI_ERROR;
goto cleanup;
}
}
// Receive Data
if (dataInSize > 0 && dataIn) {
// Some HALs require transmitting dummy data while receiving.
// Others have a dedicated Receive function that handles dummy TX internally.
// Assuming HAL_SPI_Receive handles dummy TX:
hal_status = HAL_SPI_Receive(context->hspi, dataIn, (uint16_t)dataInSize, context->timeout_ms);
// If HAL_SPI_Receive doesn't exist or requires simultaneous TX, use TransmitReceive:
// Create a dummy TX buffer if needed
// uint8_t* dummyTx = malloc(dataInSize);
// if (dummyTx) {
// memset(dummyTx, 0xFF, dataInSize);
// hal_status = HAL_SPI_TransmitReceive(context->hspi, dummyTx, dataIn, (uint16_t)dataInSize, context->timeout_ms);
// free(dummyTx);
// } else { hal_status = HAL_ERROR; /* Allocation failed */ }
if (hal_status != HAL_OK) {
plat_status = (hal_status == HAL_TIMEOUT) ? PLAT_SPI_TIMEOUT : PLAT_SPI_ERROR;
goto cleanup;
}
}
cleanup:
// --- 4. Release Mutex ---
xSemaphoreGive(context->mutex);
// --- 5. De-assert CS# (if manually controlled) ---
// If HAL doesn't manage CS# automatically across calls or based on flags,
// add manual CS# de-assertion here (outside the mutex if CS# pin is shared).
// Check PLAT_SPI_FLAG_KEEP_CS_ASSERTED flag if needed.
return plat_status;
}
