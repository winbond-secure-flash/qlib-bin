// --- Platform Specific Definitions (Zephyr) ---
// Context structure for Zephyr
typedef struct {
const struct device *spi_dev; // Pointer to the Zephyr SPI device structure
struct spi_cs_control cs_ctrl; // Chip select control structure (if needed)
uint32_t spi_freq_hz; // Default SPI frequency
// Add other config if needed
} ZephyrSpiContext;
// --- PLAT_SPI_WriteReadTransaction Implementation (Zephyr) ---
int PLAT_SPI_WriteReadTransaction(const void* userData,
QLIB_BUS_MODE_T format,
uint32_t flags,
const uint8_t* dataOutStream,
uint32_t cmdSize,
uint32_t addressSize,
uint32_t dataOutSize,
uint32_t dummyCycles,
uint8_t* dataIn,
uint32_t dataInSize)
{
const ZephyrSpiContext* context = (const ZephyrSpiContext*)userData;
if (!context || !context->spi_dev) {
LOG_ERR("Invalid userData or SPI device pointer");
return PLAT_SPI_INV_PARAM;
}
// --- 1. Prepare SPI Configuration ---
// Map QLIB format/flags to Zephyr spi_config operation bits
uint16_t operation = SPI_WORD_SET(8) | SPI_TRANSFER_MSB | SPI_OP_MODE_MASTER; // Common defaults
switch (format) {
case QLIB_BUS_MODE_SPI_MODE0: operation |= SPI_MODE_CPOL(0) | SPI_MODE_CPHA(0); break;
case QLIB_BUS_MODE_SPI_MODE1: operation |= SPI_MODE_CPOL(0) | SPI_MODE_CPHA(1); break;
case QLIB_BUS_MODE_SPI_MODE2: operation |= SPI_MODE_CPOL(1) | SPI_MODE_CPHA(0); break;
case QLIB_BUS_MODE_SPI_MODE3: operation |= SPI_MODE_CPOL(1) | SPI_MODE_CPHA(1); break;
default:
LOG_ERR("Unsupported QLIB_BUS_MODE_T: %d", format);
return PLAT_SPI_INV_PARAM;
}
// Add LSB/MSB, specific flags handling if needed based on 'format' or 'flags'
struct spi_config spi_cfg = {
.frequency = context->spi_freq_hz, // Use frequency from context
.operation = operation,
.slave = 0, // Master mode, index of slave device on bus (usually 0 if only one)
.cs = context->cs_ctrl.gpio ? context->cs_ctrl : (struct spi_cs_control){0}, // Use CS# control from context if valid
};
// Adjust frequency based on PLAT_SPI_FLAG_HIGH_SPEED if needed
// if (flags & PLAT_SPI_FLAG_HIGH_SPEED) { spi_cfg.frequency = HIGHER_FREQ; }
// --- 2. Prepare Buffers (spi_buf and spi_buf_set) ---
// Zephyr's spi_transceive allows defining sequences of TX and RX buffers.
// We need up to 3 distinct phases: Write(cmd+addr+data), Dummy TX, Read RX.
// Count how many buffer sets we need (max 3: write, dummy, read)
int buf_set_count = 0;
if (cmdSize + addressSize + dataOutSize > 0) buf_set_count++;
if (dummyCycles > 0) buf_set_count++;
if (dataInSize > 0 && dataIn) buf_set_count++;
if (buf_set_count == 0) return PLAT_SPI_OK; // Nothing to do
struct spi_buf tx_bufs[3]; // Max 3 TX buffers needed (write, dummy, dummy-for-read)
struct spi_buf rx_bufs[3]; // Max 3 RX buffers needed (discard, discard, read)
struct spi_buf_set buf_sets[3];
int current_set = 0;
// Set 1: Write Phase (Command + Address + Data)
uint32_t totalWriteSize = cmdSize + addressSize + dataOutSize;
if (totalWriteSize > 0) {
tx_bufs[current_set].buf = (void*)dataOutStream;
tx_bufs[current_set].len = totalWriteSize;
rx_bufs[current_set].buf = NULL; // Discard received data during write
rx_bufs[current_set].len = totalWriteSize;
buf_sets[current_set].buffers = &tx_bufs[current_set];
buf_sets[current_set].count = 1; // Only one buffer in this set
// We need to set the rx_buffers field in the spi_buf_set if we want to receive during tx
// For write-only phase, often better to set rx_buf to NULL in spi_buf
// and not set rx_buffers in spi_buf_set. Let's adjust:
buf_sets[current_set].buffers = &tx_bufs[current_set]; // TX only
buf_sets[current_set].count = 1;
current_set++;
}
// Set 2: Dummy Cycle Phase (Transmit dummy bytes)
// Zephyr doesn't have explicit dummy cycles, implement by transmitting dummy data.
uint8_t dummy_byte_tx = 0xFF; // Or 0x00
uint32_t num_dummy_bytes = (dummyCycles + 7) / 8;
if (num_dummy_bytes > 0) {
// Need a buffer for the dummy bytes. If > 1, might need dynamic allocation or a static buffer.
// For simplicity, assuming we can send one byte at a time if needed,
// or have a small static buffer. Let's assume sending one byte repeatedly.
// This is inefficient. Better: Create a single buffer for all dummies.
// For this example, let's represent it as one buffer set with one buffer.
// NOTE: This requires a buffer holding `num_dummy_bytes` bytes of dummy data.
// Using a single byte buffer and repeating the set is NOT standard usage.
// A better approach would be to allocate/use a static buffer for dummies.
// Let's assume a single dummy byte buffer for simplicity of example structure:
tx_bufs[current_set].buf = &dummy_byte_tx; // Needs a real buffer if num_dummy_bytes > 1
tx_bufs[current_set].len = num_dummy_bytes; // This assumes the buffer has num_dummy_bytes bytes!
rx_bufs[current_set].buf = NULL; // Discard RX during dummy phase
rx_bufs[current_set].len = num_dummy_bytes;
buf_sets[current_set].buffers = &tx_bufs[current_set];
buf_sets[current_set].count = 1;
current_set++;
}
// Set 3: Read Phase (Transmit dummy bytes to clock in data)
if (dataInSize > 0 && dataIn) {
// Need to transmit dummy bytes while receiving.
// Again, requires a buffer for dummy TX data.
tx_bufs[current_set].buf = NULL; // Transmit default bytes (usually 0xFF or 0x00 depending on driver)
tx_bufs[current_set].len = dataInSize;
rx_bufs[current_set].buf = dataIn; // Receive into user buffer
rx_bufs[current_set].len = dataInSize;
// Structure for combined TX/RX in one set element:
struct spi_buf combined_bufs[2];
combined_bufs[0].buf = NULL; // TX dummy
combined_bufs[0].len = dataInSize;
combined_bufs[1].buf = dataIn; // RX data
combined_bufs[1].len = dataInSize;
buf_sets[current_set].buffers = combined_bufs; // Point to the combined array
buf_sets[current_set].count = 2; // Two buffers in this set (TX and RX)
current_set++;
}
// --- 3. Perform Transaction ---
// The spi_transceive function takes an array of spi_buf_set.
int rc = spi_transceive(context->spi_dev, &spi_cfg, buf_sets, current_set);
// --- 4. Handle Result ---
if (rc == 0) {
return PLAT_SPI_OK;
} else {
LOG_ERR("SPI transaction failed: %d", rc);
// Translate Zephyr error codes (-errno) to platform codes
if (rc == -EBUSY) return PLAT_SPI_BUSY;
if (rc == -ETIMEDOUT) return PLAT_SPI_TIMEOUT;
return PLAT_SPI_ERROR; // General error
}
}
