58QLIB_STATUS_T QLIB_SAMPLE_KD1Config(void* userData_pv)
59{
60 QLIB_CONTEXT_T qlibContext;
61 QLIB_STATUS_T status = QLIB_STATUS__OK;
62 QLIB_BUS_FORMAT_T busFormat = QLIB_BUS_FORMAT(QLIB_BUS_MODE_1_1_1, (bool)false);
63
64 /*-------------------------------------------------------------------------------------------------------
65 Standard QLIB initialization sequence used by all samples
66 -------------------------------------------------------------------------------------------------------*/
67 QLIB_SAMPLE_PRINTF("> QLIB_SAMPLE_KD1Config: Initializing QLIB and connecting to the device...\r\n");
68
69 QLIB_STATUS_RET_CHECK(QLIB_InitLib(&qlibContext));
70 QLIB_SetUserData(&qlibContext, userData_pv);
71 QLIB_STATUS_RET_CHECK(QLIB_Connect(&qlibContext));
72 QLIB_STATUS_RET_CHECK_GOTO(QLIB_InitDevice(&qlibContext, busFormat), status, disconnect);
73
74 QLIB_SAMPLE_PRINTF("> QLIB_SAMPLE_KD1Config: Initializing finished successfully\r\n");
75
76 /*-------------------------------------------------------------------------------------------------------
77 Provision the device using a KD1-derived pre-provisioned master key
78 -------------------------------------------------------------------------------------------------------*/
79 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_KD1Config_provision(&qlibContext), status, disconnect);
80
81disconnect:
82 (void)QLIB_Disconnect(&qlibContext);
83 return status;
84}

86QLIB_STATUS_T QLIB_SAMPLE_KD1Config_provision(QLIB_CONTEXT_T* ctx)
87{
88 QLIB_ID_T id;
89 KEY_T kd1_key;
90#if QLIB_MAX_DIES_SUPPORTED > 1u && defined W77Q_KDF_KD1
91 uint8_t origMaxDie_u8 = ctx->MaxDieId;
92#endif // QLIB_MAX_DIES_SUPPORTED > 1u && W77Q_KDF_KD1
93
94 QLIB_ASSERT_RET(NULL != ctx, QLIB_STATUS__INVALID_PARAMETER);
95
96 /*-------------------------------------------------------------------------------------------------------
97 Force the active die to die 0. KD1 is a per-die pre-provisioned key but has same value for all dies
98 according to die 0 WID.
99 -------------------------------------------------------------------------------------------------------*/
100#if QLIB_MAX_DIES_SUPPORTED > 1u
101 QLIB_SAMPLE_PRINTF("> QLIB_SAMPLE_KD1Config: Set active die to 0\r\n");
102 QLIB_STATUS_RET_CHECK(QLIB_SetActiveDie(ctx, 0u));
103#ifdef W77Q_KDF_KD1
104 ctx->MaxDieId = 0u;
105#endif // defined W77Q_KDF_KD1
106#endif // QLIB_MAX_DIES_SUPPORTED > 1u
107
108 /*-------------------------------------------------------------------------------------------------------
109 Read the flash IDs to obtain the die 0 Winbond ID (WID)
110 -------------------------------------------------------------------------------------------------------*/
111 QLIB_SAMPLE_PRINTF("> QLIB_SAMPLE_KD1Config: obtain the die 0 Winbond ID (WID)\r\n");
112 QLIB_STATUS_RET_CHECK(QLIB_GetId(ctx, &id));
113
114 QLIB_SAMPLE_PRINTF("> QLIB_SAMPLE_KD1Config: die 0 WID = 0x%08lx%08lx\r\n",
115 (unsigned long)id.sec.wid[1],
116 (unsigned long)id.sec.wid[0]);
117
118 /*-------------------------------------------------------------------------------------------------------
119 Derive the KD1 pre-provisioned master key from the WID
120 -------------------------------------------------------------------------------------------------------*/
121 QLIB_SAMPLE_PRINTF("> QLIB_SAMPLE_KD1Config: deriving KD1 from the die 0 WID\r\n");
122 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetDeviceKey(ctx, id.sec.wid, QLIB_KID__DEVICE_KEY_PRE_PROVISIONING, kd1_key));
123 QLIB_SAMPLE_PRINTF("> QLIB_SAMPLE_KD1Config: retrieved KD1 = 0x");
124 for (uint32_t i_u32 = 0u; i_u32 < (uint32_t)sizeof(KEY_T); ++i_u32)
125 {
126 QLIB_SAMPLE_PRINTF("%02x", (unsigned int)((const uint8_t*)kd1_key)[i_u32]);
127 }
128 QLIB_SAMPLE_PRINTF("\r\n");
129
130 /*-------------------------------------------------------------------------------------------------------
131 Use KD1 to authenticate the device configuration: install the customer's new master key, configure
132 section 1 with a full/restricted access key and a secure-only access policy
133 -------------------------------------------------------------------------------------------------------*/
134 QLIB_SAMPLE_PRINTF("> QLIB_SAMPLE_KD1Config: use Kd1 to config new master key and secure sections\r\n");
135 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_KD1Config_configureDevice_L(ctx, kd1_key, id.sec.wid));
136
137 /*-------------------------------------------------------------------------------------------------------
138 Verify that the provisioning succeeded by opening a full-access session and doing a write/read
139 -------------------------------------------------------------------------------------------------------*/
140 QLIB_SAMPLE_PRINTF("> QLIB_SAMPLE_KD1Config: Verify configured secure section access (read, write)\r\n");
141 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_KD1Config_verifyWriteRead_L(ctx, id.sec.wid));
142
143#if QLIB_MAX_DIES_SUPPORTED > 1u && defined W77Q_KDF_KD1
144 ctx->MaxDieId = origMaxDie_u8;
145#endif // QLIB_MAX_DIES_SUPPORTED > 1u && W77Q_KDF_KD1
146 return QLIB_STATUS__OK;
147}

164static QLIB_STATUS_T QLIB_SAMPLE_KD1Config_configureDevice_L(QLIB_CONTEXT_T* ctx, const KEY_T kd1_key, QLIB_WID_T wid)
165{
166 QLIB_FLASH_CONFIG_T flashCfg = {0u};
167 QLIB_DIE_CONFIG_T dieCfg = {0u};
168 QLIB_SECTIONS_CONF_TABLE_T cfgLine[QLIB_MAX_DIES_SUPPORTED];
169 QLIB_EXTENDED_SECTION_CONF_T* sectCfg;
170 uint8_t diesNum_u8 = QLIB_GET_NUM_OF_DIES(ctx);
171 uint8_t die_u8 = 0u;
172
173 /*-------------------------------------------------------------------------------------------------------
174 Read the existing configuration so we can modify only what we need
175 -------------------------------------------------------------------------------------------------------*/
176 QLIB_STATUS_RET_CHECK(QLIB_GetDeviceConfigMultiDie(ctx, &flashCfg, &dieCfg, cfgLine, diesNum_u8));
177
178 /*-------------------------------------------------------------------------------------------------------
179 Make sure flash is formatted
180 -------------------------------------------------------------------------------------------------------*/
181 if ((bool)true == dieCfg.nonSecureFormatEn)
182 {
183 QLIB_SAMPLE_PRINTF("> QLIB_SAMPLE_KD1Config: Format Flash\r\n");
184 QLIB_STATUS_RET_CHECK(QLIB_Format(ctx, NULL, (bool)false, (bool)false));
185 QLIB_SAMPLE_PRINTF("> QLIB_SAMPLE_KD1Config: Format Flash Success\r\n");
186 // re-read the configuration after format
187 QLIB_STATUS_RET_CHECK(QLIB_GetDeviceConfigMultiDie(ctx, &flashCfg, &dieCfg, cfgLine, diesNum_u8));
188 }
189 else
190 {
191 QLIB_SAMPLE_PRINTF("> QLIB_SAMPLE_KD1Config: Skip Flash Format\r\n");
192 }
193
194 /*-------------------------------------------------------------------------------------------------------
195 Install the KD1-derived pre-provisioned master key. This is what authenticates the configuration
196 transaction on a freshly manufactured, pre-provisioned device.
197 -------------------------------------------------------------------------------------------------------*/
198 (void)memcpy((void*)dieCfg.preProvisionedMasterKey, (const void*)kd1_key, sizeof(KEY_T));
199
200 /*-------------------------------------------------------------------------------------------------------
201 Install a new customer-chosen device master key, obtained via the standard sample helper
202 -------------------------------------------------------------------------------------------------------*/
203 QLIB_SAMPLE_PRINTF("> QLIB_SAMPLE_KD1Config: Set new master key\r\n");
204 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetDeviceKey(ctx, wid, QLIB_KID__DEVICE_MASTER, dieCfg.deviceMasterKey));
205
206 /*-------------------------------------------------------------------------------------------------------
207 Configure section 1 as a secure section (no plain access, full and restricted access keys installed)
208 -------------------------------------------------------------------------------------------------------*/
209 QLIB_SAMPLE_PRINTF("> QLIB_SAMPLE_KD1Config: Set secure (no plain access) section id %u with full access and restricted "
210 "access "
211 "keys\r\n",
212 (unsigned int)QLIB_SAMPLE_KD1_CONFIG_SECTION);
213 sectCfg = cfgLine[die_u8].sectionConfigTable;
214
215 sectCfg[QLIB_SAMPLE_KD1_CONFIG_SECTION].baseAddr = QLIB_SAMPLE_RESERVED_BOOT_SECTION_SIZE;
216 sectCfg[QLIB_SAMPLE_KD1_CONFIG_SECTION].size = QLIB_SAMPLE_KD1_CONFIG_SECTION_SIZE;
217
218 sectCfg[QLIB_SAMPLE_KD1_CONFIG_SECTION].sectionConfig.policy.plainAccessReadEnable = 0u;
219 sectCfg[QLIB_SAMPLE_KD1_CONFIG_SECTION].sectionConfig.policy.plainAccessWriteEnable = 0u;
220 sectCfg[QLIB_SAMPLE_KD1_CONFIG_SECTION].sectionConfig.policy.authPlainAccess = 0u;
221 sectCfg[QLIB_SAMPLE_KD1_CONFIG_SECTION].sectionConfig.policy.checksumIntegrity = 0u;
222 sectCfg[QLIB_SAMPLE_KD1_CONFIG_SECTION].sectionConfig.policy.digestIntegrity = 0u;
223 sectCfg[QLIB_SAMPLE_KD1_CONFIG_SECTION].sectionConfig.policy.digestIntegrityOnAccess = 0u;
224 sectCfg[QLIB_SAMPLE_KD1_CONFIG_SECTION].sectionConfig.policy.rollbackProt = 0u;
225 sectCfg[QLIB_SAMPLE_KD1_CONFIG_SECTION].sectionConfig.policy.slog = 0u;
226 sectCfg[QLIB_SAMPLE_KD1_CONFIG_SECTION].sectionConfig.policy.writeProt = 0u;
227
228 /*-------------------------------------------------------------------------------------------------------
229 Full and restricted access keys for the provisioned section, obtained via the standard sample helper
230 -------------------------------------------------------------------------------------------------------*/
231 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSectionAccessKey(ctx,
232 wid,
233 QLIB_KID__FULL_ACCESS_SECTION,
234 QLIB_SAMPLE_KD1_CONFIG_SECTION,
235 (uint32_t)die_u8,
236 sectCfg[QLIB_SAMPLE_KD1_CONFIG_SECTION].fullAccessKey));
237
238 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSectionAccessKey(ctx,
239 wid,
240 QLIB_KID__RESTRICTED_ACCESS_SECTION,
241 QLIB_SAMPLE_KD1_CONFIG_SECTION,
242 (uint32_t)die_u8,
243 sectCfg[QLIB_SAMPLE_KD1_CONFIG_SECTION].restrictedKey));
244
245 /*-------------------------------------------------------------------------------------------------------
246 Write the new configuration to the device. This authenticates using preProvisionedMasterKey (=KD1)
247 and programs the new deviceMasterKey and section 1 configuration.
248 -------------------------------------------------------------------------------------------------------*/
249 QLIB_STATUS_RET_CHECK(QLIB_ConfigDeviceMultiDie(ctx, &flashCfg, &dieCfg, cfgLine, diesNum_u8));
250 QLIB_SAMPLE_PRINTF("> QLIB_SAMPLE_KD1Config: Configuration success\r\n");
251 return QLIB_STATUS__OK;
252}

263static QLIB_STATUS_T QLIB_SAMPLE_KD1Config_verifyWriteRead_L(QLIB_CONTEXT_T* ctx, QLIB_WID_T wid)
264{
265 QLIB_STATUS_T status = QLIB_STATUS__OK;
266 KEY_T qlibSampleKd1FullKey;
267 uint8_t writeBuf_u8[QLIB_FLASH_PAGE_SIZE];
268 uint8_t readBuf_u8[QLIB_FLASH_PAGE_SIZE];
269
270 (void)memset(writeBuf_u8, 0xa5, sizeof(writeBuf_u8));
271 (void)memset(readBuf_u8, 0x00, sizeof(readBuf_u8));
272
273 /*-------------------------------------------------------------------------------------------------------
274 Retrieve the full access key configured for the provisioned section (same helper as in configure)
275 -------------------------------------------------------------------------------------------------------*/
276 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSectionAccessKey(ctx,
277 wid,
278 QLIB_KID__FULL_ACCESS_SECTION,
279 QLIB_SAMPLE_KD1_CONFIG_SECTION,
280 0u,
281 qlibSampleKd1FullKey));
282
283 /*-------------------------------------------------------------------------------------------------------
284 Load the full access key for the provisioned section and open a full-access session
285 -------------------------------------------------------------------------------------------------------*/
286 QLIB_SAMPLE_PRINTF("> QLIB_SAMPLE_KD1Config: Load secure section %u full access key\r\n",
287 (unsigned int)QLIB_SAMPLE_KD1_CONFIG_SECTION);
288 QLIB_STATUS_RET_CHECK(QLIB_LoadKey(ctx, QLIB_SAMPLE_KD1_CONFIG_SECTION, qlibSampleKd1FullKey, (bool)true));
289
290 QLIB_SAMPLE_PRINTF("> QLIB_SAMPLE_KD1Config: Open full-access session on secure section %u\r\n",
291 (unsigned int)QLIB_SAMPLE_KD1_CONFIG_SECTION);
292 QLIB_STATUS_RET_CHECK_GOTO(QLIB_OpenSession(ctx, QLIB_SAMPLE_KD1_CONFIG_SECTION, QLIB_SESSION_ACCESS_FULL),
293 status,
294 remove_key);
295
296 /*-------------------------------------------------------------------------------------------------------
297 Erase the section, write one page, read it back and verify
298 -------------------------------------------------------------------------------------------------------*/
299 QLIB_SAMPLE_PRINTF("> QLIB_SAMPLE_KD1Config: Erase, write, and read back section %u data\r\n",
300 (unsigned int)QLIB_SAMPLE_KD1_CONFIG_SECTION);
301 QLIB_STATUS_RET_CHECK_GOTO(QLIB_EraseSection(ctx, QLIB_SAMPLE_KD1_CONFIG_SECTION, (bool)true), status, close_session);
302
303 QLIB_STATUS_RET_CHECK_GOTO(QLIB_Write(ctx, writeBuf_u8, QLIB_SAMPLE_KD1_CONFIG_SECTION, 0u, QLIB_FLASH_PAGE_SIZE, (bool)true),
304 status,
305 close_session);
306
307 QLIB_STATUS_RET_CHECK_GOTO(QLIB_Read(ctx,
308 readBuf_u8,
309 QLIB_SAMPLE_KD1_CONFIG_SECTION,
310 0u,
311 QLIB_FLASH_PAGE_SIZE,
312 (bool)true,
313 (bool)false),
314 status,
315 close_session);
316
317 QLIB_ASSERT_WITH_ERROR_GOTO(0 == memcmp(readBuf_u8, writeBuf_u8, sizeof(writeBuf_u8)),
318 QLIB_STATUS__TEST_FAIL,
319 status,
320 close_session);
321
322 QLIB_SAMPLE_PRINTF("> QLIB_SAMPLE_KD1Config: section %lu write/read round trip passed\r\n",
323 (unsigned long)QLIB_SAMPLE_KD1_CONFIG_SECTION);
324
325 status = QLIB_STATUS__OK;
326
327close_session:
328 (void)QLIB_CloseSession(ctx, QLIB_SAMPLE_KD1_CONFIG_SECTION);
329
330remove_key:
331 (void)QLIB_RemoveKey(ctx, QLIB_SAMPLE_KD1_CONFIG_SECTION, (bool)true);
332
333 return status;
334}
