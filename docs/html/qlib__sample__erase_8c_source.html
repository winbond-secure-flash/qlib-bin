34QLIB_STATUS_T QLIB_SAMPLE_Erase(void* userData_pv)
35{
36 QLIB_CONTEXT_T qlibContext;
37 QLIB_STATUS_T status = QLIB_STATUS__OK;
38 QLIB_BUS_FORMAT_T busFormat = QLIB_BUS_FORMAT(QLIB_BUS_MODE_1_1_1, (bool)false);
39
40 /*-------------------------------------------------------------------------------------------------------
41 Init QLIB - needed when running QLIB either on a device or on a remote server.
42 -------------------------------------------------------------------------------------------------------*/
43 QLIB_STATUS_RET_CHECK(QLIB_InitLib(&qlibContext));
44
45 /*-------------------------------------------------------------------------------------------------------
46 Set the user data into Qlib's context
47 -------------------------------------------------------------------------------------------------------*/
48 QLIB_SetUserData(&qlibContext, userData_pv);
49
50 /*-------------------------------------------------------------------------------------------------------
51 Take the ownership of flash communication channel (it belongs to local MCU or to remote server, exclusively)
52 -------------------------------------------------------------------------------------------------------*/
53 QLIB_STATUS_RET_CHECK(QLIB_Connect(&qlibContext));
54
55 /*-------------------------------------------------------------------------------------------------------
56 Init Flash Device - not needed when using QLIB on a remote server
57 -------------------------------------------------------------------------------------------------------*/
58 QLIB_STATUS_RET_CHECK_GOTO(QLIB_InitDevice(&qlibContext, busFormat), status, disconnect);
59
60 /*-------------------------------------------------------------------------------------------------------
61 Run sample code
62 -------------------------------------------------------------------------------------------------------*/
63
64 /*-------------------------------------------------------------------------------------------------------
65 configuration for sample
66 -------------------------------------------------------------------------------------------------------*/
67 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_Erase_manuf_flow(&qlibContext), status, disconnect);
68
69 /*-------------------------------------------------------------------------------------------------------
70 sample flow
71 -------------------------------------------------------------------------------------------------------*/
72 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_Erase_field_flow(&qlibContext), status, disconnect);
73
74disconnect:
75 (void)QLIB_Disconnect(&qlibContext);
76 return status;
77}

79QLIB_STATUS_T QLIB_SAMPLE_Erase_manuf_flow(QLIB_CONTEXT_T* ctx)
80{
81 QLIB_HW_VER_T hwVer;
82 QLIB_ID_T id;
83 uint32_t sect_u32;
84 uint8_t die_u8 = 0u;
85 QLIB_FLASH_CONFIG_T flashCfg = {0u};
86 QLIB_DIE_CONFIG_T dieCfg = {0u};
87 QLIB_SECTIONS_CONF_TABLE_T cfgLine[QLIB_MAX_DIES_SUPPORTED];
88 QLIB_EXTENDED_SECTION_CONF_T* sectCfg;
89 uint8_t diesNum_u8 = QLIB_GET_NUM_OF_DIES(ctx);
90
91 // See what chip is connected and its unique ids. We might need this info to obtain keys for this
92 // chips according to formula or via DB see QLIB_SAMPLE_GetDeviceKey() function
93 QLIB_STATUS_RET_CHECK(QLIB_GetHWVersion(ctx, &hwVer));
94 QLIB_STATUS_RET_CHECK(QLIB_GetId(ctx, &id));
95
96#if QLIB_MAX_DIES_SUPPORTED > 1u
97 QLIB_STATUS_RET_CHECK(QLIB_GetActiveDie(ctx, &die_u8));
98#endif // QLIB_MAX_DIES_SUPPORTED > 1u
99
100 /*-------------------------------------------------------------------------------------------------*/
101 /* Configure flash device */
102 /*-------------------------------------------------------------------------------------------------*/
103#ifdef QLIB_SAMPLE_CDI_PRESERVE_BOOT_SECTION
104 // Read-modify-write: read existing config and only configure sections 1, 2, 3.
105 // Format is skipped to preserve the boot loader in section 0.
106 QLIB_STATUS_RET_CHECK(QLIB_GetDeviceConfigMultiDie(ctx, &flashCfg, &dieCfg, cfgLine, diesNum_u8));
107#else
108 /*-------------------------------------------------------------------------------------------------*/
109 /* Perform full device format */
110 /* this will erase all data and keys from flash */
111 /*-------------------------------------------------------------------------------------------------*/
112 QLIB_STATUS_RET_CHECK(QLIB_GetDeviceConfigMultiDie(ctx, &flashCfg, &dieCfg, NULL, diesNum_u8));
113
114 if ((bool)true == dieCfg.nonSecureFormatEn)
115 {
116 // non secure format
117 QLIB_STATUS_RET_CHECK(QLIB_Format(ctx, NULL, (bool)false, (bool)false));
118 }
119 else
120 {
121 // secure format
122 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetDeviceKey(ctx, id.std.uniqueID, QLIB_KID__DEVICE_MASTER, dieCfg.deviceMasterKey));
123 QLIB_STATUS_RET_CHECK(QLIB_Format(ctx, dieCfg.deviceMasterKey, (bool)false, (bool)false));
124 }
125
126 /*-------------------------------------------------------------------------------------------------*/
127 /* Set new configuration on flash */
128 /*-------------------------------------------------------------------------------------------------*/
129 // We do read / modify /write configuration since we want to change only what is needed
130 // read configuration
131 QLIB_STATUS_RET_CHECK(QLIB_GetDeviceConfigMultiDie(ctx, &flashCfg, &dieCfg, cfgLine, diesNum_u8));
132
133 // modify configuration
134#endif // QLIB_SAMPLE_CDI_PRESERVE_BOOT_SECTION
135
136 // set pre provision master key if needed
137 QLIB_STATUS_RET_CHECK(
138 QLIB_SAMPLE_GetDeviceKey(ctx, id.std.uniqueID, QLIB_KID__DEVICE_KEY_PRE_PROVISIONING, dieCfg.preProvisionedMasterKey));
139
140 // set master key
141 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetDeviceKey(ctx, id.std.uniqueID, QLIB_KID__DEVICE_MASTER, dieCfg.deviceMasterKey));
142
143 //set secret key
144 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetDeviceKey(ctx, id.std.uniqueID, QLIB_KID__DEVICE_SECRET, dieCfg.deviceSecretKey));
145
146 // set SUID
147 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSecureUID(ctx, id.std.uniqueID, dieCfg.suid));
148
149 // setting layout and policy of section 1
150 // this section is plain read enabled and plain write enabled
151 sect_u32 = 1u;
152
153 sectCfg = cfgLine[die_u8].sectionConfigTable;
154
155 sectCfg[sect_u32].baseAddr = QLIB_SAMPLE_RESERVED_BOOT_SECTION_SIZE;
156 sectCfg[sect_u32].size = SIZE_256KB_;
157
158 sectCfg[sect_u32].sectionConfig.policy.plainAccessReadEnable = 1u;
159 sectCfg[sect_u32].sectionConfig.policy.plainAccessWriteEnable = 1u;
160 sectCfg[sect_u32].sectionConfig.policy.authPlainAccess = 0u;
161 sectCfg[sect_u32].sectionConfig.policy.checksumIntegrity = 0u;
162 sectCfg[sect_u32].sectionConfig.policy.digestIntegrity = 0u;
163 sectCfg[sect_u32].sectionConfig.policy.digestIntegrityOnAccess = 0u;
164 sectCfg[sect_u32].sectionConfig.policy.rollbackProt = 0u;
165 sectCfg[sect_u32].sectionConfig.policy.slog = 0u;
166 sectCfg[sect_u32].sectionConfig.policy.writeProt = 0u;
167
168 // Full access key configuration.This is the key to use later for the section configuration
169 // changes or write access
170 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSectionAccessKey(ctx,
171 id.std.uniqueID,
172 QLIB_KID__FULL_ACCESS_SECTION,
173 sect_u32,
174 die_u8,
175 sectCfg[sect_u32].fullAccessKey));
176
177 // setting layout of section 2
178 // this section is plain write disabled
179 sect_u32 = 2u;
180
181 sectCfg[sect_u32].baseAddr = sectCfg[sect_u32 - 1].baseAddr + sectCfg[sect_u32 - 1].size;
182 sectCfg[sect_u32].size = SIZE_256KB_;
183
184 sectCfg[sect_u32].sectionConfig.policy.plainAccessReadEnable = 1u;
185 sectCfg[sect_u32].sectionConfig.policy.plainAccessWriteEnable = 0u;
186 sectCfg[sect_u32].sectionConfig.policy.authPlainAccess = 0u;
187 sectCfg[sect_u32].sectionConfig.policy.checksumIntegrity = 0u;
188 sectCfg[sect_u32].sectionConfig.policy.digestIntegrity = 0u;
189 sectCfg[sect_u32].sectionConfig.policy.digestIntegrityOnAccess = 0u;
190 sectCfg[sect_u32].sectionConfig.policy.rollbackProt = 0u;
191 sectCfg[sect_u32].sectionConfig.policy.slog = 0u;
192 sectCfg[sect_u32].sectionConfig.policy.writeProt = 0u;
193
194 // get restricted and full access key to configure into chip.
195 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSectionAccessKey(ctx,
196 id.std.uniqueID,
197 QLIB_KID__RESTRICTED_ACCESS_SECTION,
198 sect_u32,
199 die_u8,
200 sectCfg[sect_u32].restrictedKey));
201 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSectionAccessKey(ctx,
202 id.std.uniqueID,
203 QLIB_KID__FULL_ACCESS_SECTION,
204 sect_u32,
205 die_u8,
206 sectCfg[sect_u32].fullAccessKey));
207
208 // setting layout of the section 3
209 // this section is with authenticated plain read and write access
210 sect_u32 = 3u;
211
212 sectCfg[sect_u32].baseAddr = sectCfg[sect_u32 - 1].baseAddr + sectCfg[sect_u32 - 1].size;
213 sectCfg[sect_u32].size = SIZE_256KB_;
214
215 sectCfg[sect_u32].sectionConfig.policy.plainAccessReadEnable = 1u;
216 sectCfg[sect_u32].sectionConfig.policy.plainAccessWriteEnable = 1u;
217 sectCfg[sect_u32].sectionConfig.policy.authPlainAccess = 1u;
218 sectCfg[sect_u32].sectionConfig.policy.checksumIntegrity = 0u;
219 sectCfg[sect_u32].sectionConfig.policy.digestIntegrity = 0u;
220 sectCfg[sect_u32].sectionConfig.policy.digestIntegrityOnAccess = 0u;
221 sectCfg[sect_u32].sectionConfig.policy.rollbackProt = 0u;
222 sectCfg[sect_u32].sectionConfig.policy.slog = 0u;
223 sectCfg[sect_u32].sectionConfig.policy.writeProt = 0u;
224
225 // get full access key to configure into chip.
226 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSectionAccessKey(ctx,
227 id.std.uniqueID,
228 QLIB_KID__FULL_ACCESS_SECTION,
229 sect_u32,
230 die_u8,
231 sectCfg[sect_u32].fullAccessKey));
232
233 // write configuration
234 QLIB_STATUS_RET_CHECK(QLIB_ConfigDeviceMultiDie(ctx, &flashCfg, &dieCfg, cfgLine, diesNum_u8));
235
236#if QLIB_MAX_DIES_SUPPORTED > 1u
237 // write configuration sets die to 0. return to orig die
238 QLIB_STATUS_RET_CHECK(QLIB_SetActiveDie(ctx, die_u8));
239#endif // QLIB_MAX_DIES_SUPPORTED > 1u
240
241 return QLIB_STATUS__OK;
242}

252QLIB_STATUS_T QLIB_SAMPLE_Erase_field_flow_plainErase(QLIB_CONTEXT_T* ctx)
253{
254 uint32_t sect_u32;
255 uint8_t buf_u8[QLIB_FLASH_PAGE_SIZE];
256 QLIB_STATUS_T status;
257
258 (void)memset(buf_u8, 0u, sizeof(buf_u8));
259
260 // Section 1 should be has plain access
261 sect_u32 = 1u;
262
263 // zero flash test area
264 QLIB_STATUS_RET_CHECK(QLIB_Write(ctx, buf_u8, sect_u32, 0u, QLIB_FLASH_PAGE_SIZE, (bool)false));
265
266 // plain erase to single sector is possible
267 QLIB_STATUS_RET_CHECK(QLIB_Erase(ctx, sect_u32, 0u, QLIB_FLASH_SECTOR_SIZE, (bool)false));
268
269 // plain erase to entire section_u32 is possible
270 QLIB_STATUS_RET_CHECK(QLIB_EraseSection(ctx, sect_u32, (bool)false));
271
272 // verify data
273 QLIB_STATUS_RET_CHECK(QLIB_Read(ctx, buf_u8, sect_u32, 0u, QLIB_FLASH_PAGE_SIZE, (bool)false, (bool)false));
274 for (uint32_t i = 0u; i < (uint32_t)QLIB_FLASH_PAGE_SIZE; i++)
275 {
276 QLIB_ASSERT_RET(0xFFu == buf_u8[i], QLIB_STATUS__TEST_FAIL);
277 }
278
279 // Section 2 has no write plain access, which means no plain erase is possible
280 sect_u32 = 2u;
281
282 QLIB_SAMPLE_ALLOW_TO_FAIL__START();
283 status = QLIB_Erase(ctx, sect_u32, 0u, QLIB_FLASH_SECTOR_SIZE, (bool)false);
284 QLIB_SAMPLE_ALLOW_TO_FAIL__END();
285 QLIB_ASSERT_RET(QLIB_STATUS__DEVICE_PRIVILEGE_ERR == status, QLIB_STATUS__TEST_FAIL);
286
287 QLIB_SAMPLE_ALLOW_TO_FAIL__START();
288 status = QLIB_EraseSection(ctx, sect_u32, (bool)false);
289 QLIB_SAMPLE_ALLOW_TO_FAIL__END();
290 QLIB_ASSERT_RET(QLIB_STATUS__DEVICE_PRIVILEGE_ERR == status, QLIB_STATUS__TEST_FAIL);
291
292 return QLIB_STATUS__OK;
293}

295QLIB_STATUS_T QLIB_SAMPLE_Erase_field_flow_secureErase(QLIB_CONTEXT_T* ctx)
296{
297 QLIB_HW_VER_T hwVer;
298 QLIB_ID_T id;
299 uint8_t die_u8 = 0u;
300 uint32_t section_u32;
301 CONST_KEY_P_T keyF;
302 CONST_KEY_P_T keyR;
303 QLIB_STATUS_T status;
304 uint32_t openedSessionSection_u32;
305 bool fullAccess_b;
306
307 // See what chip is connected and its unique ids. We might need this info to obtain keys for this
308 // chips according to formula or via DB see QLIB_SAMPLE_GetSectionAccessKey() function
309 QLIB_STATUS_RET_CHECK(QLIB_GetHWVersion(ctx, &hwVer));
310 QLIB_STATUS_RET_CHECK(QLIB_GetId(ctx, &id));
311
312#if QLIB_MAX_DIES_SUPPORTED > 1u
313 QLIB_STATUS_RET_CHECK(QLIB_GetActiveDie(ctx, &die_u8));
314#endif // QLIB_MAX_DIES_SUPPORTED > 1u
315
316 /*-------------------------------------------------------------------------------------------------------
317 Section 2 supports secure write/erase with full access key
318 -------------------------------------------------------------------------------------------------------*/
319 section_u32 = 2u;
320
321 /*-------------------------------------------------------------------------------------------------------
322 plain access attempt should fail
323 -------------------------------------------------------------------------------------------------------*/
324 QLIB_SAMPLE_ALLOW_TO_FAIL__START();
325 status = QLIB_EraseSection(ctx, section_u32, (bool)false);
326 QLIB_SAMPLE_ALLOW_TO_FAIL__END();
327 QLIB_ASSERT_RET(QLIB_STATUS__DEVICE_PRIVILEGE_ERR == status, QLIB_STATUS__TEST_FAIL);
328
329 /*-------------------------------------------------------------------------------------------------------
330 obtain access keys and load the keys to library
331 -------------------------------------------------------------------------------------------------------*/
332 QLIB_STATUS_RET_CHECK(
333 QLIB_SAMPLE_GetSectionKeyPtr(ctx, id.std.uniqueID, QLIB_KID__RESTRICTED_ACCESS_SECTION, section_u32, die_u8, &keyR));
334 QLIB_STATUS_RET_CHECK(
335 QLIB_SAMPLE_GetSectionKeyPtr(ctx, id.std.uniqueID, QLIB_KID__FULL_ACCESS_SECTION, section_u32, die_u8, &keyF));
336
337 QLIB_STATUS_RET_CHECK(QLIB_LoadKey(ctx, section_u32, keyR, (bool)false));
338 QLIB_STATUS_RET_CHECK_GOTO(QLIB_LoadKey(ctx, section_u32, keyF, (bool)true), status, remove_key);
339
340 /*-------------------------------------------------------------------------------------------------------
341 Check that no session is opened, should return UINT32_MAX as openedSessionSection_u32
342 -------------------------------------------------------------------------------------------------------*/
343 QLIB_STATUS_RET_CHECK_GOTO(QLIB_CurrOpenedSession(ctx, die_u8, &openedSessionSection_u32, &fullAccess_b), status, remove_key);
344 QLIB_ASSERT_RET(UINT32_MAX == openedSessionSection_u32, QLIB_STATUS__TEST_FAIL);
345
346 /*-------------------------------------------------------------------------------------------------------
347 Open restricted access secure session for secure read
348 -------------------------------------------------------------------------------------------------------*/
349 QLIB_STATUS_RET_CHECK_GOTO(QLIB_OpenSession(ctx, section_u32, QLIB_SESSION_ACCESS_RESTRICTED), status, remove_key);
350
351 /*-------------------------------------------------------------------------------------------------------
352 Check that a secure session is opened, should return the section opened and if it is fullAccess_b or not
353 -------------------------------------------------------------------------------------------------------*/
354 QLIB_STATUS_RET_CHECK_GOTO(QLIB_CurrOpenedSession(ctx, die_u8, &openedSessionSection_u32, &fullAccess_b), status, remove_key);
355 QLIB_ASSERT_RET(section_u32 == openedSessionSection_u32, QLIB_STATUS__TEST_FAIL);
356 QLIB_ASSERT_RET(false == fullAccess_b, QLIB_STATUS__TEST_FAIL);
357
358 /*-------------------------------------------------------------------------------------------------------
359 section_u32 can not be erased via restricted session
360 -------------------------------------------------------------------------------------------------------*/
361 QLIB_SAMPLE_ALLOW_TO_FAIL__START();
362 status = QLIB_EraseSection(ctx, section_u32, (bool)true);
363 QLIB_SAMPLE_ALLOW_TO_FAIL__END();
364 QLIB_ASSERT_WITH_ERROR_GOTO(QLIB_STATUS__DEVICE_SESSION_ERR == status, QLIB_STATUS__TEST_FAIL, status, close_session);
365
366 QLIB_STATUS_RET_CHECK_GOTO(QLIB_CloseSession(ctx, section_u32), status, remove_key);
367
368 /*-------------------------------------------------------------------------------------------------------
369 Open full access secure session for secure write
370 -------------------------------------------------------------------------------------------------------*/
371 QLIB_STATUS_RET_CHECK_GOTO(QLIB_OpenSession(ctx, section_u32, QLIB_SESSION_ACCESS_FULL), status, remove_key);
372
373 /*-------------------------------------------------------------------------------------------------------
374 Erase access is allowed
375 -------------------------------------------------------------------------------------------------------*/
376 QLIB_STATUS_RET_CHECK_GOTO(QLIB_EraseSection(ctx, section_u32, (bool)true), status, close_session);
377
378 QLIB_STATUS_RET_CHECK_GOTO(QLIB_CloseSession(ctx, section_u32), status, remove_key);
379
380 status = QLIB_STATUS__OK;
381 goto remove_key;
382
383close_session:
384 (void)QLIB_CloseSession(ctx, section_u32);
385
386remove_key:
387 (void)QLIB_RemoveKey(ctx, section_u32, (bool)true);
388 (void)QLIB_RemoveKey(ctx, section_u32, (bool)false);
389
390 return status;
391}

393QLIB_STATUS_T QLIB_SAMPLE_Erase_field_flow_authPlainErase(QLIB_CONTEXT_T* ctx)
394{
395 QLIB_HW_VER_T hwVer;
396 QLIB_ID_T id;
397 uint8_t die_u8 = 0u;
398 uint32_t section_u32;
399 CONST_KEY_P_T keyF;
400 QLIB_STATUS_T status = QLIB_STATUS__OK;
401 uint8_t writeBuf_u8[QLIB_FLASH_PAGE_SIZE];
402 uint8_t readBuf_u8[QLIB_FLASH_PAGE_SIZE];
403 uint8_t eraseBuf_u8[QLIB_FLASH_PAGE_SIZE];
404
405 (void)memset(readBuf_u8, 0u, sizeof(writeBuf_u8));
406 (void)memset(writeBuf_u8, 0u, sizeof(writeBuf_u8));
407 (void)memset(eraseBuf_u8, 0xFF, sizeof(eraseBuf_u8));
408
409 // See what chip is connected and its unique ids. We might need this info to obtain keys for this
410 // chips according to formula or via DB see QLIB_SAMPLE_GetSectionKeyPtr() function
411 QLIB_STATUS_RET_CHECK(QLIB_GetHWVersion(ctx, &hwVer));
412 QLIB_STATUS_RET_CHECK(QLIB_GetId(ctx, &id));
413
414#if QLIB_MAX_DIES_SUPPORTED > 1u
415 QLIB_STATUS_RET_CHECK(QLIB_GetActiveDie(ctx, &die_u8));
416#endif // QLIB_MAX_DIES_SUPPORTED > 1u
417
418 /*-------------------------------------------------------------------------------------------------------
419 Section 3 should be authenticated plain read and write
420 -------------------------------------------------------------------------------------------------------*/
421 section_u32 = 3u;
422
423 /*-------------------------------------------------------------------------------------------------------
424 obtain full access key and load the key to library
425 -------------------------------------------------------------------------------------------------------*/
426 QLIB_STATUS_RET_CHECK(
427 QLIB_SAMPLE_GetSectionKeyPtr(ctx, id.std.uniqueID, QLIB_KID__FULL_ACCESS_SECTION, section_u32, die_u8, &keyF));
428
429 QLIB_STATUS_RET_CHECK(QLIB_LoadKey(ctx, section_u32, keyF, (bool)true));
430
431 /*-------------------------------------------------------------------------------------------------------
432 Open full access secure session for the secure commands
433 -------------------------------------------------------------------------------------------------------*/
434 QLIB_STATUS_RET_CHECK_GOTO(QLIB_OpenSession(ctx, section_u32, QLIB_SESSION_ACCESS_FULL), status, remove_key);
435
436 /*-------------------------------------------------------------------------------------------------------
437 Secure write
438 -------------------------------------------------------------------------------------------------------*/
439 QLIB_STATUS_RET_CHECK_GOTO(QLIB_Write(ctx, writeBuf_u8, section_u32, 0u, (uint32_t)sizeof(writeBuf_u8), (bool)true),
440 status,
441 close_session);
442
443 /*-------------------------------------------------------------------------------------------------------
444 Close the secure session
445 -------------------------------------------------------------------------------------------------------*/
446 QLIB_STATUS_RET_CHECK_GOTO(QLIB_CloseSession(ctx, section_u32), status, remove_key);
447
448 /*-------------------------------------------------------------------------------------------------------
449 Non-secure Erase for authenticated plain access write section is still allowed after session was closed
450 -------------------------------------------------------------------------------------------------------*/
451 QLIB_STATUS_RET_CHECK_GOTO(QLIB_Erase(ctx, section_u32, 0u, QLIB_FLASH_SECTOR_SIZE, (bool)false), status, remove_key);
452
453 /*-------------------------------------------------------------------------------------------------------
454 Verify data
455 -------------------------------------------------------------------------------------------------------*/
456 QLIB_STATUS_RET_CHECK_GOTO(QLIB_Read(ctx,
457 readBuf_u8,
458 section_u32,
459 0u,
460 (uint32_t)sizeof(readBuf_u8),
461 (bool)false,
462 (bool)false),
463 status,
464 remove_key);
465 QLIB_ASSERT_WITH_ERROR_GOTO(0u == memcmp(readBuf_u8, eraseBuf_u8, sizeof(eraseBuf_u8)),
466 QLIB_STATUS__TEST_FAIL,
467 status,
468 remove_key);
469
470 /*-------------------------------------------------------------------------------------------------------
471 Revoke Plain Access privileges
472 -------------------------------------------------------------------------------------------------------*/
473 QLIB_STATUS_RET_CHECK_GOTO(QLIB_PlainAccessRevoke(ctx, section_u32, QLIB_PA_REVOKE_ALL_ACCESS), status, remove_key);
474
475 /*-------------------------------------------------------------------------------------------------------
476 In W77Q security version 2 flash, non-secure erase for authenticated plain access section is not allowed
477 after revoke.
478 In W77Q/T security version 3 flash, the erase function implicitly grants plain access.
479 -------------------------------------------------------------------------------------------------------*/
480 QLIB_SAMPLE_ALLOW_TO_FAIL__START();
481 status = QLIB_Erase(ctx, section_u32, 0u, QLIB_FLASH_SECTOR_SIZE, (bool)false);
482 QLIB_SAMPLE_ALLOW_TO_FAIL__END();
483 QLIB_ASSERT_WITH_ERROR_GOTO(status == (QLIB_FeatureSupported(ctx, W77Q_FEATURE_PA_GRANT_REVOKE) == (bool)false
484 ? QLIB_STATUS__DEVICE_PRIVILEGE_ERR
485 : QLIB_STATUS__OK),
486 QLIB_STATUS__TEST_FAIL,
487 status,
488 remove_key);
489
490 status = QLIB_STATUS__OK;
491 goto remove_key;
492
493close_session:
494 (void)QLIB_CloseSession(ctx, section_u32);
495
496remove_key:
497 (void)QLIB_RemoveKey(ctx, section_u32, (bool)true);
498
499 return status;
500}
