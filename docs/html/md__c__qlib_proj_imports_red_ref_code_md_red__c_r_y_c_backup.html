// context of the monotonic counter (MC) counter over w77q flash.
typedef struct _mc_w77q32
{
bool read_secure; // if true read operation is protected, false otherwise
bool write_secure; // if true read operation is protected, false otherwise
uint8_t w77q_section; // section index MC resides on
QLIB_CONTEXT_T *qlibContext; // qlib context needed to operate w77q flash
} MC_W77Q_CONTEXT;
//
// 3 callbacks needed to be implemented, interface is defined in imports/flash_access.h in FMC release
//
DEFS_STATUS FMC_PLAT_FLASH_Write(void* flashHandle, const uint32_t addr, const void* data, uint32_t size)
{
MC_W77Q_CONTEXT *ctx = (MC_W77Q_CONTEXT *)flashHandle;
DEFS_STATUS_COND_CHECK(ctx, DEFS_STATUS_INVALID_PARAMETER);
QLIB_STATUS_T status = QLIB_Write(ctx->qlibContext, (uint8_t*)data, ctx->w77q_section, addr, size, ctx->write_secure);
ASSERT_MSG_RET( QLIB_STATUS__OK == status, DEFS_STATUS_FAIL,"QLIB_Write failed with qlib status %d\n ", status);
return DEFS_STATUS_OK;
}
DEFS_STATUS FMC_PLAT_FLASH_Read(void* flashHandle, const uint32_t addr, void* data, uint32_t size)
{
MC_W77Q_CONTEXT *ctx = (MC_W77Q_CONTEXT *)flashHandle;
DEFS_STATUS_COND_CHECK(ctx, DEFS_STATUS_INVALID_PARAMETER);
QLIB_STATUS_T status = QLIB_Read( ctx->qlibContext, (uint8_t*)data, ctx->w77q_section, addr, size, ctx->read_secure, true);
ASSERT_MSG_RET( QLIB_STATUS__OK == status, DEFS_STATUS_FAIL,"QLIB_Read failed with qlib status %d\n ", status);
return DEFS_STATUS_OK;
}
DEFS_STATUS FMC_PLAT_FLASH_SectorErase(void* flashHandle, const uint32_t addr)
{
MC_W77Q_CONTEXT *ctx = (MC_W77Q_CONTEXT *)flashHandle;
DEFS_STATUS_COND_CHECK(ctx, DEFS_STATUS_INVALID_PARAMETER);
QLIB_STATUS_T status = QLIB_Erase( ctx->qlibContext, ctx->w77q_section, addr, FMC_FLASH_SECTOR_SIZE, ctx->write_secure);
ASSERT_MSG_RET( QLIB_STATUS__OK == status, DEFS_STATUS_FAIL,"QLIB_Erase failed with qlib status %d\n ", status);
return DEFS_STATUS_OK;
}
// flash configuration function which configures 2 sections corresponding to the 2 MC we want to work with
QLIB_STATUS_T w77q_config_flash(QLIB_CONTEXT_T* ctx)
{
QLIB_ID_T id;
uint32_t sect = 0, die = 0;
QLIB_FLASH_CONFIG_T flashCfg = {0};
QLIB_DIE_CONFIG_T dieCfg = {0};
QLIB_SECTIONS_CONF_TABLE_T cfgLine[QLIB_MAX_DIES_SUPPORTED];
uint8_t diesNum = QLIB_GET_NUM_OF_DIES(ctx);
// See what chip is connected and its unique ids. We might need this info to obtain keys for this
// chips according to formula or via DB see fwGetSectionKey() function
QLIB_STATUS_RET_CHECK(QLIB_GetId(ctx, &id));
// We do read / modify /write configuration since we want to change only what is needed
// read configuration
QLIB_STATUS_RET_CHECK(QLIB_GetDeviceConfigMultiDie( ctx, &flashCfg, &dieCfg, cfgLine, diesNum));
QLIB_EXTENDED_SECTION_CONF_T * sectCfg = cfgLine[die].sectionConfigTable;
// modify configuration
// set pre provision master key if needed
QLIB_STATUS_RET_CHECK(fwGetMasterKey(ctx, &id, QLIB_KID__DEVICE_KEY_PRE_PROVISIONING, dieCfg.preProvisionedMasterKey));
// set new master key
QLIB_STATUS_RET_CHECK(fwGetMasterKey(ctx, &id, QLIB_KID__DEVICE_MASTER, dieCfg.deviceMasterKey));
// setting layout and policy of the section 0
// this section plain read enabled and plain write enabled
sect = 0;
sectCfg[sect].baseAddr = 0;
sectCfg[sect].size = _1MB_;
sectCfg[sect].sectionConfig.policy.plainAccessReadEnable = 1;
sectCfg[sect].sectionConfig.policy.plainAccessWriteEnable = 1;
// Full access key configuration.This is the key to use later for the section configuration
// changes or write access
QLIB_STATUS_RET_CHECK( fwGetSectionKey(
ctx, &id, QLIB_KID__FULL_ACCESS_SECTION, sect, die, sectCfg[sect].fullAccessKey, sizeof(KEY_T)));
// setting layout of the section 1
// this section plain read enabled and plain write disabled
sect = 1;
sectCfg[sect].baseAddr = _1MB_;
sectCfg[sect].size = _1MB_;
sectCfg[sect].sectionConfig.policy.plainAccessReadEnable = 1;
sectCfg[sect].sectionConfig.policy.plainAccessWriteEnable = 0;
// get full access key to configure into chip.
QLIB_STATUS_RET_CHECK( fwGetSectionKey(
ctx, &id, QLIB_KID__FULL_ACCESS_SECTION, sect, die, sectCfg[sect].fullAccessKey, sizeof(KEY_T)));
// write configuration
QLIB_STATUS_RET_CHECK(QLIB_ConfigDeviceMultiDie(ctx, &flashCfg, &dieCfg, cfgLine, diesNum));
return QLIB_STATUS__OK;
}
// opens secure session to the flash section, thus enabling work with protected section
QLIB_STATUS_T w77q_session_start(QLIB_CONTEXT_T* ctx, uint32_t sect, KEY_T key, QLIB_KID_TYPE_T keyType)
{
QLIB_ASSERT_RET(keyType == QLIB_KID__RESTRICTED_ACCESS_SECTION ||
keyType == QLIB_KID__FULL_ACCESS_SECTION, QLIB_STATUS__INVALID_PARAMETER);
bool fullAccess = (keyType == QLIB_KID__FULL_ACCESS_SECTION) ? true : false;
// load key pointer to library, open session
QLIB_STATUS_RET_CHECK(QLIB_LoadKey(ctx, sect, key, fullAccess));
QLIB_STATUS_RET_CHECK(QLIB_OpenSession(ctx, sect, fullAccess ? QLIB_SESSION_ACCESS_FULL : QLIB_SESSION_ACCESS_RESTRICTED));
return QLIB_STATUS__OK;
}
// closes secure session to the flash section
QLIB_STATUS_T w77q_session_stop(QLIB_CONTEXT_T* ctx, uint32_t sect, QLIB_KID_TYPE_T keyType)
{
QLIB_ASSERT_RET(keyType == QLIB_KID__RESTRICTED_ACCESS_SECTION ||
keyType == QLIB_KID__FULL_ACCESS_SECTION, QLIB_STATUS__INVALID_PARAMETER);
bool fullAccess = (keyType == QLIB_KID__FULL_ACCESS_SECTION) ? true : false;
// close secure session and remove key pointer from the library
QLIB_STATUS_RET_CHECK(QLIB_CloseSession(ctx, sect));
QLIB_STATUS_RET_CHECK(QLIB_RemoveKey(ctx, sect, fullAccess));
return QLIB_STATUS__OK;
}
// function which checks incrementation of the MC, secure or not
QLIB_STATUS_T w77q_mc_check(FMC_MC_T *mc, KEY_T key)
{
uint32_t counterVal = 0;
uint32_t counterValNext = 0;
MC_W77Q_CONTEXT *mc_ctx = (MC_W77Q_CONTEXT *)mc->flashContext;
QLIB_KID_TYPE_T keyType;
// if key is provided, we treat it as a secure counter and open secure session according to security policy
if (key)
{
keyType = mc_ctx->write_secure ? QLIB_KID__FULL_ACCESS_SECTION : QLIB_KID__RESTRICTED_ACCESS_SECTION;
QLIB_STATUS_RET_CHECK(w77q_session_start(mc_ctx->qlibContext, mc_ctx->w77q_section, key, keyType));
}
// using the counter
ASSERT_MSG_RET(DEFS_STATUS_OK == FMC_MC_Init(mc), QLIB_STATUS__COMMAND_FAIL, "FMC_MC_Init fail \n");
ASSERT_MSG_RET(DEFS_STATUS_OK == FMC_MC_Get(mc, &counterVal), QLIB_STATUS__COMMAND_FAIL, "FMC_MC_Get fail \n");
ASSERT_MSG_RET(DEFS_STATUS_OK == FMC_MC_Inc(mc), QLIB_STATUS__COMMAND_FAIL, "FMC_MC_Inc fail \n");
ASSERT_MSG_RET(DEFS_STATUS_OK == FMC_MC_Get(mc, &counterValNext), QLIB_STATUS__COMMAND_FAIL, "FMC_MC_Get fail \n");
if (key)
{
QLIB_STATUS_RET_CHECK(w77q_session_stop(mc_ctx->qlibContext, mc_ctx->w77q_section, keyType));
}
ASSERT_MSG_RET(counterVal + 1 == counterValNext, QLIB_STATUS__COMMAND_FAIL, "MC counters' check fail \n");
return QLIB_STATUS__OK;;
}
// formats MC, secire or not
QLIB_STATUS_T w77q_mc_format(FMC_MC_T *mc, KEY_T key)
{
MC_W77Q_CONTEXT *mc_ctx = NULL;
QLIB_KID_TYPE_T keyType;
// if key is provided, we treat it as a secure counter and open secure session according to security policy
if (key)
{
mc_ctx = (MC_W77Q_CONTEXT *)mc->flashContext;
keyType = mc_ctx->write_secure ? QLIB_KID__FULL_ACCESS_SECTION : QLIB_KID__RESTRICTED_ACCESS_SECTION;
QLIB_STATUS_RET_CHECK(w77q_session_start(mc_ctx->qlibContext, mc_ctx->w77q_section, key, keyType));
}
// using the counter
ASSERT_MSG_RET(DEFS_STATUS_OK == FMC_MC_Format(mc, true), QLIB_STATUS__COMMAND_FAIL, "FMC_MC_Format fail \n");
if (key)
{
QLIB_STATUS_RET_CHECK(w77q_session_stop(mc_ctx->qlibContext, mc_ctx->w77q_section, keyType));
}
return QLIB_STATUS__OK;
}
// full manufactire flow for MC over w77q
QLIB_STATUS_T red_CRY_manuf_flow(QLIB_CONTEXT_T* ctx)
{
QLIB_STATUS_T qlib_ret;
DEFS_STATUS fmc_ret;
KEY_T keyF;
QLIB_ID_T id;
// See what chip is connected and its unique ids. We might need this info to obtain keys for this
// chips according to formula or via DB see fwGetSectionKey() function
QLIB_STATUS_RET_CHECK(QLIB_GetId(ctx, &id));
// in this example we will build three counters with different permissions.To do that we
// will configure 3 sections over flash with different policy which corresponds to the
// counter policy.
// section# | 0 | 1 |
//-----------------------------------------------------------------|
// size | 1M | 1M |
// policy | Plain Read / Plain Write | Plain Read / Secure Write |
ASSERT_MSG_RET(QLIB_STATUS__OK == (qlib_ret = w77q_config_flash(ctx)), qlib_ret, "cfg_three_sections failed\n");
// non protected mc0 will be using non protected section 0 of flash, taking 2 sectors from the beginning ( offset 0 ).
// there might be more counters in the same section each taking two sectors aka 2*4K bytes
FMC_MC_T mc0;
MC_W77Q_CONTEXT mc0_ctx = { false, false, 0, ctx };
fmc_ret = FMC_MC_SetParams(&mc0, 0x0, &mc0_ctx);
ASSERT_MSG_RET(DEFS_STATUS_OK == fmc_ret, QLIB_STATUS__COMMAND_FAIL, "FMC_MC_SetParams fail \n");
// formatting counter requires write permission
// since counter is over plain read/ plain write section, no secure session is needed, so key is NULL
QLIB_STATUS_RET_CHECK(w77q_mc_format(&mc0, NULL));
// mc1 will be using write protected section 1 of flash, taking 2 sectors from the beginning ( offset 0 ).
// there might be more counters in the same section each taking two sectors aka 2*4K bytes
//
FMC_MC_T mc1;
MC_W77Q_CONTEXT mc1_ctx = { false, true, 1, ctx};
fmc_ret = FMC_MC_SetParams(&mc1, 0x0, &mc1_ctx);
ASSERT_MSG_RET(DEFS_STATUS_OK == fmc_ret, QLIB_STATUS__COMMAND_FAIL, "FMC_MC_SetParams fail \n");
// formatting counter requires write permission
// since counter is secure write section, full access key is needed
QLIB_STATUS_RET_CHECK(fwGetSectionKey(ctx, &id, QLIB_KID__FULL_ACCESS_SECTION, 1, 0, keyF, sizeof(KEY_T)));
QLIB_STATUS_RET_CHECK(w77q_mc_format(&mc1, keyF));
return QLIB_STATUS__OK;
}
// full field flow for MC over w77q
QLIB_STATUS_T red_CRY_field_flow( QLIB_CONTEXT_T* ctx)
{
DEFS_STATUS fmc_ret;
QLIB_ID_T id;
KEY_T keyF;
// See what chip is connected and its unique ids. We might need this info to obtain keys for this
// chips according to formula or via DB see fwGetSectionKey() function
QLIB_STATUS_RET_CHECK(QLIB_GetId(ctx, &id));
// mc0 configuration is as previously configured.
FMC_MC_T mc0;
MC_W77Q_CONTEXT mc0_ctx = { false, false, 0, ctx};
fmc_ret = FMC_MC_SetParams(&mc0, 0x0, &mc0_ctx);
ASSERT_MSG_RET(DEFS_STATUS_OK == fmc_ret, QLIB_STATUS__COMMAND_FAIL, "FMC_MC_SetParams fail \n");
// this is non protected counter so key is NULL
ASSERT_MSG_RET(QLIB_STATUS__OK == w77q_mc_check(&mc0, NULL), QLIB_STATUS__COMMAND_FAIL,"w77q_mc_check mc0 fail\n");
// mc1 will is located over protected flash so it it is secure counter which requres key
FMC_MC_T mc1;
MC_W77Q_CONTEXT mc1_ctx = { false, true, 1, ctx};
fmc_ret = FMC_MC_SetParams(&mc1, 0x0, &mc1_ctx);
ASSERT_MSG_RET(DEFS_STATUS_OK == fmc_ret, QLIB_STATUS__COMMAND_FAIL, "FMC_MC_SetParams fail \n");
// if we try without the key should fail
ASSERT_MSG_RET(QLIB_STATUS__OK != w77q_mc_check(&mc1, NULL), QLIB_STATUS__COMMAND_FAIL, "w77q_mc_check mc1 neg. test fail \n");
// obtain full, access key and use it
QLIB_STATUS_RET_CHECK(fwGetSectionKey(ctx, &id, QLIB_KID__FULL_ACCESS_SECTION, 1, 0, keyF, sizeof(KEY_T)));
ASSERT_MSG_RET(QLIB_STATUS__OK == w77q_mc_check(&mc1, keyF), QLIB_STATUS__COMMAND_FAIL, "w77q_mc_check mc1 fail\n");
return QLIB_STATUS__OK;
}
