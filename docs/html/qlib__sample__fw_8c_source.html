42QLIB_STATUS_T QLIB_SAMPLE_GetSectionAccessKey(QLIB_CONTEXT_T* ctx,
43 QLIB_WID_T wid,
44 QLIB_KID_TYPE_T keyType,
45 uint32_t section_u32,
46 uint32_t die_u32,
47 KEY_T key)
48{
49 QLIB_ASSERT_RET((keyType == QLIB_KID__FULL_ACCESS_SECTION) || (keyType == QLIB_KID__RESTRICTED_ACCESS_SECTION),
50 QLIB_STATUS__INVALID_PARAMETER);
51 QLIB_ASSERT_RET(section_u32 < QLIB_NUM_OF_SECTIONS, QLIB_STATUS__INVALID_PARAMETER);
52 QLIB_ASSERT_RET(key != NULL, QLIB_STATUS__INVALID_PARAMETER);
53 QLIB_ASSERT_RET(ctx != NULL, QLIB_STATUS__INVALID_PARAMETER);
54 QLIB_ASSERT_RET(die_u32 < QLIB_GET_NUM_OF_DIES(ctx), QLIB_STATUS__INVALID_PARAMETER);
55 QLIB_ASSERT_RET(wid != NULL, QLIB_STATUS__INVALID_PARAMETER);
56#ifdef QLIB_KDF_DEV_KEYS
57 QLIB_ASSERT_RET(QLIB_UTILS_CMKDF_KeyDerive(devSecret_au8_L,
58 (uint8_t*)wid,
59 keyType == QLIB_KID__FULL_ACCESS_SECTION ? QLIB_UTILS_CMKDF_LABEL_FK
60 : QLIB_UTILS_CMKDF_LABEL_RK,
61 (QLIB_NUM_OF_SECTIONS * die_u32) + section_u32,
62 QLIB_NUM_OF_SECTIONS * QLIB_GET_NUM_OF_DIES(ctx),
63 (uint8_t*)key,
64 sizeof(KEY_T)) == 0,
65 QLIB_STATUS__COMMAND_FAIL);
66#else
67 {
68 static const KEY_ARRAY_T rKeys = {QLIB_SAMPLE_RESTRICTED_K_0,
69 QLIB_SAMPLE_RESTRICTED_K_1,
70 QLIB_SAMPLE_RESTRICTED_K_2,
71 QLIB_SAMPLE_RESTRICTED_K_3,
72 QLIB_SAMPLE_RESTRICTED_K_4,
73 QLIB_SAMPLE_RESTRICTED_K_5,
74 QLIB_SAMPLE_RESTRICTED_K_6,
75 QLIB_SAMPLE_RESTRICTED_K_7,
76 QLIB_SAMPLE_VAULT_RESTRICTED};
77 static const KEY_ARRAY_T fKeys = {QLIB_SAMPLE_FULL_ACCESS_K_0,
78 QLIB_SAMPLE_FULL_ACCESS_K_1,
79 QLIB_SAMPLE_FULL_ACCESS_K_2,
80 QLIB_SAMPLE_FULL_ACCESS_K_3,
81 QLIB_SAMPLE_FULL_ACCESS_K_4,
82 QLIB_SAMPLE_FULL_ACCESS_K_5,
83 QLIB_SAMPLE_FULL_ACCESS_K_6,
84 QLIB_SAMPLE_FULL_ACCESS_K_7,
85 QLIB_SAMPLE_VAULT_FULL_ACCESS};
86
87 (void)memcpy(key, keyType == QLIB_KID__FULL_ACCESS_SECTION ? fKeys[section_u32] : rKeys[section_u32], sizeof(KEY_T));
88 }
89#endif // QLIB_KDF_DEV_KEYS
90 return QLIB_STATUS__OK;
91}

123{
124 QLIB_ASSERT_RET(ctx != NULL, QLIB_STATUS__INVALID_PARAMETER);
125 QLIB_ASSERT_RET(wid != NULL, QLIB_STATUS__INVALID_PARAMETER);
126 QLIB_ASSERT_RET(deviceKey != NULL, QLIB_STATUS__INVALID_PARAMETER);
127
128 if (keyType == QLIB_KID__DEVICE_KEY_PRE_PROVISIONING)
129 {
130 if (QLIB_FeatureSupported(ctx, W77Q_FEATURE_PRE_PROV_MASTER_KEY) != (bool)false)
131 {
132#ifdef W77Q_KDF_KD1
133 QLIB_ASSERT_RET(QLIB_UTILS_CMKDF_KeyDerive(cms_au8_L,
134 (uint8_t*)wid,
135 QLIB_UTILS_CMKDF_LABEL_KD1,
136 0u,
137 1u,
138 (uint8_t*)deviceKey,
139 sizeof(KEY_T)) == 0,
140 QLIB_STATUS__COMMAND_FAIL);
141#else
142 KEY_T key = QLIB_SAMPLE_KD1;
143 (void)memcpy(deviceKey, key, sizeof(KEY_T));
144#endif // W77Q_KDF_KD1
145 }
146 else
147 {
148 (void)memset(deviceKey, 0u, sizeof(KEY_T));
149 }
150 }
151 else if (keyType == QLIB_KID__DEVICE_MASTER)
152 {
153#ifdef QLIB_KDF_DEV_KEYS
154 QLIB_ASSERT_RET(QLIB_UTILS_CMKDF_KeyDerive(devSecret_au8_L,
155 (uint8_t*)wid,
156 QLIB_UTILS_CMKDF_LABEL_KD,
157 0u,
158 1u,
159 (uint8_t*)deviceKey,
160 sizeof(KEY_T)) == 0,
161 QLIB_STATUS__COMMAND_FAIL);
162#else
163 KEY_T key = QLIB_SAMPLE_KD;
164 (void)memcpy(deviceKey, key, sizeof(KEY_T));
165#endif // QLIB_KDF_DEV_KEYS
166 }
167 else if (keyType == QLIB_KID__DEVICE_SECRET)
168 {
169#ifdef QLIB_KDF_DEV_KEYS
170 QLIB_ASSERT_RET(QLIB_UTILS_CMKDF_KeyDerive(devSecret_au8_L,
171 (uint8_t*)wid,
172 QLIB_UTILS_CMKDF_LABEL_KS,
173 0u,
174 1u,
175 (uint8_t*)deviceKey,
176 sizeof(KEY_T)) == 0,
177 QLIB_STATUS__COMMAND_FAIL);
178#else
179 KEY_T key = QLIB_SAMPLE_KDS;
180 (void)memcpy(deviceKey, key, sizeof(KEY_T));
181#endif // QLIB_KDF_DEV_KEYS
182 }
183 else
184 {
185 return QLIB_STATUS__INVALID_PARAMETER;
186 }
187 return QLIB_STATUS__OK;
188}
