88QLIB_STATUS_T QLIB_SAMPLE_GetSectionKeyPtr(QLIB_CONTEXT_T* ctx,
89 UNIQUE_ID_T uid,
90 QLIB_KID_TYPE_T keyType,
91 uint32_t section_u32,
92 uint32_t die_u32,
93 CONST_KEY_P_T* key)
94{
95 (void)ctx;
96 (void)uid;
97 (void)die_u32;
98
99 QLIB_ASSERT_RET((keyType == QLIB_KID__FULL_ACCESS_SECTION) || (keyType == QLIB_KID__RESTRICTED_ACCESS_SECTION),
100 QLIB_STATUS__INVALID_PARAMETER);
101 QLIB_ASSERT_RET(section_u32 < QLIB_NUM_OF_SECTIONS, QLIB_STATUS__INVALID_PARAMETER);
102 QLIB_ASSERT_RET(key != NULL, QLIB_STATUS__INVALID_PARAMETER);
103
104 if (keyType == QLIB_KID__FULL_ACCESS_SECTION)
105 {
106 *key = fKeys_L[section_u32];
107 }
108 else
109 {
110 *key = rKeys_L[section_u32];
111 }
112
113 return QLIB_STATUS__OK;
114}

118{
119 (void)uid;
120 if (keyType == QLIB_KID__DEVICE_KEY_PRE_PROVISIONING)

123 {
124 KEY_T key = QLIB_SAMPLE_KD1;
125 (void)memcpy(deviceKey, key, sizeof(KEY_T));
126 }
127 else
128 {
129 (void)memset(deviceKey, 0u, sizeof(KEY_T));
130 }
131 }
132 else if (keyType == QLIB_KID__DEVICE_MASTER)
133 {
134 KEY_T key = QLIB_SAMPLE_KD;
135 (void)memcpy(deviceKey, key, sizeof(KEY_T));
136 }
137 else if (keyType == QLIB_KID__DEVICE_SECRET)
138 {
139 KEY_T key = QLIB_SAMPLE_KDS;
140 (void)memcpy(deviceKey, key, sizeof(KEY_T));
141 }
142 else
143 {
144 return QLIB_STATUS__INVALID_PARAMETER;
145 }
146 return QLIB_STATUS__OK;
147}
