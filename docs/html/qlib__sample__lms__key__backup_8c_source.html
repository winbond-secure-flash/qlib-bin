101QLIB_STATUS_T QLIB_SAMPLE_LmsKeyBackup(void* userData_pv)
102{
103 QLIB_CONTEXT_T qlibContext;
104 QLIB_STATUS_T status = QLIB_STATUS__OK;
105 QLIB_BUS_FORMAT_T busFormat = QLIB_BUS_FORMAT(QLIB_BUS_MODE_1_1_1, (bool)false);
106
107 /*-------------------------------------------------------------------------------------------------------
108 Init QLIB - needed when running QLIB either on a device or on a remote server.
109 -------------------------------------------------------------------------------------------------------*/
110 QLIB_STATUS_RET_CHECK(QLIB_InitLib(&qlibContext));
111
112 /*-------------------------------------------------------------------------------------------------------
113 Set the user data into Qlib's context
114 -------------------------------------------------------------------------------------------------------*/
115 QLIB_SetUserData(&qlibContext, userData_pv);
116
117 /*-------------------------------------------------------------------------------------------------------
118 Take the ownership of flash communication channel (it belongs to local MCU or to remote server, exclusively)
119 -------------------------------------------------------------------------------------------------------*/
120 QLIB_STATUS_RET_CHECK(QLIB_Connect(&qlibContext));
121
122 /*-------------------------------------------------------------------------------------------------------
123 Init Flash Device - not needed when using QLIB on a remote server
124 -------------------------------------------------------------------------------------------------------*/
125 QLIB_STATUS_RET_CHECK_GOTO(QLIB_InitDevice(&qlibContext, busFormat), status, disconnect);
126
127 /*-------------------------------------------------------------------------------------------------------
128 configure flash for sample flow
129 -------------------------------------------------------------------------------------------------------*/
130 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_LmsKeyBackup_manuf_flow(&qlibContext), status, disconnect);
131
132 /*-------------------------------------------------------------------------------------------------------
133 Run sample code
134 -------------------------------------------------------------------------------------------------------*/
135 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_LmsKeyBackup_field_flow(&qlibContext), status, disconnect);
136
137#ifdef LMS_SAMPLE_PREPARE_IN_ADVANCE
138 // we can change the keys multiple times if we prepared a backup key when the last backup is used
139 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_LmsKeyBackup_field_flow(&qlibContext), status, disconnect);
140#endif // LMS_SAMPLE_PREPARE_IN_ADVANCE
141
142disconnect:
143 /*-------------------------------------------------------------------------------------------------------
144 Release the ownership of flash communication channel (it belongs to local MCU or to remote server, exclusively)
145 -------------------------------------------------------------------------------------------------------*/
146 (void)QLIB_Disconnect(&qlibContext);
147
148 return status;
149}

151QLIB_STATUS_T QLIB_SAMPLE_LmsKeyBackup_manuf_flow(QLIB_CONTEXT_T* qlibContext)
152{
153 QLIB_HW_VER_T hwVer;
154 QLIB_ID_T id;
155 uint8_t die_u8 = 0u;
156 uint32_t sect_u32;
157 QLIB_FLASH_CONFIG_T flashCfg = {0u};
158 QLIB_DIE_CONFIG_T dieCfg = {0u};
159 QLIB_SECTIONS_CONF_TABLE_T cfgLine[QLIB_MAX_DIES_SUPPORTED];
160 QLIB_EXTENDED_SECTION_CONF_T* sectCfg;
161 uint8_t diesNum_u8 = QLIB_GET_NUM_OF_DIES(qlibContext);
162 bool isProvisioned_b;
163
164 // See what chip is connected and its unique ids. We might need this info to obtain keys for this
165 // chips according to formula or via DB see QLIB_SAMPLE_GetDeviceKey() function
166 QLIB_STATUS_RET_CHECK(QLIB_GetHWVersion(qlibContext, &hwVer));
167 QLIB_STATUS_RET_CHECK(QLIB_GetId(qlibContext, &id));
168
169 // We do read / modify /write configuration since we want to change only what is needed
170 // read configuration
171 QLIB_STATUS_RET_CHECK(QLIB_GetDeviceConfigMultiDie(qlibContext, &flashCfg, &dieCfg, cfgLine, diesNum_u8));
172
173 // modify configuration
174
175 // set pre provision master key if needed
176 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetDeviceKey(qlibContext,
177 id.std.uniqueID,
178 QLIB_KID__DEVICE_KEY_PRE_PROVISIONING,
179 dieCfg.preProvisionedMasterKey));
180
181 // set new master key
182 QLIB_STATUS_RET_CHECK(
183 QLIB_SAMPLE_GetDeviceKey(qlibContext, id.std.uniqueID, QLIB_KID__DEVICE_MASTER, dieCfg.deviceMasterKey));
184
185#if QLIB_MAX_DIES_SUPPORTED > 1u
186 QLIB_STATUS_RET_CHECK(QLIB_GetActiveDie(qlibContext, &die_u8));
187#endif // QLIB_MAX_DIES_SUPPORTED > 1u
188 sectCfg = cfgLine[die_u8].sectionConfigTable;
189
190 sect_u32 = LMS_SAMPLE_SECTION_ID;
191
192 sectCfg[sect_u32].baseAddr = QLIB_SAMPLE_RESERVED_BOOT_SECTION_SIZE;
193 sectCfg[sect_u32].size = SIZE_512KB_;
194
195 // section configured usually will be set with rollback protected for FW update
196 sectCfg[sect_u32].sectionConfig.policy.plainAccessReadEnable = 1u;
197 sectCfg[sect_u32].sectionConfig.policy.plainAccessWriteEnable = 0u;
198 sectCfg[sect_u32].sectionConfig.policy.authPlainAccess = 0u;
199 sectCfg[sect_u32].sectionConfig.policy.checksumIntegrity = 0u;
200 sectCfg[sect_u32].sectionConfig.policy.digestIntegrity = 0u;
201 sectCfg[sect_u32].sectionConfig.policy.digestIntegrityOnAccess = 0u;
202 sectCfg[sect_u32].sectionConfig.policy.rollbackProt = 1u;
203 sectCfg[sect_u32].sectionConfig.policy.slog = 0u;
204 sectCfg[sect_u32].sectionConfig.policy.writeProt = 0u;
205
206 // Full access key configuration.This is the key to use later for the section configuration
207 // changes or write access
208 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSectionAccessKey(qlibContext,
209 id.std.uniqueID,
210 QLIB_KID__FULL_ACCESS_SECTION,
211 sect_u32,
212 die_u8,
213 sectCfg[sect_u32].fullAccessKey));
214
215 // configure LMS public key.
216 QLIB_STATUS_RET_CHECK(QLIB_IsKeyProvisioned(qlibContext, QLIB_KID__SECTION_LMS, sect_u32, &isProvisioned_b));
217
218 if ((bool)false == isProvisioned_b)
219 {
220#ifndef LMS_SAMPLE_PREPARE_IN_ADVANCE
221 // lms public key is obtained from hsm
222 LMS_PUB_KEY_T publicKey_u8;
223 QLIB_LMS_KEY_ID_T keyId;
224
225 QLIB_STATUS_RET_CHECK(QLIB_LMS_HsmGenerateKeyPair(keyId));
226 QLIB_STATUS_RET_CHECK(QLIB_LMS_HsmGetKey(keyId, publicKey_u8));
227#else
228 const QLIB_LMS_KEY_ID_T keyId = LMS_KEY_ID_1;
229 const uint8_t publicKey_u8[] = LMS_PUB_KEY_1;
230#endif // LMS_SAMPLE_PREPARE_IN_ADVANCE
231 QLIB_SAMPLE_LmsKeyBackup_storeKid_L(keyId);
232 // copy to chip configuration data structure
233 (void)memcpy(sectCfg[sect_u32].lmsKey, publicKey_u8, sizeof(LMS_PUB_KEY_T));
234 }
235 else
236 {
237#ifndef LMS_SAMPLE_PREPARE_IN_ADVANCE
238 // Can not run the sample as LMS key is already set on section
239 return QLIB_STATUS__SYSTEM_IN_INCORRECT_STATE;
240#else
241 // lms public key can be already inside, we assume that this sample was already ran before
242 const QLIB_LMS_KEY_ID_T keyId = LMS_KEY_ID_1;
243 QLIB_SAMPLE_LmsKeyBackup_storeKid_L(keyId);
244#endif // LMS_SAMPLE_PREPARE_IN_ADVANCE
245 }
246
247 // write configuration
248 QLIB_STATUS_RET_CHECK(QLIB_ConfigDeviceMultiDie(qlibContext, &flashCfg, &dieCfg, cfgLine, diesNum_u8));
249
250#if QLIB_MAX_DIES_SUPPORTED > 1u
251 // write configuration sets die to 0. return to orig die
252 QLIB_STATUS_RET_CHECK(QLIB_SetActiveDie(qlibContext, die_u8));
253#endif // QLIB_MAX_DIES_SUPPORTED > 1u
254
255 // Create backup key for redundancy and sign the LMS key replacement command using the current LMS key
256 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_LmsKeyBackup_createBackupKey_L());
257
258 return QLIB_STATUS__OK;
259}

261QLIB_STATUS_T QLIB_SAMPLE_LmsKeyBackup_field_flow(QLIB_CONTEXT_T* qlibContext)
262{
263 QLIB_LMS_KEY_ID_T keyId;
264 QLIB_LMS_KEY_ID_T backupKeyId_pu8;
265 QLIB_LMS_MSG_DATA_STRUCT_T lmsMsg_pu8;
266 QLIB_LMS_SIG_BUFFER_T signature_pu8;
267 /*-------------------------------------------------------------------------------------------------------
268 Initialize KeyId to the stored key id that match the configured public key on flash
269 -------------------------------------------------------------------------------------------------------*/
270 QLIB_SAMPLE_LmsKeyBackup_retrieveKid_L(keyId);
271
272 QLIB_SAMPLE_LmsKeyBackup_retrieveBackupCmd_L(backupKeyId_pu8, lmsMsg_pu8, signature_pu8);
273
274 /*---------------------------------------------------------------------------------------------------
275 Send the LMS command to the flash
276 ---------------------------------------------------------------------------------------------------*/
277 QLIB_STATUS_RET_CHECK(QLIB_SendLMSCommand(qlibContext, lmsMsg_pu8, keyId, (uint8_t*)signature_pu8, LMS_SAMPLE_SECTION_ID));
278
279 /*---------------------------------------------------------------------------------------------------
280 Now the active key became the new one we just provisioned to flash
281 ---------------------------------------------------------------------------------------------------*/
282 QLIB_SAMPLE_LmsKeyBackup_storeKid_L(backupKeyId_pu8);
283
284 /*---------------------------------------------------------------------------------------------------
285 Optionally, user can create new backup for future use.
286 In this example we avoid it when the key pair generation is needed since this is time consuming
287 ---------------------------------------------------------------------------------------------------*/
288#ifdef LMS_SAMPLE_PREPARE_IN_ADVANCE
289 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_LmsKeyBackup_createBackupKey_L());
290#endif // LMS_SAMPLE_PREPARE_IN_ADVANCE
291
292 return QLIB_STATUS__OK;
293}
