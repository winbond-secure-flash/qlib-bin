uint8_t PLAT_SPI_TransferByte(void* spi_regs_base, uint8_t byte_out) {
volatile uint32_t timeout;
// Wait for TX buffer to be empty
timeout = 1000000; // Add a reasonable timeout
while (!(SPI_SR(spi_regs_base) & SPI_SR_TXE)) {
if (--timeout == 0) return 0xFF; // Indicate timeout error (or handle differently)
}
// Write byte to Data Register
SPI_DR(spi_regs_base) = byte_out;
// Wait for RX buffer to be not empty
timeout = 1000000; // Add a reasonable timeout
while (!(SPI_SR(spi_regs_base) & SPI_SR_RXNE)) {
if (--timeout == 0) return 0xFF; // Indicate timeout error
}
// Read received byte (clears RXNE flag)
return SPI_DR(spi_regs_base);
}
// --- PLAT_SPI_WriteReadTransaction Implementation (Bare Metal) ---
int PLAT_SPI_WriteReadTransaction(const void* userData,
QLIB_BUS_MODE_T format,
uint32_t flags,
const uint8_t* dataOutStream,
uint32_t cmdSize,
uint32_t addressSize,
uint32_t dataOutSize,
uint32_t dummyCycles,
uint8_t* dataIn,
uint32_t dataInSize)
{
// Assume userData points to a BareMetalSpiContext structure
const BareMetalSpiContext* context = (const BareMetalSpiContext*)userData;
if (!context || !context->spi_regs_base) {
return PLAT_SPI_INV_PARAM;
}
void* spi_base = context->spi_regs_base;
uint32_t cs_pin = context->cs_pin;
// --- 1. Configure SPI Mode (CPOL/CPHA) ---
// This might involve reading CR1, masking bits, and writing back
// Or completely re-initializing if settings change drastically
uint32_t cr1_val = SPI_CR1(spi_base);
cr1_val &= ~(SPI_CR1_CPOL | SPI_CR1_CPHA); // Clear existing mode bits
switch (format) {
case QLIB_BUS_MODE_SPI_MODE0: /* CPOL=0, CPHA=0 */ break; // Already cleared
case QLIB_BUS_MODE_SPI_MODE1: cr1_val |= SPI_CR1_CPHA; break;
case QLIB_BUS_MODE_SPI_MODE2: cr1_val |= SPI_CR1_CPOL; break;
case QLIB_BUS_MODE_SPI_MODE3: cr1_val |= (SPI_CR1_CPOL | SPI_CR1_CPHA); break;
default: return PLAT_SPI_INV_PARAM;
}
// Ensure SPI is enabled and in Master mode (assuming it was set up before)
cr1_val |= SPI_CR1_SPE | SPI_CR1_MSTR;
SPI_CR1(spi_base) = cr1_val;
// Add configuration for speed, LSB/MSB based on 'format' or 'flags' if needed
// --- 2. Assert Chip Select ---
PLAT_GPIO_SetLow(cs_pin);
// --- 3. Write Phase (Command + Address + Data) ---
uint32_t totalWriteSize = cmdSize + addressSize + dataOutSize;
for (uint32_t i = 0; i < totalWriteSize; ++i) {
// Transfer byte, ignore received data during write phase
if (PLAT_SPI_TransferByte(spi_base, dataOutStream[i]) == 0xFF) {
PLAT_GPIO_SetHigh(cs_pin); // Release CS# on error
return PLAT_SPI_TIMEOUT; // Or PLAT_SPI_ERROR
}
}
// --- 4. Dummy Cycles Phase ---
uint32_t num_dummy_bytes = (dummyCycles + 7) / 8;
// Often implemented by writing dummy bytes (e.g., 0xFF or 0x00)
for (uint32_t i = 0; i < num_dummy_bytes; ++i) {
if (PLAT_SPI_TransferByte(spi_base, 0xFF) == 0xFF) { // Send dummy byte 0xFF
PLAT_GPIO_SetHigh(cs_pin); // Release CS# on error
return PLAT_SPI_TIMEOUT; // Or PLAT_SPI_ERROR
}
}
// --- 5. Read Phase ---
for (uint32_t i = 0; i < dataInSize; ++i) {
uint8_t received_byte = PLAT_SPI_TransferByte(spi_base, 0xFF); // Send dummy byte 0xFF to clock data in
if (received_byte == 0xFF) { // Check if helper indicated an error
PLAT_GPIO_SetHigh(cs_pin); // Release CS# on error
return PLAT_SPI_TIMEOUT; // Or PLAT_SPI_ERROR
}
if (dataIn) { // Check if dataIn buffer is valid
dataIn[i] = received_byte;
}
}
// --- 6. De-assert Chip Select ---
// Check flags if CS# should be kept asserted (optional, depends on library needs)
if (!(flags & PLAT_SPI_FLAG_KEEP_CS_ASSERTED)) {
// Wait for SPI bus to become idle before raising CS# (important!)
volatile uint32_t timeout = 100000;
while (SPI_SR(spi_base) & SPI_SR_BSY) {
if (--timeout == 0) {
// Handle timeout waiting for BSY flag - may still raise CS#
PLAT_GPIO_SetHigh(cs_pin);
return PLAT_SPI_TIMEOUT; // Indicate potential issue
}
}
PLAT_GPIO_SetHigh(cs_pin);
}
return PLAT_SPI_OK;
}
