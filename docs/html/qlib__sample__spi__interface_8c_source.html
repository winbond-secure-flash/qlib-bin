41QLIB_STATUS_T QLIB_SAMPLE_SetSpiInterface(void* userData_pv, QLIB_SAMPLE_SetFetchCmd_CB setFetchCmd)
42{
43 QLIB_CONTEXT_T qlibContext;
44 QLIB_STATUS_T status = QLIB_STATUS__OK;
45
46 /*-------------------------------------------------------------------------------------------------------
47 Init QLIB - needed when running QLIB either on a device or on a remote server.
48 -------------------------------------------------------------------------------------------------------*/
49 QLIB_STATUS_RET_CHECK(QLIB_InitLib(&qlibContext));
50
51 /*-------------------------------------------------------------------------------------------------------
52 Set the user data into Qlib's context
53 -------------------------------------------------------------------------------------------------------*/
54 QLIB_SetUserData(&qlibContext, userData_pv);
55
56 /*-------------------------------------------------------------------------------------------------------
57 Take the ownership of flash communication channel (it belongs to local MCU or to remote server, exclusively)
58 -------------------------------------------------------------------------------------------------------*/
59 QLIB_STATUS_RET_CHECK(QLIB_Connect(&qlibContext));
60
61 /*-------------------------------------------------------------------------------------------------------
62 Init Flash Device - not needed when using QLIB on a remote server
63 -------------------------------------------------------------------------------------------------------*/
64 QLIB_STATUS_RET_CHECK_GOTO(QLIB_InitDevice(&qlibContext, QLIB_BUS_FORMAT(QLIB_BUS_MODE_1_1_1, (bool)false)),
65 status,
66 disconnect);
67
68 /*-------------------------------------------------------------------------------------------------------
69 Sample for setting new SPI interface parameters
70 -------------------------------------------------------------------------------------------------------*/
71 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_SetSpiInterface_flow(&qlibContext, setFetchCmd), status, disconnect);
72
73 /*-------------------------------------------------------------------------------------------------------
74 Release the ownership of flash communication channel (it belongs to local MCU or to remote server, exclusively)
75 -------------------------------------------------------------------------------------------------------*/
76 QLIB_STATUS_RET_CHECK(QLIB_Disconnect(&qlibContext));
77
78 return QLIB_STATUS__OK;
79
80disconnect:
81 (void)QLIB_Disconnect(&qlibContext);
82 return status;
83}

85QLIB_STATUS_T QLIB_SAMPLE_SetSpiInterface_flow(QLIB_CONTEXT_T* qlibContext, QLIB_SAMPLE_SetFetchCmd_CB setFetchCmd)
86{
87 QLIB_STATUS_T status = QLIB_STATUS__OK;
88 QLIB_BUS_FORMAT_T busFormat;
89 uint32_t dummyCycles_pu32;
90 QLIB_STD_ADDR_MODE_T addrMode;
91 QLIB_BUS_FORMAT_T origBusFormat = QLIB_BUS_FORMAT(QLIB_BUS_MODE_1_1_1, (bool)false); // Initial value for current sample
92 uint32_t origDummyCycles_u32 = FLASH_CR_DUMMY_DEFAULT; // Initial value for current sample
93 QLIB_STD_ADDR_MODE_T origAddrMode = QLIB_STD_ADDR_MODE__3_BYTE; // Initial value for current sample
94
95#ifndef QLIB_NO_DIRECT_FLASH_ACCESS
96 QLIB_ASSERT_RET(setFetchCmd != NULL, QLIB_STATUS__INVALID_PARAMETER);
97#endif // QLIB_NO_DIRECT_FLASH_ACCESS
98
99#ifdef QLIB_SUPPORT_QPI
100 busFormat = QLIB_BUS_FORMAT(QLIB_BUS_MODE_4_4_4, (bool)false);
101#else
102 busFormat = QLIB_BUS_FORMAT(QLIB_BUS_MODE_1_1_1, (bool)false);
103#endif // QLIB_SUPPORT_QPI
104
105 if (QLIB_FeatureSupported(qlibContext, W77Q_FEATURE_FAST_READ_DUMMY_CONFIG) == (bool)false)
106 {
107 dummyCycles_pu32 = QLIB_QPI_READ_DUMMY_CYCLES;
108 }
109 else
110 {
111#ifdef QLIB_SUPPORT_QPI
112 dummyCycles_pu32 = 16u; // max needed dummy cycles for QPI SDR in W77Q/T 256Mb to 1Gb
113#else
114 dummyCycles_pu32 = 8u; // max needed dummy cycles for SPI SDR in W77Q/T 256Mb to 1Gb
115#endif // QLIB_SUPPORT_QPI
116 }
117
118 addrMode = (QLIB_FeatureSupported(qlibContext, W77Q_FEATURE_4_BYTES_ADDRESS_MODE) == (bool)true) ? QLIB_STD_ADDR_MODE__4_BYTE
119 : QLIB_STD_ADDR_MODE__3_BYTE;
120#ifndef QLIB_NO_DIRECT_FLASH_ACCESS
121 if (QLIB_FeatureSupported(qlibContext, W77Q_FEATURE_FAST_READ_DUMMY_CONFIG) == (bool)false)
122 {
123 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_DummyCyclesForBusMode_L(origBusFormat, &origDummyCycles_u32));
124 }
125#endif // QLIB_NO_DIRECT_FLASH_ACCESS
126
127 /*-------------------------------------------------------------------------------------------------------
128 configure chip basic registers to some value (just to make sure GMC, GMT are configured)
129 -------------------------------------------------------------------------------------------------------*/
130 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_SecureStorage_manuf_flow(qlibContext), status, recover_spi);
131
132 /*-------------------------------------------------------------------------------------------------------
133 Set SPI interface
134 In case XIP is used, the SPI interface parameters should be set in QLIB, flash and in core simultaneously,
135 to allow both proper QLIB operation and command fetch operations from core.
136 -------------------------------------------------------------------------------------------------------*/
137 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_SetSPI(qlibContext, busFormat, dummyCycles_pu32, addrMode, setFetchCmd));
138
139 /*-------------------------------------------------------------------------------------------------------
140 Perform some commands to test new SPI mode
141 -------------------------------------------------------------------------------------------------------*/
142 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_SecureStorage_field_flow_plainRW(qlibContext), status, recover_spi);
143
144 /*-------------------------------------------------------------------------------------------------------
145 return to original SPI parameters
146 -------------------------------------------------------------------------------------------------------*/
147 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_SetSPI(qlibContext, origBusFormat, origDummyCycles_u32, origAddrMode, setFetchCmd));
148
149 return status;
150
151recover_spi:
152 (void)QLIB_SAMPLE_SetSPI(qlibContext, origBusFormat, origDummyCycles_u32, origAddrMode, setFetchCmd);
153 return status;
154}

156QLIB_STATUS_T QLIB_SAMPLE_SetSPI(QLIB_CONTEXT_T* qlibContext,
157 QLIB_BUS_FORMAT_T busFormat,
158 uint32_t dummyCycles_pu32,
159 QLIB_STD_ADDR_MODE_T addrMode,
160 QLIB_SAMPLE_SetFetchCmd_CB setFetchCmd)
161{
162 bool support4ByteAddr_b = (QLIB_FeatureSupported(qlibContext, W77Q_FEATURE_4_BYTES_ADDRESS_MODE) != (bool)false);
163#ifndef EXCLUDE_FAST_READ_DUMMY_CONFIG
164 bool supportDummyConf_b = (QLIB_FeatureSupported(qlibContext, W77Q_FEATURE_FAST_READ_DUMMY_CONFIG) != (bool)false);
165#endif // EXCLUDE_FAST_READ_DUMMY_CONFIG
166#ifndef QLIB_NO_DIRECT_FLASH_ACCESS
167 if (setFetchCmd != NULL)
168 {
169 // If direct flash access is needed or QLIB runs from flash, We need to set the SPI fetch command in core.
170 QLIB_ASSERT_RET(setFetchCmd(busFormat, dummyCycles_pu32, addrMode == QLIB_STD_ADDR_MODE__3_BYTE ? true : false) == 0u,
171 QLIB_STATUS__HARDWARE_FAILURE);
172 }
173#else
174 (void)setFetchCmd; // Suppress unused parameter warning when direct flash access is disabled
175#endif // QLIB_NO_DIRECT_FLASH_ACCESS
176
177 if (support4ByteAddr_b)
178 {
179 QLIB_STATUS_RET_CHECK(QLIB_SetAddressMode(qlibContext, addrMode));
180 }
181#ifndef EXCLUDE_FAST_READ_DUMMY_CONFIG
182 if (supportDummyConf_b)
183 {
184 QLIB_STATUS_RET_CHECK(QLIB_SetFastReadDummyCycles(qlibContext, dummyCycles_pu32));
185 }
186#else
187 (void)dummyCycles_pu32;
188#endif // EXCLUDE_FAST_READ_DUMMY_CONFIG
189 QLIB_STATUS_RET_CHECK(QLIB_SetInterface(qlibContext, busFormat));
190 return QLIB_STATUS__OK;
191}

205static QLIB_STATUS_T QLIB_SAMPLE_DummyCyclesForBusMode_L(QLIB_BUS_FORMAT_T busFormat, uint32_t* dummyCycles_pu32)
206{
207 QLIB_STATUS_T ret = QLIB_STATUS__OK;
208 bool dtr_b = QLIB_BUS_FORMAT_GET_DTR(busFormat);
209
210 QLIB_ASSERT_RET(dummyCycles_pu32 != NULL, QLIB_STATUS__INVALID_PARAMETER);
211
212 switch (QLIB_BUS_FORMAT_GET_MODE(busFormat))
213 {
214 case QLIB_BUS_MODE_1_1_1:
215 *dummyCycles_pu32 = ((bool)true == dtr_b) ? SPI_DC_FAST_READ_DTR_111 : SPI_DC_FAST_READ_11X;
216 break;
217 case QLIB_BUS_MODE_1_1_2:
218 case QLIB_BUS_MODE_1_2_2:
219 *dummyCycles_pu32 = ((bool)true == dtr_b) ? SPI_DC_FAST_READ_DTR_122 : SPI_DC_FAST_READ_122;
220 break;
221 case QLIB_BUS_MODE_1_1_4:
222 case QLIB_BUS_MODE_1_4_4:
223 *dummyCycles_pu32 = ((bool)true == dtr_b) ? SPI_DC_FAST_READ_DTR_144 : SPI_DC_FAST_READ_144;
224 break;
225 case QLIB_BUS_MODE_4_4_4:
226 *dummyCycles_pu32 = ((bool)true == dtr_b) ? SPI_DC_FAST_READ_DTR_444 : QLIB_QPI_READ_DUMMY_CYCLES;
227 break;
228 default:
229 ret = QLIB_STATUS__INVALID_PARAMETER;
230 break;
231 }
232
233 return ret;
234}
