50QLIB_STATUS_T QLIB_SAMPLE_Integrity(void* userData_pv)
51{
52 QLIB_CONTEXT_T qlibContext;
53 QLIB_STATUS_T status = QLIB_STATUS__OK;
54 QLIB_BUS_FORMAT_T busFormat = QLIB_BUS_FORMAT(QLIB_BUS_MODE_1_1_1, (bool)false);
55
56 /*-------------------------------------------------------------------------------------------------------
57 Init QLIB - needed when running QLIB either on a device or on a remote server.
58 -------------------------------------------------------------------------------------------------------*/
59 QLIB_STATUS_RET_CHECK(QLIB_InitLib(&qlibContext));
60
61 /*-------------------------------------------------------------------------------------------------------
62 Set the user data into Qlib's context
63 -------------------------------------------------------------------------------------------------------*/
64 QLIB_SetUserData(&qlibContext, userData_pv);
65
66 /*-------------------------------------------------------------------------------------------------------
67 Take the ownership of flash communication channel
68 -------------------------------------------------------------------------------------------------------*/
69 QLIB_STATUS_RET_CHECK(QLIB_Connect(&qlibContext));
70
71 /*-------------------------------------------------------------------------------------------------------
72 Init Flash Device - not needed when using QLIB on a remote server
73 -------------------------------------------------------------------------------------------------------*/
74 QLIB_STATUS_RET_CHECK_GOTO(QLIB_InitDevice(&qlibContext, busFormat), status, disconnect);
75
76 /*-------------------------------------------------------------------------------------------------------
77 Configuration for sample - set up sections -
78 section 1 with both CRC and digest integrity,
79 and section 2 with plain read/write access (used to corrupt section 1)
80 -------------------------------------------------------------------------------------------------------*/
81 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_Integrity_manuf_flow(&qlibContext), status, disconnect);
82
83 /*-------------------------------------------------------------------------------------------------------
84 Sample flow - verify integrity and demonstrate corruption detection
85 -------------------------------------------------------------------------------------------------------*/
86 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_Integrity_field_flow(&qlibContext), status, disconnect);
87
88disconnect:
89 (void)QLIB_Disconnect(&qlibContext);
90 return status;
91}

93QLIB_STATUS_T QLIB_SAMPLE_Integrity_manuf_flow(QLIB_CONTEXT_T* qlibContext)
94{
95 QLIB_HW_VER_T hwVer;
96 QLIB_ID_T id;
97 uint8_t die_u8 = 0u;
98 QLIB_FLASH_CONFIG_T flashCfg;
99 QLIB_DIE_CONFIG_T dieCfg;
100 QLIB_SECTIONS_CONF_TABLE_T cfgLine[QLIB_MAX_DIES_SUPPORTED];
101 QLIB_EXTENDED_SECTION_CONF_T* sectCfg;
102 uint8_t diesNum_u8 = QLIB_GET_NUM_OF_DIES(qlibContext);
103
104 (void)memset(&flashCfg, 0, sizeof(flashCfg));
105 (void)memset(&dieCfg, 0, sizeof(dieCfg));
106 (void)memset(cfgLine, 0, sizeof(cfgLine));
107
108 /*-------------------------------------------------------------------------------------------------------
109 See what chip is connected and its unique ids. We might need this info to obtain keys for this
110 chip according to formula or via DB. See QLIB_SAMPLE_GetDeviceKey() function.
111 -------------------------------------------------------------------------------------------------------*/
112 QLIB_STATUS_RET_CHECK(QLIB_GetHWVersion(qlibContext, &hwVer));
113 QLIB_STATUS_RET_CHECK(QLIB_GetId(qlibContext, &id));
114
115 /*-------------------------------------------------------------------------------------------------------
116 We do read / modify / write configuration since we want to change only what is needed.
117 -------------------------------------------------------------------------------------------------------*/
118 QLIB_STATUS_RET_CHECK(QLIB_GetDeviceConfigMultiDie(qlibContext, &flashCfg, &dieCfg, cfgLine, diesNum_u8));
119
120 /*-------------------------------------------------------------------------------------------------------
121 Set pre-provision master key if needed
122 -------------------------------------------------------------------------------------------------------*/
123 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetDeviceKey(qlibContext,
124 id.std.uniqueID,
125 QLIB_KID__DEVICE_KEY_PRE_PROVISIONING,
126 dieCfg.preProvisionedMasterKey));
127
128 /*-------------------------------------------------------------------------------------------------------
129 Set new master key
130 -------------------------------------------------------------------------------------------------------*/
131 QLIB_STATUS_RET_CHECK(
132 QLIB_SAMPLE_GetDeviceKey(qlibContext, id.std.uniqueID, QLIB_KID__DEVICE_MASTER, dieCfg.deviceMasterKey));
133
134#if QLIB_MAX_DIES_SUPPORTED > 1u
135 QLIB_STATUS_RET_CHECK(QLIB_GetActiveDie(qlibContext, &die_u8));
136#endif // QLIB_MAX_DIES_SUPPORTED > 1u
137
138 sectCfg = cfgLine[die_u8].sectionConfigTable;
139
140 /*-------------------------------------------------------------------------------------------------------
141 Section 1 (check section): CRC and digest integrity protected, plain read enabled, plain write disabled.
142 Both the CRC and the digest are pre-calculated for SIZE_512KB_ of 0xFF (erased flash).
143 CRC integrity is verified by the device without a session; digest integrity requires a restricted session.
144 -------------------------------------------------------------------------------------------------------*/
145 sectCfg[QLIB_SAMPLE_INTEGRITY_CHECK_SECTION].baseAddr = QLIB_SAMPLE_RESERVED_BOOT_SECTION_SIZE;
146 sectCfg[QLIB_SAMPLE_INTEGRITY_CHECK_SECTION].size = SIZE_512KB_;
147
148 sectCfg[QLIB_SAMPLE_INTEGRITY_CHECK_SECTION].sectionConfig.policy.plainAccessReadEnable = 1u;
149 sectCfg[QLIB_SAMPLE_INTEGRITY_CHECK_SECTION].sectionConfig.policy.plainAccessWriteEnable = 0u;
150 sectCfg[QLIB_SAMPLE_INTEGRITY_CHECK_SECTION].sectionConfig.policy.checksumIntegrity = 1u;
151 sectCfg[QLIB_SAMPLE_INTEGRITY_CHECK_SECTION].sectionConfig.policy.digestIntegrity = 1u;
152 sectCfg[QLIB_SAMPLE_INTEGRITY_CHECK_SECTION].sectionConfig.policy.digestIntegrityOnAccess =
153 (QLIB_FeatureSupported(qlibContext, W77Q_FEATURE_AUTH_AC) != (bool)false) ? 1u : 0u;
154 sectCfg[QLIB_SAMPLE_INTEGRITY_CHECK_SECTION].sectionConfig.crc = QLIB_SAMPLE_INTEGRITY_CRC_EXPECTED;
155 sectCfg[QLIB_SAMPLE_INTEGRITY_CHECK_SECTION].sectionConfig.digest = QLIB_SAMPLE_INTEGRITY_DIGEST_EXPECTED;
156
157 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSectionAccessKey(qlibContext,
158 id.std.uniqueID,
159 QLIB_KID__RESTRICTED_ACCESS_SECTION,
160 QLIB_SAMPLE_INTEGRITY_CHECK_SECTION,
161 die_u8,
162 sectCfg[QLIB_SAMPLE_INTEGRITY_CHECK_SECTION].restrictedKey));
163
164 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSectionAccessKey(qlibContext,
165 id.std.uniqueID,
166 QLIB_KID__FULL_ACCESS_SECTION,
167 QLIB_SAMPLE_INTEGRITY_CHECK_SECTION,
168 die_u8,
169 sectCfg[QLIB_SAMPLE_INTEGRITY_CHECK_SECTION].fullAccessKey));
170
171 /*-------------------------------------------------------------------------------------------------------
172 Section 2 (test section): plain read and write enabled. This section overlaps section 1 memory area
173 and will be used to corrupt section 1 data to demonstrate integrity detection.
174 -------------------------------------------------------------------------------------------------------*/
175 sectCfg[QLIB_SAMPLE_INTEGRITY_TEST_SECTION].baseAddr = QLIB_SAMPLE_RESERVED_BOOT_SECTION_SIZE;
176 sectCfg[QLIB_SAMPLE_INTEGRITY_TEST_SECTION].size = SIZE_2MB_;
177
178 sectCfg[QLIB_SAMPLE_INTEGRITY_TEST_SECTION].sectionConfig.policy.plainAccessReadEnable = 1u;
179 sectCfg[QLIB_SAMPLE_INTEGRITY_TEST_SECTION].sectionConfig.policy.plainAccessWriteEnable = 1u;
180
181 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSectionAccessKey(qlibContext,
182 id.std.uniqueID,
183 QLIB_KID__FULL_ACCESS_SECTION,
184 QLIB_SAMPLE_INTEGRITY_TEST_SECTION,
185 die_u8,
186 sectCfg[QLIB_SAMPLE_INTEGRITY_TEST_SECTION].fullAccessKey));
187
188 /*-------------------------------------------------------------------------------------------------------
189 Write configuration to device
190 -------------------------------------------------------------------------------------------------------*/
191 QLIB_STATUS_RET_CHECK(QLIB_ConfigDeviceMultiDie(qlibContext, &flashCfg, &dieCfg, cfgLine, diesNum_u8));
192
193#if QLIB_MAX_DIES_SUPPORTED > 1u
194 QLIB_STATUS_RET_CHECK(QLIB_SetActiveDie(qlibContext, die_u8));
195#endif // QLIB_MAX_DIES_SUPPORTED > 1u
196
197 return QLIB_STATUS__OK;
198}

200QLIB_STATUS_T QLIB_SAMPLE_Integrity_field_flow(QLIB_CONTEXT_T* qlibContext)
201{
202 QLIB_ID_T id;
203 uint8_t die_u8 = 0u;
204 KEY_T digestSectKey;
205 uint8_t corruptedData_u8[] = {0xDEu, 0xADu, 0xBEu, 0xEFu};
206 QLIB_STATUS_T ret = QLIB_STATUS__OK;
207
208 QLIB_STATUS_RET_CHECK(QLIB_GetId(qlibContext, &id));
209#if QLIB_MAX_DIES_SUPPORTED > 1u
210 QLIB_STATUS_RET_CHECK(QLIB_GetActiveDie(qlibContext, &die_u8));
211#endif // QLIB_MAX_DIES_SUPPORTED > 1u
212
213 /*-------------------------------------------------------------------------------------------------------
214 Obtain the restricted key for the digest section (needed to open a session for digest checks)
215 -------------------------------------------------------------------------------------------------------*/
216 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSectionAccessKey(qlibContext,
217 id.std.uniqueID,
218 QLIB_KID__RESTRICTED_ACCESS_SECTION,
219 QLIB_SAMPLE_INTEGRITY_CHECK_SECTION,
220 die_u8,
221 digestSectKey));
222
223 QLIB_STATUS_RET_CHECK(QLIB_LoadKey(qlibContext, QLIB_SAMPLE_INTEGRITY_CHECK_SECTION, digestSectKey, (bool)false));
224 QLIB_STATUS_RET_CHECK_GOTO(QLIB_OpenSession(qlibContext, QLIB_SAMPLE_INTEGRITY_CHECK_SECTION, QLIB_SESSION_ACCESS_RESTRICTED),
225 ret,
226 remove_key);
227
228 /*-------------------------------------------------------------------------------------------------------
229 Verify section 1 CRC integrity is valid (flash is erased, matches pre-configured CRC)
230 -------------------------------------------------------------------------------------------------------*/
231 QLIB_ASSERT_WITH_ERROR_GOTO(QLIB_STATUS__OK ==
232 QLIB_CheckIntegrity(qlibContext, QLIB_SAMPLE_INTEGRITY_CHECK_SECTION, QLIB_INTEGRITY_CRC),
233 QLIB_STATUS__TEST_FAIL,
234 ret,
235 close_session);
236
237 /*-------------------------------------------------------------------------------------------------------
238 Verify section 1 digest integrity is valid (flash is erased, matches pre-configured digest).
239 Unlike CRC, digest check requires an authenticated session.
240 -------------------------------------------------------------------------------------------------------*/
241 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_CheckDigestIntegrity_L(qlibContext, (bool)true), ret, close_session);
242
243 QLIB_SAMPLE_ALLOW_TO_FAIL__START();
244 /*-------------------------------------------------------------------------------------------------------
245 Corrupt section 1 data through section 2 (overlapping plain write)
246 -------------------------------------------------------------------------------------------------------*/
247 QLIB_STATUS_RET_CHECK(
248 QLIB_Write(qlibContext, corruptedData_u8, QLIB_SAMPLE_INTEGRITY_TEST_SECTION, 0u, sizeof(corruptedData_u8), (bool)false));
249
250 /*-------------------------------------------------------------------------------------------------------
251 Verify section 1 CRC integrity now fails (data was corrupted)
252 -------------------------------------------------------------------------------------------------------*/
253 QLIB_ASSERT_RET(QLIB_STATUS__OK != QLIB_CheckIntegrity(qlibContext, QLIB_SAMPLE_INTEGRITY_CHECK_SECTION, QLIB_INTEGRITY_CRC),
254 QLIB_STATUS__TEST_FAIL);
255 QLIB_SAMPLE_ALLOW_TO_FAIL__END();
256 /*-------------------------------------------------------------------------------------------------------
257 Verify section 3 digest integrity now fails (data was corrupted)
258 -------------------------------------------------------------------------------------------------------*/
259 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_CheckDigestIntegrity_L(qlibContext, (bool)false), ret, close_session);
260
261close_session:
262 (void)QLIB_CloseSession(qlibContext, QLIB_SAMPLE_INTEGRITY_CHECK_SECTION);
263
264remove_key:
265 (void)QLIB_RemoveKey(qlibContext, QLIB_SAMPLE_INTEGRITY_CHECK_SECTION, (bool)false);
266 return ret;
267}

290static QLIB_STATUS_T QLIB_SAMPLE_CheckDigestIntegrity_L(QLIB_CONTEXT_T* qlibContext, bool expectPass_b)
291{
292 QLIB_STATUS_T ret = QLIB_STATUS__OK;
293 uint8_t digestBuff_u8[sizeof(uint64_t)];
294 uint64_t expectedDigest_u64 = QLIB_SAMPLE_INTEGRITY_DIGEST_EXPECTED;
295 bool digestMatches_b = (bool)false;
296
297 /*-----------------------------------------------------------------------------------------------------
298 Method 0: QLIB_CheckIntegrity
299 The device compares the section digest against the configured value and signals the result through
300 the return status: QLIB_STATUS__OK on match, QLIB_STATUS__SECURITY_ERR on mismatch.
301 -----------------------------------------------------------------------------------------------------*/
302 QLIB_SAMPLE_ALLOW_TO_FAIL__START();
303 ret = QLIB_CheckIntegrity(qlibContext, QLIB_SAMPLE_INTEGRITY_CHECK_SECTION, QLIB_INTEGRITY_DIGEST);
304 QLIB_SAMPLE_ALLOW_TO_FAIL__END();
305
306 if (expectPass_b)
307 {
308 QLIB_ASSERT_RET(QLIB_STATUS__OK == ret, QLIB_STATUS__TEST_FAIL);
309 }
310 else
311 {
312 QLIB_ASSERT_RET(QLIB_STATUS__OK != ret, QLIB_STATUS__TEST_FAIL);
313 ret = QLIB_STATUS__OK;
314 }
315 /*-----------------------------------------------------------------------------------------------------
316 Method 1: QLIB_GetSignedData + manual comparison
317 Retrieves the current section digest as raw bytes. Unlike QLIB_CheckIntegrity, this always returns
318 QLIB_STATUS__OK - integrity must be determined by comparing the output against the expected value.
319 -----------------------------------------------------------------------------------------------------*/
320 QLIB_STATUS_RET_CHECK(QLIB_GetSignedData(qlibContext,
321 QLIB_SAMPLE_INTEGRITY_CHECK_SECTION,
322 QLIB_SIGNED_DATA_TYPE_SECTION_DIGEST,
323 digestBuff_u8,
324 sizeof(digestBuff_u8),
325 NULL));
326
327 QLIB_SAMPLE_ALLOW_TO_FAIL__START();
328 digestMatches_b = (memcmp(digestBuff_u8, &expectedDigest_u64, sizeof(expectedDigest_u64)) == 0);
329 QLIB_SAMPLE_ALLOW_TO_FAIL__END();
330
331 QLIB_ASSERT_RET(expectPass_b == digestMatches_b, QLIB_STATUS__TEST_FAIL);
332
333 return ret;
334}
