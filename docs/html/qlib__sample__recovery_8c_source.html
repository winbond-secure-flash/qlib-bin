50QLIB_STATUS_T QLIB_SAMPLE_Recovery(void* userData_pv)
51{
52#ifdef QLIB_SUPPORT_XIP
53 TOUCH(userData_pv);
54 /*-------------------------------------------------------------------------------------------------------
55 recovery functionality is designed to run from RAM only since it formats the flash
56 -------------------------------------------------------------------------------------------------------*/
57 return QLIB_STATUS__NOT_SUPPORTED;
58#else
59 QLIB_CONTEXT_T qlibContext;
60 QLIB_BUS_FORMAT_T busFormat = QLIB_BUS_FORMAT(QLIB_BUS_MODE_1_1_1, (bool)false);
61 QLIB_STATUS_T status = QLIB_STATUS__OK;
62
63 /*-------------------------------------------------------------------------------------------------------
64 Init QLIB - needed when running QLIB either on a device or on a remote server.
65 -------------------------------------------------------------------------------------------------------*/
66 QLIB_STATUS_RET_CHECK(QLIB_InitLib(&qlibContext));
67
68 /*-------------------------------------------------------------------------------------------------------
69 Set the user data into Qlib's context
70 -------------------------------------------------------------------------------------------------------*/
71 QLIB_SetUserData(&qlibContext, userData_pv);
72
73 /*-------------------------------------------------------------------------------------------------------
74 Take the ownership of flash communication channel (it belongs to local MCU or to remote server, exclusively)
75 -------------------------------------------------------------------------------------------------------*/
76 QLIB_STATUS_RET_CHECK(QLIB_Connect(&qlibContext));
77
78 /*-------------------------------------------------------------------------------------------------------
79 Init Flash Device - not needed when using QLIB on a remote server
80 -------------------------------------------------------------------------------------------------------*/
81 QLIB_STATUS_RET_CHECK_GOTO(QLIB_InitDevice(&qlibContext, busFormat), status, disconnect);
82
83 /*-------------------------------------------------------------------------------------------------------
84 Flash will obtain legacy configuration
85 -------------------------------------------------------------------------------------------------------*/
86 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_Recovery_flow(&qlibContext), status, disconnect);
87
88disconnect:
89 (void)QLIB_Disconnect(&qlibContext);
90 return status;
91#endif // QLIB_SUPPORT_XIP
92}

94QLIB_STATUS_T QLIB_SAMPLE_Recovery_flow(QLIB_CONTEXT_T* qlibContext)
95{
96 bool nonSecureFormatEn_b;
97 QLIB_DIE_CONFIG_T dieCfg = {0u};
98 QLIB_DIE_CONFIG_T origDieCfg = {0u};
99 uint32_t sectionId_u32;
100 QLIB_ID_T id;
101 QLIB_FLASH_CONFIG_T flashCfg;
102 QLIB_FLASH_CONFIG_T tempFlashCfg;
103 QLIB_SECTIONS_CONF_TABLE_T cfgSectionArray[QLIB_MAX_DIES_SUPPORTED];
104 uint8_t dieId_u8;
105
106 QLIB_WATCHDOG_CONF_T watchdogDefault = {
107 (bool)false, // enable
108 (bool)false, // lfOscEn
109 (bool)false, // swResetEn
110 (bool)false, // authenticated
111 0u, // sectionID
112 QLIB_AWDT_TH_12_DAYS, // threshold
113 (bool)false, // lock
114 0u, // oscRateHz
115 (bool)false, // fallbackEn (in flash that supports Q2_SUPPORT_AWDTCFG_FALLBACK)
116 };
117
118
119 QLIB_SAMPLE_initLegacyConfig_L(qlibContext, &flashCfg, &dieCfg, cfgSectionArray, QLIB_GET_NUM_OF_DIES(qlibContext));
120
121 // See what chip is connected and its unique ids. We might need this info to obtain keys for this
122 // chips according to formula or via DB, see QLIB_SAMPLE_GetDeviceKey() function
123 QLIB_STATUS_RET_CHECK(QLIB_GetId(qlibContext, &id));
124
125 // get pre-provision key
126 QLIB_STATUS_RET_CHECK(
127 QLIB_SAMPLE_GetDeviceKey(qlibContext, id.sec.wid, QLIB_KID__DEVICE_KEY_PRE_PROVISIONING, dieCfg.preProvisionedMasterKey));
128
129 // get master key
130 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetDeviceKey(qlibContext, id.sec.wid, QLIB_KID__DEVICE_MASTER, dieCfg.deviceMasterKey));
131
132 //get secret key
133 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetDeviceKey(qlibContext, id.sec.wid, QLIB_KID__DEVICE_SECRET, dieCfg.deviceSecretKey));
134
135 // get SUID
136 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSecureUID(qlibContext, id.sec.wid, dieCfg.suid));
137
138 /*-------------------------------------------------------------------------------------------------------
139 Ensure device can be formatted and configured
140 ------------------------------------------------------------------------------------------------------*/
141 QLIB_STATUS_RET_CHECK(
142 QLIB_GetDeviceConfigMultiDie(qlibContext, &tempFlashCfg, &origDieCfg, NULL, QLIB_GET_NUM_OF_DIES(qlibContext)));
143 nonSecureFormatEn_b = origDieCfg.nonSecureFormatEn;
144 if ((bool)false == nonSecureFormatEn_b)
145 {
146
147 (void)memcpy(&flashCfg.watchdogDefault, &watchdogDefault, sizeof(QLIB_WATCHDOG_CONF_T));
148
149 QLIB_STATUS_RET_CHECK(
150 QLIB_ConfigDeviceMultiDie(qlibContext, &flashCfg, &dieCfg, NULL, QLIB_GET_NUM_OF_DIES(qlibContext)));
151 }
152
153 /*-------------------------------------------------------------------------------------------------------
154 Format the flash - clean from all it's data and configurations
155 -------------------------------------------------------------------------------------------------------*/
156 QLIB_STATUS_RET_CHECK(QLIB_Format(qlibContext, NULL, (bool)false, (bool)false));
157
158 /*-------------------------------------------------------------------------------------------------------
159 Write the configuration to the flash device
160 -------------------------------------------------------------------------------------------------------*/
161
162 (void)memcpy(&flashCfg.watchdogDefault, &watchdogDefault, sizeof(QLIB_WATCHDOG_CONF_T));
163
164 for (dieId_u8 = 0u; dieId_u8 < QLIB_GET_NUM_OF_DIES(qlibContext); ++dieId_u8)
165 {
166 for (sectionId_u32 = 0u; sectionId_u32 < QLIB_NUM_OF_SECTIONS; ++sectionId_u32)
167 {
168 if (cfgSectionArray[dieId_u8].sectionConfigTable[sectionId_u32].size > 0u)
169 {
170 QLIB_STATUS_RET_CHECK(
171 QLIB_SAMPLE_GetSectionAccessKey(qlibContext,
172 id.sec.wid,
173 QLIB_KID__FULL_ACCESS_SECTION,
174 sectionId_u32,
175 0u,
176 cfgSectionArray[dieId_u8].sectionConfigTable[sectionId_u32].fullAccessKey));
177 QLIB_STATUS_RET_CHECK(
178 QLIB_SAMPLE_GetSectionAccessKey(qlibContext,
179 id.sec.wid,
180 QLIB_KID__RESTRICTED_ACCESS_SECTION,
181 sectionId_u32,
182 0u,
183 cfgSectionArray[dieId_u8].sectionConfigTable[sectionId_u32].restrictedKey));
184 }
185 }
186 }
187 QLIB_STATUS_RET_CHECK(
188 QLIB_ConfigDeviceMultiDie(qlibContext, &flashCfg, &dieCfg, cfgSectionArray, QLIB_GET_NUM_OF_DIES(qlibContext)));
189
190
191#ifndef EXCLUDE_4_BYTES_ADDRESS_MODE
192 if (QLIB_FeatureSupported(qlibContext, W77Q_FEATURE_4_BYTES_ADDRESS_MODE) != 0u)
193 {
194#ifdef SPI_INIT_ADDRESS_MODE_4_BYTES
195 QLIB_STATUS_RET_CHECK(QLIB_SetAddressMode(qlibContext, QLIB_STD_ADDR_MODE__4_BYTE));
196#else
197 QLIB_STATUS_RET_CHECK(QLIB_SetAddressMode(qlibContext, QLIB_STD_ADDR_MODE__3_BYTE));
198#endif // SPI_INIT_ADDRESS_MODE_4_BYTES
199 }
200#endif // EXCLUDE_4_BYTES_ADDRESS_MODE
201
202#ifndef EXCLUDE_FAST_READ_DUMMY_CONFIG
203 if (QLIB_FeatureSupported(qlibContext, W77Q_FEATURE_FAST_READ_DUMMY_CONFIG) != 0u)
204 {
205 QLIB_STATUS_RET_CHECK(QLIB_SetFastReadDummyCycles(qlibContext, flashCfg.fastReadDummyCycles));
206 }
207#endif // EXCLUDE_FAST_READ_DUMMY_CONFIG
208
209 return QLIB_STATUS__OK;
210}

225 QLIB_FLASH_CONFIG_T* flashCfg,
226 QLIB_DIE_CONFIG_T* dieCfg,

231
232 // map section 0,1 for legacy access to all the flash size
233#if !defined(__cplusplus) || (__cplusplus >= 202002L)
234#define FLASH_DIE_SECTION_CONFIG_TABLE_DESIGNATOR .sectionConfigTable =
235#else

240 { \
241 FLASH_DIE_SECTION_CONFIG_TABLE_DESIGNATOR \
242 { \

244 /* Section 0 */ \
245 0u, /* baseAddr */ \
246 0u, /* size */ \

263 {0u} /* postActions */ \
264 }, \
265 (bool)true, /* resetPA */ \
266 {0u}, /* restrictedKey */ \
267 {0u}, /* fullAccessKey */ \
268 {0u}, /* lmsKey */ \
269 }, \
270 { \
271 /* Section 1 */ \
272 0u, /* baseAddr */ \
273 0u, /* size */ \
274 { \
275 { \
276 /* policy */ \
277 0u, /* digestIntegrity */ \
278 0u, /* checksumIntegrity */ \
279 0u, /* writeProt */ \
280 0u, /* rollbackProt */ \
281 1u, /* plainAccessWriteEnable */ \
282 1u, /* plainAccessReadEnable */ \
283 0u, /* authPlainAccess */ \
284 0u, /* digestIntegrityOnAccess */ \
285 0u, /* slog */ \
286 }, \
287 0u, /* section crc */ \
288 0u, /* section digest */ \
289 0u, /* version */ \
290 {0u} /* postActions */ \
291 }, \
292 (bool)false, /* resetPA */ \
293 {0u}, /* restrictedKey */ \
294 {0u}, /* fullAccessKey */ \
295 {0u}, /* lmsKey */ \
296 }, \
297 {0u}, /* Section 2 */ \
298 {0u}, /* Section 3 */ \
299 {0u}, /* Section 4 */ \
300 {0u}, /* Section 5 */ \
301 {0u}, /* Section 6 */ \
302 {0u}, /* Section 7 */ \
303 { \
304 0u \
305 } /* Section 8u (vault) */ \
306 } \
307 }
308
309 QLIB_SECTIONS_CONF_TABLE_T cfgSectionArray_l[QLIB_MAX_DIES_SUPPORTED] =
310 { FLASH_DIE_LEGACY_SECTION_ARRAY, // die 0u
311#if QLIB_MAX_DIES_SUPPORTED > 1u
312 FLASH_DIE_LEGACY_SECTION_ARRAY, // die 1
313
314#endif // QLIB_MAX_DIES_SUPPORTED > 1u
315#if QLIB_MAX_DIES_SUPPORTED == 4u
316 FLASH_DIE_LEGACY_SECTION_ARRAY, // die 2
317 FLASH_DIE_LEGACY_SECTION_ARRAY, // die 3
318
319#endif // QLIB_MAX_DIES_SUPPORTED == 4u
320 };
321
322 QLIB_FLASH_CONFIG_T flashCfg_l = {
323 {0u}, // watchdogDefault
324 {
325 // pinMux
326 QLIB_IO23_MODE__QUAD, // IO2 and IO3 mux
327 (bool)true, // Dedicated reset-in mux
328 },
329 {
330 24u, // addrLen
331 3u, // sidLen
332#ifdef SPI_INIT_ADDRESS_MODE_4_BYTES
333 QLIB_STD_ADDR_MODE__4_BYTE, // addrMode
334#else
335 QLIB_STD_ADDR_MODE__3_BYTE, // addrMode
336#endif // SPI_INIT_ADDRESS_MODE_4_BYTES
337 },
338 RECOVERY_FAST_READ_DUMMY, // fastReadDummyCycles
339 (bool)false // dqsDisable
340 };
341
342 QLIB_DIE_CONFIG_T dieCfg_l =
343 { {0u}, // deviceMasterKey
344 {0u}, // deviceSecretKey
345 {0u}, // preProvisionedMasterKey
346 {0u}, // suid
347 (bool)false, // safeFB
348 (bool)false, // Fallback fail
349 (bool)false, // Fallback select
350 1u, // Fallback mapping
351 (bool)false, // speculCK
352 (bool)true, // nonSecureFormatEn_b
353 (bool)false, // rngPlainAccessEn
354 (bool)false, // ctagModeMulti
355 (bool)false, // devcfgLock
356#if QLIB_MAX_DIES_SUPPORTED > 1
357 (bool)false, // FBIsPerDieConfig
358#endif // QLIB_MAX_DIES_SUPPORTED > 1
359 };
360
361 TOUCH(qlibContext);
362
363 if (NULL != flashCfg)
364 {
365 (void)memcpy((uint8_t*)flashCfg, (uint8_t*)&flashCfg_l, sizeof(QLIB_FLASH_CONFIG_T));
366 flashCfg->stdAddrSize.addrLen_u8 =
367 (LOG2(QLIB_GetFlashSize(qlibContext)) >= 24u ? 24u : (LOG2(QLIB_GetFlashSize(qlibContext)) == 23u ? 23u : 22u));
368 }
369
370 if (NULL != dieCfg)
371 {
372 (void)memcpy((uint8_t*)dieCfg, (uint8_t*)&dieCfg_l, sizeof(QLIB_DIE_CONFIG_T));
373 }
374
375 if (NULL != cfgSectionArray)
376 {
377 (void)memcpy((uint8_t*)cfgSectionArray, (uint8_t*)cfgSectionArray_l, sizeof(cfgSectionArray_l));
378
379 for (dieId_u32 = 0u; dieId_u32 < numOfDies_u32; ++dieId_u32)
380 {
381 cfgSectionArray[dieId_u32].sectionConfigTable[0].size = MIN(QLIB_GetFlashSize(qlibContext), QLIB_MAX_SECTION_SIZE);
382 cfgSectionArray[dieId_u32].sectionConfigTable[1].size =
383 (cfgSectionArray[dieId_u32].sectionConfigTable[0].size == QLIB_GetFlashSize(qlibContext))
384 ? 0u
385 : MIN((QLIB_GetFlashSize(qlibContext) - QLIB_MAX_SECTION_SIZE), QLIB_MAX_SECTION_SIZE);
386 cfgSectionArray[dieId_u32].sectionConfigTable[1].baseAddr =
387 (cfgSectionArray[dieId_u32].sectionConfigTable[1].size > 0u) ? QLIB_MAX_SECTION_SIZE : 0u;
388
389 if (QLIB_FeatureSupported(qlibContext, W77Q_FEATURE_VAULT) != 0u)
390 {
391 cfgSectionArray[dieId_u32].sectionConfigTable[QLIB_SECTION_ID_VAULT].size =
392 (QLIB_FeatureSupported(qlibContext, W77Q_FEATURE_VAULT_128KB) != 0u) ? SIZE_128KB_ : SIZE_64KB_;
393 }
394 }
395 }
396}
