92QLIB_STATUS_T QLIB_SAMPLE_LmsAttest(void* userData_pv)
93{
94 QLIB_CONTEXT_T qlibContext;
95 QLIB_STATUS_T status = QLIB_STATUS__OK;
96 QLIB_BUS_FORMAT_T busFormat = QLIB_BUS_FORMAT(QLIB_BUS_MODE_1_1_1, (bool)false);
97
98 /*-------------------------------------------------------------------------------------------------------
99 Init QLIB - needed when running QLIB either on a device or on a remote server.
100 -------------------------------------------------------------------------------------------------------*/
101 QLIB_STATUS_RET_CHECK(QLIB_InitLib(&qlibContext));
102
103 /*-------------------------------------------------------------------------------------------------------
104 Set the user data into Qlib's context
105 -------------------------------------------------------------------------------------------------------*/
106 QLIB_SetUserData(&qlibContext, userData_pv);
107
108 /*-------------------------------------------------------------------------------------------------------
109 Take the ownership of flash communication channel (it belongs to local MCU or to remote server, exclusively)
110 -------------------------------------------------------------------------------------------------------*/
111 QLIB_STATUS_RET_CHECK(QLIB_Connect(&qlibContext));
112
113 /*-------------------------------------------------------------------------------------------------------
114 Init Flash Device - not needed when using QLIB on a remote server
115 -------------------------------------------------------------------------------------------------------*/
116 QLIB_STATUS_RET_CHECK_GOTO(QLIB_InitDevice(&qlibContext, busFormat), status, disconnect);
117
118 /*-------------------------------------------------------------------------------------------------------
119 Run manufacture stage code
120 -------------------------------------------------------------------------------------------------------*/
121 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_LmsAttest_manuf_flow(&qlibContext), status, disconnect);
122
123 /*-------------------------------------------------------------------------------------------------------
124 Run sample flow
125 -------------------------------------------------------------------------------------------------------*/
126 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_LmsAttest_field_flow(&qlibContext), status, disconnect);
127
128 /*-------------------------------------------------------------------------------------------------------
129 Release the ownership of flash communication channel (it belongs to local MCU or to remote server, exclusively)
130 -------------------------------------------------------------------------------------------------------*/
131 QLIB_STATUS_RET_CHECK(QLIB_Disconnect(&qlibContext));
132
133 return QLIB_STATUS__OK;
134
135disconnect:
136 (void)QLIB_Disconnect(&qlibContext);
137 return status;
138}

140QLIB_STATUS_T QLIB_SAMPLE_LmsAttest_manuf_flow(QLIB_CONTEXT_T* qlibContext)
141{
142 QLIB_STATUS_T status = QLIB_STATUS__OK;
143 LMS_ATTEST_CHUNK_T pubCache[LMS_ATTEST_CACHE_SIZE] = LMS_ATTEST_CACHE;
144 LMS_ATTEST_PUB_KEY_T lmsPubKey;
145
146 /*-------------------------------------------------------------------------------------------------------
147 Run sample code
148 -------------------------------------------------------------------------------------------------------*/
149
150#if QLIB_MAX_DIES_SUPPORTED > 1u
151 // DIE 0 is used for supply chain protection
152 QLIB_STATUS_RET_CHECK(QLIB_SetActiveDie(qlibContext, 0u));
153#endif // QLIB_MAX_DIES_SUPPORTED > 1u
154
155 /*-------------------------------------------------------------------------------------------------------
156 Generate LMS public key and public cache
157 -------------------------------------------------------------------------------------------------------*/
158 status = QLIB_LMS_Attest_GetPublicKey(qlibContext, lmsPubKey, pubCache, LMS_ATTEST_CACHE_SIZE);
159
160 if (status == QLIB_STATUS__SYSTEM_IN_INCORRECT_STATE)
161 {
162#ifndef QLIB_FPGA_TARGET
163 QLIB_SAMPLE_PRINTF("Error: The LMS private key must be set for this test");
164 /*---------------------------------------------------------------------------------------------------
165 please note that setting the LMS private key can be done only once in flash life cycle
166 consider carefully before setting it. In order to set the private key use the following command
167 QLIB_LMS_KEY_ID_T keyId = {LMS_PRIV_KEY};
168 LMS_ATTEST_PRIVATE_SEED_T seed = {LMS_PRIV_SEED};
169 QLIB_STATUS_RET_CHECK(QLIB_LMS_Attest_SetPrivateKey(qlibContext, seed, keyId));
170 ---------------------------------------------------------------------------------------------------*/
171 goto exit;
172#else // QLIB_FPGA_TARGET
173 /*---------------------------------------------------------------------------------------------------
174 Provision private key - only required for FPGA
175 ---------------------------------------------------------------------------------------------------*/
176 QLIB_LMS_KEY_ID_T keyId = LMS_ATTEST_SAMPLE_LMS_KEY_ID;
177 LMS_ATTEST_PRIVATE_SEED_T seed = LMS_ATTEST_PRIVATE_SEED;
178 QLIB_STATUS_RET_CHECK(QLIB_LMS_Attest_SetPrivateKey(qlibContext, seed, keyId));
179 QLIB_STATUS_RET_CHECK(QLIB_LMS_Attest_GetPublicKey(qlibContext, lmsPubKey, pubCache, LMS_ATTEST_CACHE_SIZE));
180#endif // QLIB_FPGA_TARGET
181 }
182 else
183 {
184 QLIB_STATUS_RET_CHECK(status);
185 }
186
187 /*-------------------------------------------------------------------------------------------------------
188 public key is stored during manufacture stage (in vault section, DB or other secure location)
189 and used later when verifying the signature on field
190 -------------------------------------------------------------------------------------------------------*/
191 QLIB_SAMPLE_LmsAttest_storePub_L(lmsPubKey, pubCache);
192
193exit:
194 return QLIB_STATUS__OK;
195}

197QLIB_STATUS_T QLIB_SAMPLE_LmsAttest_field_flow(QLIB_CONTEXT_T* qlibContext)
198{
199 QLIB_STATUS_T status = QLIB_STATUS__OK;
200 uint8_t msg_u8[] = LMS_ATTEST_MSG;
201 LMS_ATTEST_NONCE_T nonce = LMS_ATTEST_NONCE;
202 LMS_ATTEST_CHUNK_T pubCache[LMS_ATTEST_CACHE_SIZE] = LMS_ATTEST_CACHE;
203 LMS_ATTEST_SIG_T lmsSig;
204 QLIB_LMS_KEY_ID_T wKeyId;
205 ALG_LMS_T lmsAlg = {0u};
206 LMS_ATTEST_PUB_KEY_T lmsPubKey;
207
208 /*-------------------------------------------------------------------------------------------------------
209 Run sample code
210 -------------------------------------------------------------------------------------------------------*/
211
212#if QLIB_MAX_DIES_SUPPORTED > 1u
213 // DIE 0 is used for supply chain protection
214 QLIB_STATUS_RET_CHECK(QLIB_SetActiveDie(qlibContext, 0u));
215#endif // QLIB_MAX_DIES_SUPPORTED > 1u
216
217 /*-------------------------------------------------------------------------------------------------------
218 Get the stored public key and public cache
219 -------------------------------------------------------------------------------------------------------*/
220 QLIB_SAMPLE_LmsAttest_retrievePub_L(lmsPubKey, pubCache);
221
222 /*-------------------------------------------------------------------------------------------------------
223 Sign using flash internal private key
224 -------------------------------------------------------------------------------------------------------*/
225 QLIB_STATUS_RET_CHECK(
226 QLIB_LMS_Attest_Sign(qlibContext, msg_u8, sizeof(msg_u8), nonce, pubCache, LMS_ATTEST_CACHE_SIZE, lmsSig, wKeyId));
227
228 /*-------------------------------------------------------------------------------------------------------
229 Verify the signature itself
230 -------------------------------------------------------------------------------------------------------*/
231 {
232 QLIB_ASSERT_RET(ALG_SUCCESS == lmsInit(&lmsAlg), QLIB_STATUS__COMMAND_FAIL);
233 LMS_VERIFY_SIGNATURE_OP_T verifyParams = {lmsPubKey,
234 (uint32_t)sizeof(lmsPubKey),
235 msg_u8,
236 (uint32_t)sizeof(msg_u8),
237 lmsSig,
238 (uint32_t)sizeof(LMS_ATTEST_SIG_T)};
239 QLIB_ASSERT_WITH_ERROR_GOTO(ALG_SUCCESS == lmsVerifySignature(&lmsAlg, &verifyParams),
240 QLIB_STATUS__COMMAND_FAIL,
241 status,
242 lms_finit);
243 (void)lmsFinit(&lmsAlg);
244 }
245
246 goto exit;
247
248lms_finit:
249 (void)lmsFinit(&lmsAlg);
250 goto exit;
251
252exit:
253 return status;
254}
