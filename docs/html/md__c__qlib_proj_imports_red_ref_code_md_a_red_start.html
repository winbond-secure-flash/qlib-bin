All C code samples are functions that accept a pointer to an initialized QLIB context. For example, call the sample function security_flow() as follows:

#include "qlib.h"
int fw_code()
{
QLIB_BUS_FORMAT_T busFormat = QLIB_BUS_FORMAT(QLIB_BUS_MODE_1_1_1, FALSE);
QLIB_CONTEXT_T qlibContext;
void* deviceUserData = getDeviceHandle();
QLIB_STATUS_RET_CHECK(QLIB_InitLib(&qlibContext));
// The device handle is passed to the library's SPI callback.
QLIB_SetUserData(qlibContext, deviceUserData);
// Initialize the target. This step is not needed on a remote server.
QLIB_STATUS_RET_CHECK(QLIB_Connect(&qlibContext));
QLIB_STATUS_RET_CHECK(QLIB_InitDevice(&qlibContext, busFormat));
QLIB_STATUS_T status = security_flow(&qlibContext);
//...rest of the code
}
