33QLIB_STATUS_T QLIB_SAMPLE_DirectAttestation(void* userData_pv)
34{
35 QLIB_CONTEXT_T qlibContext;
36 QLIB_STATUS_T status = QLIB_STATUS__OK;
37 QLIB_BUS_FORMAT_T busFormat = QLIB_BUS_FORMAT(QLIB_BUS_MODE_1_1_1, (bool)false);
38
39 /*-------------------------------------------------------------------------------------------------------
40 Initialize QLIB - needed when running QLIB either on a device or on a remote server.
41 -------------------------------------------------------------------------------------------------------*/
42 QLIB_STATUS_RET_CHECK(QLIB_InitLib(&qlibContext));
43
44 /*-------------------------------------------------------------------------------------------------------
45 Set the user data into Qlib's context
46 -------------------------------------------------------------------------------------------------------*/
47 QLIB_SetUserData(&qlibContext, userData_pv);
48
49 /*-------------------------------------------------------------------------------------------------------
50 Take the ownership of flash communication channel (it belongs to local MCU or to remote server, exclusively)
51 -------------------------------------------------------------------------------------------------------*/
52 QLIB_STATUS_RET_CHECK(QLIB_Connect(&qlibContext));
53
54 /*-------------------------------------------------------------------------------------------------------
55 Init Flash Device - not needed when using QLIB on a remote server
56 -------------------------------------------------------------------------------------------------------*/
57 QLIB_STATUS_RET_CHECK_GOTO(QLIB_InitDevice(&qlibContext, busFormat), status, disconnect);
58
59 /*-------------------------------------------------------------------------------------------------------
60 configuration for sample
61 -------------------------------------------------------------------------------------------------------*/
62 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_DirectAttestation_manuf_flow(&qlibContext), status, disconnect);
63
64 /*-------------------------------------------------------------------------------------------------------
65 Run sample flow
66 -------------------------------------------------------------------------------------------------------*/
67 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_DirectAttestation_field_flow(&qlibContext), status, disconnect);
68
69disconnect:
70 (void)QLIB_Disconnect(&qlibContext);
71 return status;
72}

74QLIB_STATUS_T QLIB_SAMPLE_DirectAttestation_manuf_flow(QLIB_CONTEXT_T* qlibContext)
75{
76 QLIB_HW_VER_T hwVer;
77 QLIB_ID_T id;
78 uint32_t sect_u32;
79 uint8_t die_u8 = 0u;
80 QLIB_FLASH_CONFIG_T flashCfg = {0u};
81 QLIB_DIE_CONFIG_T dieCfg = {0u};
82 QLIB_SECTIONS_CONF_TABLE_T cfgLine[QLIB_MAX_DIES_SUPPORTED];
83 QLIB_EXTENDED_SECTION_CONF_T* sectCfg;
84 uint8_t diesNum_u8 = QLIB_GET_NUM_OF_DIES(qlibContext);
85 uint8_t writeBuf_u8[QLIB_FLASH_PAGE_SIZE] = {0u};
86 QLIB_STATUS_T status = QLIB_STATUS__OK;
87 uint64_t digest_u64 = 0u;
88
89#if QLIB_MAX_DIES_SUPPORTED > 1u
90 QLIB_GetActiveDie(qlibContext, &die_u8);
91#endif // QLIB_MAX_DIES_SUPPORTED > 1u
92
93 // See what chip is connected and its unique ids. We might need this info to obtain keys for this
94 // chips according to formula or via DB. see QLIB_SAMPLE_GetDeviceKey() function
95 QLIB_STATUS_RET_CHECK(QLIB_GetHWVersion(qlibContext, &hwVer));
96 QLIB_STATUS_RET_CHECK(QLIB_GetId(qlibContext, &id));
97
98 // We do read / modify /write configuration since we want to change only what is needed
99 // read configuration
100 QLIB_STATUS_RET_CHECK(QLIB_GetDeviceConfigMultiDie(qlibContext, &flashCfg, &dieCfg, cfgLine, diesNum_u8));
101
102 // modify configuration
103
104 // set pre provision master key if needed
105 QLIB_STATUS_RET_CHECK(
106 QLIB_SAMPLE_GetDeviceKey(qlibContext, id.sec.wid, QLIB_KID__DEVICE_KEY_PRE_PROVISIONING, dieCfg.preProvisionedMasterKey));
107
108 // set new master key
109 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetDeviceKey(qlibContext, id.sec.wid, QLIB_KID__DEVICE_MASTER, dieCfg.deviceMasterKey));
110
111 // setting layout and policy of the section
112 // this section is with digest enabled
113 sect_u32 = QLIB_SAMPLE_DIRECT_ATTEST_SECTION;
114
115 sectCfg = cfgLine[die_u8].sectionConfigTable;
116
117 sectCfg[sect_u32].baseAddr = QLIB_SAMPLE_RESERVED_BOOT_SECTION_SIZE;
118 sectCfg[sect_u32].size = SIZE_256KB_;
119
120 // first create the section with digest disabled
121 sectCfg[sect_u32].sectionConfig.policy.plainAccessReadEnable = 1u;
122 sectCfg[sect_u32].sectionConfig.policy.plainAccessWriteEnable = 1u;
123 sectCfg[sect_u32].sectionConfig.policy.authPlainAccess = 0u;
124 sectCfg[sect_u32].sectionConfig.policy.checksumIntegrity = 0u;
125 sectCfg[sect_u32].sectionConfig.policy.digestIntegrity = 0u;
126 sectCfg[sect_u32].sectionConfig.policy.digestIntegrityOnAccess = 0u;
127 sectCfg[sect_u32].sectionConfig.policy.rollbackProt = 0u;
128 sectCfg[sect_u32].sectionConfig.policy.slog = 0u;
129 sectCfg[sect_u32].sectionConfig.policy.writeProt = 0u;
130
131 // Full access key configuration.This is the key to use later for the section configuration
132 // changes or write access
133 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSectionAccessKey(qlibContext,
134 id.sec.wid,
135 QLIB_KID__FULL_ACCESS_SECTION,
136 sect_u32,
137 die_u8,
138 sectCfg[sect_u32].fullAccessKey));
139
140 // Restricted access key configuration. This is the key used later by the field flow to open a
141 // restricted-access session
142 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSectionAccessKey(qlibContext,
143 id.sec.wid,
144 QLIB_KID__RESTRICTED_ACCESS_SECTION,
145 sect_u32,
146 die_u8,
147 sectCfg[sect_u32].restrictedKey));
148
149 // write configuration
150 QLIB_STATUS_RET_CHECK(QLIB_ConfigDeviceMultiDie(qlibContext, &flashCfg, &dieCfg, cfgLine, diesNum_u8));
151
152 // write configuration sets die to 0. return to orig die
153#if QLIB_MAX_DIES_SUPPORTED > 1u
154 QLIB_STATUS_RET_CHECK(QLIB_SetActiveDie(qlibContext, die_u8));
155#endif // QLIB_MAX_DIES_SUPPORTED > 1u
156
157 // now set section data and enable digest
158 QLIB_STATUS_RET_CHECK(QLIB_EraseSection(qlibContext, sect_u32, (bool)false));
159 QLIB_STATUS_RET_CHECK(QLIB_Write(qlibContext, writeBuf_u8, sect_u32, 0u, QLIB_FLASH_PAGE_SIZE, (bool)false));
160 QLIB_STATUS_RET_CHECK(QLIB_UTILS_CalcDigestWithPadding(writeBuf_u8,
161 QLIB_FLASH_PAGE_SIZE,
162 0xFF,
163 sectCfg[sect_u32].size - QLIB_FLASH_PAGE_SIZE,
164 &digest_u64,
165 NULL));
166
167 sectCfg[sect_u32].sectionConfig.policy.plainAccessReadEnable = 1u;
168 sectCfg[sect_u32].sectionConfig.policy.plainAccessWriteEnable = 0u;
169 sectCfg[sect_u32].sectionConfig.policy.digestIntegrity = 1u;
170 sectCfg[sect_u32].sectionConfig.policy.writeProt = 1u;
171 sectCfg[sect_u32].sectionConfig.digest = digest_u64;
172 sectCfg[sect_u32].sectionConfig.postActions.reload = 0u;
173 sectCfg[sect_u32].sectionConfig.postActions.reset = 0u;
174 sectCfg[sect_u32].sectionConfig.postActions.swap = 0u;
175
176 QLIB_STATUS_RET_CHECK(QLIB_LoadKey(qlibContext, sect_u32, sectCfg[sect_u32].fullAccessKey, (bool)true));
177 QLIB_STATUS_RET_CHECK_GOTO(QLIB_OpenSession(qlibContext, sect_u32, QLIB_SESSION_ACCESS_FULL), status, remove_key);
178 QLIB_STATUS_RET_CHECK_GOTO(QLIB_ConfigDeviceSection(qlibContext, sect_u32, &sectCfg[sect_u32].sectionConfig),
179 status,
180 close_session);
181 QLIB_STATUS_RET_CHECK(QLIB_RemoveKey(qlibContext, sect_u32, (bool)true));
182
183 return QLIB_STATUS__OK;
184
185close_session:
186 (void)QLIB_CloseSession(qlibContext, sect_u32);
187
188remove_key:
189 (void)QLIB_RemoveKey(qlibContext, sect_u32, (bool)true);
190
191 return status;
192}

194QLIB_STATUS_T QLIB_SAMPLE_DirectAttestation_field_flow(QLIB_CONTEXT_T* qlibContext)
195{
196 uint32_t section_u32;
197 QLIB_ID_T id;
198 KEY_T sectionKey;
199 uint8_t die_u8 = 0u;
200 QLIB_STATUS_T status = QLIB_STATUS__OK;
201
202 /*-------------------------------------------------------------------------------------------------------
203 The section should be set with digest integrity
204 -------------------------------------------------------------------------------------------------------*/
205 section_u32 = QLIB_SAMPLE_DIRECT_ATTEST_SECTION;
206
207 /*-------------------------------------------------------------------------------------------------------
208 Access to the Section must be established before using this command.
209 It can be either by granting plain access or via session with restricted or full key.
210 -------------------------------------------------------------------------------------------------------*/
211 QLIB_STATUS_RET_CHECK(QLIB_GetId(qlibContext, &id));
212#if QLIB_MAX_DIES_SUPPORTED > 1u
213 QLIB_STATUS_RET_CHECK(QLIB_GetActiveDie(qlibContext, &die_u8));
214#endif // QLIB_MAX_DIES_SUPPORTED > 1
215 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSectionAccessKey(qlibContext,
216 id.sec.wid,
217 QLIB_KID__RESTRICTED_ACCESS_SECTION,
218 section_u32,
219 die_u8,
220 sectionKey));
221 QLIB_STATUS_RET_CHECK(QLIB_LoadKey(qlibContext, section_u32, sectionKey, (bool)false));
222 QLIB_STATUS_RET_CHECK_GOTO(QLIB_OpenSession(qlibContext, section_u32, QLIB_SESSION_ACCESS_RESTRICTED), status, remove_key);
223
224 /*-------------------------------------------------------------------------------------------------------
225 Check section integrity using digest
226 -------------------------------------------------------------------------------------------------------*/
227 QLIB_STATUS_RET_CHECK_GOTO(QLIB_CheckIntegrity(qlibContext, section_u32, QLIB_INTEGRITY_DIGEST), status, close_session);
228
229close_session:
230 (void)QLIB_CloseSession(qlibContext, section_u32);
231
232remove_key:
233 (void)QLIB_RemoveKey(qlibContext, section_u32, (bool)false);
234 return status;
235}
