// Context structure for Linux spidev
typedef struct {
int fd; // File descriptor for the /dev/spidevB.C device
// Add default speed, mode etc. if needed
} LinuxSpiContext;
// --- PLAT_SPI_WriteReadTransaction Implementation (Linux/spidev) ---
int PLAT_SPI_WriteReadTransaction(const void* userData,
QLIB_BUS_MODE_T format,
uint32_t flags,
const uint8_t* dataOutStream,
uint32_t cmdSize,
uint32_t addressSize,
uint32_t dataOutSize,
uint32_t dummyCycles,
uint8_t* dataIn,
uint32_t dataInSize)
{
const LinuxSpiContext* context = (const LinuxSpiContext*)userData;
if (!context || context->fd < 0) {
fprintf(stderr, "PLAT_SPI: Invalid context or file descriptor.\n");
return PLAT_SPI_INV_PARAM;
}
int fd = context->fd;
int ret = PLAT_SPI_OK;
// --- 1. Configure SPI Mode ---
// Map QLIB format to spidev mode bits
uint8_t mode = 0;
switch (format) {
case QLIB_BUS_MODE_SPI_MODE0: mode = SPI_MODE_0; break;
case QLIB_BUS_MODE_SPI_MODE1: mode = SPI_MODE_1; break;
case QLIB_BUS_MODE_SPI_MODE2: mode = SPI_MODE_2; break;
case QLIB_BUS_MODE_SPI_MODE3: mode = SPI_MODE_3; break;
default:
fprintf(stderr, "PLAT_SPI: Unsupported QLIB_BUS_MODE_T: %d\n", format);
return PLAT_SPI_INV_PARAM;
}
// Add SPI_LSB_FIRST, SPI_3WIRE etc. based on 'format' or 'flags' if needed
// Set mode using ioctl (consider doing this less frequently if mode doesn't change often)
if (ioctl(fd, SPI_IOC_WR_MODE, &mode) == -1) {
perror("PLAT_SPI: Failed to set SPI mode (WR_MODE)");
return PLAT_SPI_ERROR;
}
// Set other parameters like bits_per_word, speed via ioctl if needed
// uint8_t bits = 8;
// if (ioctl(fd, SPI_IOC_WR_BITS_PER_WORD, &bits) == -1) { ... }
// uint32_t speed = DEFAULT_SPEED;
// if (flags & PLAT_SPI_FLAG_HIGH_SPEED) speed = HIGH_SPEED;
// if (ioctl(fd, SPI_IOC_WR_MAX_SPEED_HZ, &speed) == -1) { ... }
// --- 2. Prepare Transfers ---
// Use SPI_IOC_MESSAGE to send multiple transfers atomically.
// We'll use 3 transfers: Write(cmd+addr+data), Dummy Write, Read(with dummy write)
struct spi_ioc_transfer xfers[3]; // Max 3 transfers needed
memset(xfers, 0, sizeof(xfers));
int xfer_count = 0;
uint8_t* dummy_buf = NULL; // Buffer for dummy bytes
// Transfer 1: Write Phase (Command + Address + Data)
uint32_t totalWriteSize = cmdSize + addressSize + dataOutSize;
if (totalWriteSize > 0) {
xfers[xfer_count].tx_buf = (unsigned long)dataOutStream;
xfers[xfer_count].rx_buf = 0; // No receive buffer for write phase
xfers[xfer_count].len = totalWriteSize;
xfers[xfer_count].cs_change = 0; // Keep CS asserted after this transfer
// Set speed_hz, bits_per_word etc. here if overriding defaults
xfer_count++;
}
// Transfer 2: Dummy Cycles Phase (Write dummy bytes)
uint32_t num_dummy_bytes = (dummyCycles + 7) / 8;
if (num_dummy_bytes > 0) {
// Need a buffer for dummy bytes. Allocate dynamically or use a static one.
dummy_buf = (uint8_t*)malloc(num_dummy_bytes);
if (!dummy_buf) {
perror("PLAT_SPI: Failed to allocate dummy buffer");
return PLAT_SPI_ERROR; // Or map errno
}
memset(dummy_buf, 0xFF, num_dummy_bytes); // Fill with dummy value (e.g., 0xFF)
xfers[xfer_count].tx_buf = (unsigned long)dummy_buf;
xfers[xfer_count].rx_buf = 0; // No receive buffer
xfers[xfer_count].len = num_dummy_bytes;
xfers[xfer_count].cs_change = 0; // Keep CS asserted
xfer_count++;
}
// Transfer 3: Read Phase (Write dummy bytes while reading)
if (dataInSize > 0 && dataIn) {
// Need to provide a TX buffer (can be NULL for spidev, kernel sends 0s)
// or provide a buffer of dummy bytes. Using NULL is simpler.
xfers[xfer_count].tx_buf = 0; // Kernel will send 0s or configured dummy pattern
xfers[xfer_count].rx_buf = (unsigned long)dataIn;
xfers[xfer_count].len = dataInSize;
// Deassert CS after this final transfer? Check flags.
xfers[xfer_count].cs_change = (flags & PLAT_SPI_FLAG_KEEP_CS_ASSERTED) ? 0 : 1;
xfer_count++;
} else if (xfer_count > 0) {
// If there's no read phase, the last write/dummy phase should deassert CS
if (!(flags & PLAT_SPI_FLAG_KEEP_CS_ASSERTED)) {
xfers[xfer_count - 1].cs_change = 1;
}
}
// --- 3. Perform Transaction ---
if (xfer_count > 0) {
if (ioctl(fd, SPI_IOC_MESSAGE(xfer_count), xfers) < 0) {
perror("PLAT_SPI: Failed SPI_IOC_MESSAGE");
// Translate errno
if (errno == ETIMEDOUT) ret = PLAT_SPI_TIMEOUT;
else if (errno == EBUSY) ret = PLAT_SPI_BUSY;
else ret = PLAT_SPI_ERROR;
}
} else {
ret = PLAT_SPI_OK; // Nothing to transfer
}
// --- 4. Cleanup ---
if (dummy_buf) {
free(dummy_buf);
}
return ret;
}
