int PLAT_SPI_WriteReadTransaction(const void* userData,
QLIB_BUS_MODE_T format,
uint32_t flags,
const uint8_t* dataOutStream,
uint32_t cmdSize,
uint32_t addressSize,
uint32_t dataOutSize,
uint32_t dummyCycles, // Number of dummy BITS
uint8_t* dataIn,
uint32_t dataInSize)
{
// Get FT_HANDLE
FT_HANDLE ftHandle = *(FT_HANDLE*)userData;
// tx
FT4222_STATUS ftStatus;
uint32 sizeTransferred = 0; // To store actual bytes transferred by the call
uint32_t sizeToSend = cmdSize + addressSize + dataOutSize;// + DUMMY_CYCLES_TO_BYTES(format, dummyCycles);
ftStatus = FT4222_SPIMaster_SingleWrite(device->ftHandle, (uint8_t*)dataOutStream, sizeToSend, &sizeTransferred, (dummyCycles + dataInSize) ? 0 : 1);
assert(ftStatus == FT4222_OK && sizeToSend == sizeTransferred);
// dc
uint16_t num_dummy_bytes = (dummyCycles + 7) / 8; // Calculate dummy bytes needed
uint8_t dummyBuffer[] ={0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff};
if ( num_dummy_bytes )
{
assert(num_dummy_bytes <= sizeof(dummyBuffer));
ftStatus = FT4222_SPIMaster_SingleWrite(device->ftHandle, dummyBuffer, num_dummy_bytes, &sizeTransferred, dataInSize ? 0 : 1);
assert(ftStatus == FT4222_OK && num_dummy_bytes == sizeTransferred);
}
// rx
if (dataInSize)
{
ftStatus = FT4222_SPIMaster_SingleRead(device->ftHandle, dataIn, dataInSize, &sizeTransferred, 1);
assert(ftStatus == FT4222_OK && dataInSize == sizeTransferred);
}
return 0;
}
