QLIB_STATUS_T red_LGM_manuf_flow(QLIB_CONTEXT_T* ctx)
{
QLIB_HW_VER_T hwVer;
QLIB_ID_T id;
uint32_t sect = 0, die = 0;
QLIB_FLASH_CONFIG_T flashCfg;
QLIB_DIE_CONFIG_T dieCfg;
QLIB_SECTIONS_CONF_TABLE_T cfgArr[QLIB_MAX_DIES_SUPPORTED];
uint8_t numOfDies = QLIB_GET_NUM_OF_DIES(ctx);
// See what chip is connected and its unique ids. We might need this info to obtain keys for this
// chips according to formula or via DB see fwGetSectionKey() function
QLIB_STATUS_RET_CHECK(QLIB_GetHWVersion(ctx, &hwVer));
QLIB_STATUS_RET_CHECK(QLIB_GetId(ctx, &id));
// We do read / modify /write configuration since we want to change only what is needed
// read configuration
QLIB_STATUS_RET_CHECK(QLIB_GetDeviceConfigMultiDie(ctx, &flashCfg, &dieCfg, cfgArr, numOfDies));
// modify configuration
// set pre provision master key if needed
QLIB_STATUS_RET_CHECK(fwGetMasterKey(ctx, &id, QLIB_KID__DEVICE_KEY_PRE_PROVISIONING, dieCfg.preProvisionedMasterKey));
// set new master key
QLIB_STATUS_RET_CHECK(fwGetMasterKey(ctx, &id, QLIB_KID__DEVICE_MASTER, dieCfg.deviceMasterKey));
// set new keys ( we do only full access ) and layout
QLIB_EXTENDED_SECTION_CONF_T * sectCfg = cfgArr[die].sectionConfigTable;
// setting layout and policy of the section 0
// this section plain read enabled and plain write enabled
sect = 0;
sectCfg[sect].baseAddr = 0;
sectCfg[sect].size = SIZE_1MB_;
sectCfg[sect].sectionConfig.policy.plainAccessReadEnable = 1;
sectCfg[sect].sectionConfig.policy.plainAccessWriteEnable = 1;
sectCfg[sect].sectionConfig.policy.slog = 1;
// Full access key configuration.This is the key to use later for the section configuration
// changes or write access
QLIB_STATUS_RET_CHECK(fwGetSectionKey(
ctx, &id, QLIB_KID__FULL_ACCESS_SECTION, sect, die, sectCfg[sect].fullAccessKey, sizeof(KEY_T)));
QLIB_STATUS_RET_CHECK(fwGetSectionKey(
ctx, &id, QLIB_KID__RESTRICTED_ACCESS_SECTION, sect, die, sectCfg[sect].restrictedKey, sizeof(KEY_T)));
// setting layout of the section 1
// this section plain read enabled and plain write disabled
sect = 1;
sectCfg[sect].baseAddr = sectCfg[sect - 1].baseAddr + sectCfg[sect - 1].size;
sectCfg[sect].size = SIZE_1MB_;
sectCfg[sect].sectionConfig.policy.plainAccessReadEnable = 1;
sectCfg[sect].sectionConfig.policy.plainAccessWriteEnable = 0;
sectCfg[sect].sectionConfig.policy.slog = 1;
// get full access key to configure into chip.
QLIB_STATUS_RET_CHECK(fwGetSectionKey(
ctx, &id, QLIB_KID__FULL_ACCESS_SECTION, sect, die, sectCfg[sect].fullAccessKey, sizeof(KEY_T)));
QLIB_STATUS_RET_CHECK(fwGetSectionKey(
ctx, &id, QLIB_KID__RESTRICTED_ACCESS_SECTION, sect, die, sectCfg[sect].restrictedKey, sizeof(KEY_T)));
// setting layout of the section 2
// this section plain read enabled and plain write disabled
sect = 2;
sectCfg[sect].baseAddr = sectCfg[sect - 1].baseAddr + sectCfg[sect - 1].size;
sectCfg[sect].size = SIZE_1MB_;
sectCfg[sect].sectionConfig.policy.plainAccessReadEnable = 0;
sectCfg[sect].sectionConfig.policy.plainAccessWriteEnable = 1;
sectCfg[sect].sectionConfig.policy.slog = 1;
// get full access key to configure into chip.
QLIB_STATUS_RET_CHECK(fwGetSectionKey(
ctx, &id, QLIB_KID__FULL_ACCESS_SECTION, sect, die, sectCfg[sect].fullAccessKey, sizeof(KEY_T)));
QLIB_STATUS_RET_CHECK(fwGetSectionKey(
ctx, &id, QLIB_KID__RESTRICTED_ACCESS_SECTION, sect, die, sectCfg[sect].restrictedKey, sizeof(KEY_T)));
// setting layout of the section 3
// this section plain read disabled and plain write disabled
sect = 3;
sectCfg[sect].baseAddr = sectCfg[sect - 1].baseAddr + sectCfg[sect - 1].size;
sectCfg[sect].size = SIZE_1MB_;
sectCfg[sect].sectionConfig.policy.plainAccessReadEnable = 0;
sectCfg[sect].sectionConfig.policy.plainAccessWriteEnable = 0;
sectCfg[sect].sectionConfig.policy.slog = 1;
// get full access key to configure into chip.
QLIB_STATUS_RET_CHECK(fwGetSectionKey(
ctx, &id, QLIB_KID__FULL_ACCESS_SECTION, sect, die, sectCfg[sect].fullAccessKey, sizeof(KEY_T)));
// get restricted access key to configure into chip.
QLIB_STATUS_RET_CHECK(fwGetSectionKey(
ctx, &id, QLIB_KID__RESTRICTED_ACCESS_SECTION, sect, die, sectCfg[sect].restrictedKey, sizeof(KEY_T)));
// write configuration
QLIB_STATUS_RET_CHECK(QLIB_ConfigDeviceMultiDie(ctx, &flashCfg, &dieCfg, cfgArr, numOfDies));
return QLIB_STATUS__OK;
}
QLIB_STATUS_T check_log_access( QLIB_CONTEXT_T* ctx, QLIB_ID_T* id, uint32_t die, uint32_t sect,
bool secure, bool fullAccess,
bool *can_erase, bool *can_read, bool *can_write )
{
uint32_t maxLen, currLen;
QLIB_STATUS_T status;
LOG_ENTRY_T entry, entryTest;
KEY_T key;
QLIB_SESSION_ACCESS_T session;
QLIB_KID_TYPE_T keyType;
// how many entries are possible for this section
QLIB_STATUS_RET_CHECK( QLIB_LogMaxSize( ctx, sect, &maxLen));
if (secure)
{
session = fullAccess ? QLIB_SESSION_ACCESS_FULL : QLIB_SESSION_ACCESS_RESTRICTED;
keyType = fullAccess ? QLIB_KID__FULL_ACCESS_SECTION : QLIB_KID__RESTRICTED_ACCESS_SECTION;
QLIB_STATUS_RET_CHECK(fwGetSectionKey(ctx, id, keyType, sect, die, key, sizeof (KEY_T)));
QLIB_STATUS_RET_CHECK(QLIB_LoadKey(ctx, sect, key, fullAccess));
QLIB_STATUS_RET_CHECK(QLIB_OpenSession(ctx, sect, session));
}
// erase log section check
status = QLIB_EraseSection(ctx, sect, secure);
*can_erase = (QLIB_STATUS__OK == status);
// creating/reading log entry
memset(entry, sect, sizeof(entry));
status = QLIB_LogAppend(ctx, entry, sect);
*can_write = (QLIB_STATUS__OK == status);
// how many entries are currently in the section should be equal to reading
status = QLIB_LogUsedEntries( ctx, sect, &currLen);
*can_read = (QLIB_STATUS__OK == status);
if (*can_read && currLen > 0)
{
status = QLIB_LogAt(ctx, entryTest, sect, currLen - 1 );
QLIB_ASSERT_RET(QLIB_STATUS__OK == status, QLIB_STATUS__TEST_FAIL);
if (*can_write)
{
QLIB_ASSERT_RET(0 == memcmp(entry, entryTest, sizeof(LOG_ENTRY_T)), QLIB_STATUS__COMMAND_FAIL);
}
}
if (secure)
{
QLIB_STATUS_RET_CHECK(QLIB_CloseSession(ctx, sect));
QLIB_STATUS_RET_CHECK(QLIB_RemoveKey(ctx, sect, fullAccess));
}
return QLIB_STATUS__OK;
}
