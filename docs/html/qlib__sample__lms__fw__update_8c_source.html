77QLIB_STATUS_T QLIB_SAMPLE_LmsFwUpdate(void* userData_pv)
78{
79 QLIB_CONTEXT_T qlibContext;
80 QLIB_STATUS_T status = QLIB_STATUS__OK;
81 QLIB_BUS_FORMAT_T busFormat = QLIB_BUS_FORMAT(QLIB_BUS_MODE_1_1_1, (bool)false);
82
83 /*-------------------------------------------------------------------------------------------------------
84 Init QLIB - needed when running QLIB either on a device or on a remote server.
85 -------------------------------------------------------------------------------------------------------*/
86 QLIB_STATUS_RET_CHECK(QLIB_InitLib(&qlibContext));
87
88 /*-------------------------------------------------------------------------------------------------------
89 Set the user data into Qlib's context
90 -------------------------------------------------------------------------------------------------------*/
91 QLIB_SetUserData(&qlibContext, userData_pv);
92
93 /*-------------------------------------------------------------------------------------------------------
94 Take the ownership of flash communication channel (it belongs to local MCU or to remote server, exclusively)
95 -------------------------------------------------------------------------------------------------------*/
96 QLIB_STATUS_RET_CHECK(QLIB_Connect(&qlibContext));
97
98 /*-------------------------------------------------------------------------------------------------------
99 Init Flash Device - not needed when using QLIB on a remote server
100 -------------------------------------------------------------------------------------------------------*/
101 QLIB_STATUS_RET_CHECK_GOTO(QLIB_InitDevice(&qlibContext, busFormat), status, disconnect);
102
103 /*-------------------------------------------------------------------------------------------------------
104 configure flash for sample flow
105 -------------------------------------------------------------------------------------------------------*/
106 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_LmsFwUpdate_manuf_flow(&qlibContext), status, disconnect);
107
108 /*-------------------------------------------------------------------------------------------------------
109 Run sample code
110 -------------------------------------------------------------------------------------------------------*/
111 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_LmsFwUpdate_field_flow(&qlibContext), status, disconnect);
112
113disconnect:
114 /*-------------------------------------------------------------------------------------------------------
115 Release the ownership of flash communication channel (it belongs to local MCU or to remote server, exclusively)
116 -------------------------------------------------------------------------------------------------------*/
117 (void)QLIB_Disconnect(&qlibContext);
118
119 return status;
120}

122QLIB_STATUS_T QLIB_SAMPLE_LmsFwUpdate_manuf_flow(QLIB_CONTEXT_T* qlibContext)
123{
124 QLIB_HW_VER_T hwVer;
125 QLIB_ID_T id;
126 uint8_t die_u8 = 0u;
127 uint32_t sect_u32;
128 QLIB_FLASH_CONFIG_T flashCfg = {0u};
129 QLIB_DIE_CONFIG_T dieCfg = {0u};
130 QLIB_SECTIONS_CONF_TABLE_T cfgLine[QLIB_MAX_DIES_SUPPORTED];
131 QLIB_EXTENDED_SECTION_CONF_T* sectCfg;
132 uint8_t diesNum_u8 = QLIB_GET_NUM_OF_DIES(qlibContext);
133 bool isProvisioned_b;
134
135 // See what chip is connected and its unique ids. We might need this info to obtain keys for this
136 // chips according to formula or via DB see QLIB_SAMPLE_GetDeviceKey() function
137 QLIB_STATUS_RET_CHECK(QLIB_GetHWVersion(qlibContext, &hwVer));
138 QLIB_STATUS_RET_CHECK(QLIB_GetId(qlibContext, &id));
139
140 // We do read / modify /write configuration since we want to change only what is needed
141 // read configuration
142 QLIB_STATUS_RET_CHECK(QLIB_GetDeviceConfigMultiDie(qlibContext, &flashCfg, &dieCfg, cfgLine, diesNum_u8));
143
144 // modify configuration
145
146 // set pre provision master key if needed
147 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetDeviceKey(qlibContext,
148 id.std.uniqueID,
149 QLIB_KID__DEVICE_KEY_PRE_PROVISIONING,
150 dieCfg.preProvisionedMasterKey));
151
152 // set new master key
153 QLIB_STATUS_RET_CHECK(
154 QLIB_SAMPLE_GetDeviceKey(qlibContext, id.std.uniqueID, QLIB_KID__DEVICE_MASTER, dieCfg.deviceMasterKey));
155
156#if QLIB_MAX_DIES_SUPPORTED > 1u
157 QLIB_STATUS_RET_CHECK(QLIB_GetActiveDie(qlibContext, &die_u8));
158#endif // QLIB_MAX_DIES_SUPPORTED > 1u
159 sectCfg = cfgLine[die_u8].sectionConfigTable;
160
161 sectCfg[1].size = 0u;
162 sectCfg[2].size = 0u;
163
164 // this sample uses section 3 for LMS FW update
165 sect_u32 = 3u;
166
167 sectCfg[sect_u32].baseAddr = QLIB_SAMPLE_RESERVED_BOOT_SECTION_SIZE;
168 sectCfg[sect_u32].size = FW_UPDATE_SECTION_SIZE;
169
170 // section configured as rollback protected
171 sectCfg[sect_u32].sectionConfig.policy.plainAccessReadEnable = 1u;
172 sectCfg[sect_u32].sectionConfig.policy.plainAccessWriteEnable = 0u;
173 sectCfg[sect_u32].sectionConfig.policy.authPlainAccess = 0u;
174 sectCfg[sect_u32].sectionConfig.policy.checksumIntegrity = 0u;
175 sectCfg[sect_u32].sectionConfig.policy.digestIntegrity = 0u;
176 sectCfg[sect_u32].sectionConfig.policy.digestIntegrityOnAccess = 0u;
177 sectCfg[sect_u32].sectionConfig.policy.rollbackProt = 1u;
178 sectCfg[sect_u32].sectionConfig.policy.slog = 0u;
179 sectCfg[sect_u32].sectionConfig.policy.writeProt = 0u;
180
181 // Full access key configuration.This is the key to use later for the section configuration
182 // changes or write access
183 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSectionAccessKey(qlibContext,
184 id.std.uniqueID,
185 QLIB_KID__FULL_ACCESS_SECTION,
186 sect_u32,
187 die_u8,
188 sectCfg[sect_u32].fullAccessKey));
189
190 // configure LMS public key.
191 QLIB_STATUS_RET_CHECK(QLIB_IsKeyProvisioned(qlibContext, QLIB_KID__SECTION_LMS, sect_u32, &isProvisioned_b));
192
193 // lms public key can be already inside, we assume that this function was already ran before
194 if ((bool)false == isProvisioned_b)
195 {
196#ifndef LMS_SAMPLE_PREPARE_IN_ADVANCE
197 // lms public key is obtained from hsm
198 LMS_PUB_KEY_T publicKey_u8;
199 QLIB_LMS_KEY_ID_T keyId;
200
201 QLIB_STATUS_RET_CHECK(QLIB_LMS_HsmGenerateKeyPair(keyId));
202 QLIB_SAMPLE_Lms_storeKid_L(keyId);
203 QLIB_STATUS_RET_CHECK(QLIB_LMS_HsmGetKey(keyId, publicKey_u8));
204#else
205 // pre-prepared key pair for seed 2923be84e16cd6ae529049f1f1bbe9ebb3a6db3c870c3e99
206 uint8_t publicKey_u8[] = LMS_PUBLIC_KEY_DEF;
207#endif // LMS_SAMPLE_PREPARE_IN_ADVANCE
208
209 // copy to chip configuration data structure
210 (void)memcpy(sectCfg[sect_u32].lmsKey, publicKey_u8, sizeof(LMS_PUB_KEY_T));
211 }
212
213 // write configuration
214 QLIB_STATUS_RET_CHECK(QLIB_ConfigDeviceMultiDie(qlibContext, &flashCfg, &dieCfg, cfgLine, diesNum_u8));
215
216#if QLIB_MAX_DIES_SUPPORTED > 1u
217 // write configuration sets die to 0. return to orig die
218 QLIB_STATUS_RET_CHECK(QLIB_SetActiveDie(qlibContext, die_u8));
219#endif // QLIB_MAX_DIES_SUPPORTED > 1u
220
221 return QLIB_STATUS__OK;
222}

339 QLIB_STATUS_RET_CHECK(QLIB_LoadKey(qlibContext, sectionID_u32, key, (bool)true));
340
341 /*---------------------------------------------------------------------------------------------------
342 Erase upper half of the sector & copy the FW buffer
343 ---------------------------------------------------------------------------------------------------*/
344 QLIB_STATUS_RET_CHECK_GOTO(QLIB_OpenSession(qlibContext, sectionID_u32, QLIB_SESSION_ACCESS_FULL), status, remove_key);
345 QLIB_STATUS_RET_CHECK_GOTO(QLIB_Erase(qlibContext,
346 sectionID_u32,
347 FW_UPDATE_SECTION_SIZE / 2u,
348 FW_UPDATE_SECTION_SIZE / 2u,
349 (bool)true),
350 status,
351 close_session);
352 QLIB_STATUS_RET_CHECK_GOTO(QLIB_Write(qlibContext,
353 (const uint8_t*)fw,
354 sectionID_u32,
355 (FW_UPDATE_SECTION_SIZE / 2u),
356 sizeof(fw),
357 (bool)true),
358 status,
359 close_session);
360
361 QLIB_STATUS_RET_CHECK_GOTO(QLIB_CloseSession(qlibContext, sectionID_u32), status, remove_key);
362 QLIB_STATUS_RET_CHECK(QLIB_RemoveKey(qlibContext, sectionID_u32, (bool)true));
363
364 /*-------------------------------------------------------------------------------------------------------
365 Get current section version
366 -------------------------------------------------------------------------------------------------------*/
367 QLIB_STATUS_RET_CHECK(QLIB_GetDeviceSectionConfig(qlibContext, sectionID_u32, NULL, NULL, &sectionConfig));
368#ifndef LMS_SAMPLE_PREPARE_IN_ADVANCE
369 newVersion_u32 = sectionConfig.version + 1u;
370#else
371 /*---------------------------------------------------------------------------------------------------
372 Verify that the version in the LMS command is greater than the version in the designated section
373 ---------------------------------------------------------------------------------------------------*/
374 QLIB_ASSERT_RET(sectionConfig.version <= newVersion_u32, QLIB_STATUS__TEST_FAIL);
375
376#endif // LMS_SAMPLE_PREPARE_IN_ADVANCE
377
378#ifndef LMS_SAMPLE_PREPARE_IN_ADVANCE
379 /*-------------------------------------------------------------------------------------------------------
380 Moving FW to aligned location to compute expected section integrity
381 -------------------------------------------------------------------------------------------------------*/
382 (void)memset(fw32_u32, 0xff, sizeof(fw32_u32));
383 (void)memcpy((void*)fw32_u32, (void*)fw, sizeof(fw));
384
385 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_LmsCalcSectionIntegrity_L(fw32_u32,
386 sizeof(fw32_u32),
387 FW_UPDATE_SECTION_SIZE,
388 (bool)true,
389 &sectionConfig.digest,
390 digestExt_au8,
391 &sectionConfig.crc));
392
393 /*-------------------------------------------------------------------------------------------------------
394 Create the set_scr command
395 -------------------------------------------------------------------------------------------------------*/
396 sectionConfig.policy.digestIntegrity = 1u;
397 sectionConfig.policy.digestIntegrityOnAccess = 1u;
398 sectionConfig.policy.checksumIntegrity = 1u;
399 sectionConfig.policy.rollbackProt = 1u;
400 sectionConfig.policy.plainAccessReadEnable = 1u;
401 sectionConfig.version = newVersion_u32;
402 sectionConfig.postActions.reload = 0u;
403 sectionConfig.postActions.reset = 0u;
404 sectionConfig.postActions.swap = 1u;
405 if (QLIB_FeatureSupported(qlibContext, W77Q_FEATURE_DIGEST_EXT) == 1u)
406 {
407 QLIB_STATUS_RET_CHECK(QLIB_UTILS_LMS_CreateSetScrMsg(sectionID_u32, &sectionConfig, digestExt_au8, lmsMsg));
408 }
409 else
410 {
411 QLIB_STATUS_RET_CHECK(QLIB_UTILS_LMS_CreateSetScrMsg(sectionID_u32, &sectionConfig, NULL, lmsMsg));
412 }
413
414 /*-------------------------------------------------------------------------------------------------------
415 Initialize KeyId to the stored key id that match the configured public key on flash
416 -------------------------------------------------------------------------------------------------------*/
417 QLIB_SAMPLE_Lms_retrieveKid_L(keyId);
418
419 /*-------------------------------------------------------------------------------------------------------
420 sign the set_scr command
421 -------------------------------------------------------------------------------------------------------*/
422 QLIB_STATUS_RET_CHECK(QLIB_LMS_HsmSign(&signParams));
423
424#endif // LMS_SAMPLE_PREPARE_IN_ADVANCE
425 /*---------------------------------------------------------------------------------------------------
426 Send the LMS command to the flash
427 ---------------------------------------------------------------------------------------------------*/
428 QLIB_STATUS_RET_CHECK(QLIB_SendLMSCommand(qlibContext, lmsMsg, keyId, (uint8_t*)signature_u32, sectionID_u32));
429
430 /*---------------------------------------------------------------------------------------------------
431 Verify that after the execution of the LMS command, the version of the designated section was updated
432 ---------------------------------------------------------------------------------------------------*/
433 QLIB_STATUS_RET_CHECK(QLIB_GetDeviceSectionConfig(qlibContext, sectionID_u32, NULL, NULL, &sectionConfig));
434 QLIB_ASSERT_RET(newVersion_u32 == sectionConfig.version, QLIB_STATUS__TEST_FAIL);
435
436 goto exit;
437#ifdef LMS_SAMPLE_PREPARE_IN_ADVANCE
438lms_finit:
439 (void)lmsFinit(&lmsAlg);
440 goto exit;
441#endif // LMS_SAMPLE_PREPARE_IN_ADVANCE
442close_session:
443 (void)QLIB_CloseSession(qlibContext, sectionID_u32);
444remove_key:
445 (void)QLIB_RemoveKey(qlibContext, sectionID_u32, (bool)true);
446 goto exit;
447exit:
448 return status;
449}
