44{
45 QLIB_CONTEXT_T qlibContext;
46 QLIB_STATUS_T status = QLIB_STATUS__OK;
47 QLIB_BUS_FORMAT_T busFormat = QLIB_BUS_FORMAT(QLIB_BUS_MODE_1_1_1, (bool)false);
48
49 /*-------------------------------------------------------------------------------------------------------
50 Init QLIB - needed when running QLIB either on a device or on a remote server.
51 -------------------------------------------------------------------------------------------------------*/
52 QLIB_STATUS_RET_CHECK(QLIB_InitLib(&qlibContext));
53
54 /*-------------------------------------------------------------------------------------------------------
55 Set the user data into Qlib's context
56 -------------------------------------------------------------------------------------------------------*/
57 QLIB_SetUserData(&qlibContext, userData_pv);
58
59 /*-------------------------------------------------------------------------------------------------------
60 Take the ownership of flash communication channel (it belongs to local MCU or to remote server, exclusively)
61 -------------------------------------------------------------------------------------------------------*/
62 QLIB_STATUS_RET_CHECK(QLIB_Connect(&qlibContext));
63
64 /*-------------------------------------------------------------------------------------------------------
65 Init Flash Device - not needed when using QLIB on a remote server
66 -------------------------------------------------------------------------------------------------------*/
67 QLIB_STATUS_RET_CHECK_GOTO(QLIB_InitDevice(&qlibContext, busFormat), status, disconnect);

70 Run sample code
71 -------------------------------------------------------------------------------------------------------*/
72 /*-------------------------------------------------------------------------------------------------------
73 Get indications for sample
74 -------------------------------------------------------------------------------------------------------*/
75 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_SecureIndications_flow(&qlibContext), status, disconnect);

77disconnect:
