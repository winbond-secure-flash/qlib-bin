35 /*-------------------------------------------------------------------------------------------------------

53 (unsigned long)flashIds.sec.suid[2],

56 /*-------------------------------------------------------------------------------------------------------
57 Winbond ID is a factory programmed ID containing a device identifier.

59 QLIB_SAMPLE_PRINTF("Winbond ID = 0x%lx%08lx\r\n", (unsigned long)flashIds.sec.wid[1], (unsigned long)flashIds.sec.wid[0]);

64QLIB_STATUS_T QLIB_SAMPLE_GetIDs(void* userData_pv)
65{
66 QLIB_CONTEXT_T qlibContext;
67 QLIB_STATUS_T status = QLIB_STATUS__OK;
68
69 /*-------------------------------------------------------------------------------------------------------
70 Init QLIB - needed when running QLIB either on a device or on a remote server.
71 -------------------------------------------------------------------------------------------------------*/
72 QLIB_STATUS_RET_CHECK(QLIB_InitLib(&qlibContext));
73
74 /*-------------------------------------------------------------------------------------------------------
75 Set the user data into Qlib's context
76 -------------------------------------------------------------------------------------------------------*/
77 QLIB_SetUserData(&qlibContext, userData_pv);
78
79 /*-------------------------------------------------------------------------------------------------------
80 Take the ownership of flash communication channel (it belongs to local MCU or to remote server, exclusively)
81 -------------------------------------------------------------------------------------------------------*/
82 QLIB_STATUS_RET_CHECK(QLIB_Connect(&qlibContext));
83
84 /*-------------------------------------------------------------------------------------------------------
85 Init Flash Device - not needed when using QLIB on a remote server
86 -------------------------------------------------------------------------------------------------------*/
87 QLIB_STATUS_RET_CHECK_GOTO(QLIB_InitDevice(&qlibContext, QLIB_BUS_FORMAT(QLIB_BUS_MODE_1_1_1, (bool)false)),
88 status,
89 disconnect);
90
91 /*-------------------------------------------------------------------------------------------------------
92 Sample for getting Flash IDs
93 -------------------------------------------------------------------------------------------------------*/
94 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_GetIDs_flow(&qlibContext), status, disconnect);
95
96 /*-------------------------------------------------------------------------------------------------------
97 Release the ownership of flash communication channel (it belongs to local MCU or to remote server, exclusively)
98 -------------------------------------------------------------------------------------------------------*/
99 QLIB_STATUS_RET_CHECK(QLIB_Disconnect(&qlibContext));
100
101 return QLIB_STATUS__OK;
102
103disconnect:
104 (void)QLIB_Disconnect(&qlibContext);
105 return status;
106}
