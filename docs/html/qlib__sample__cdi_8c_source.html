34QLIB_STATUS_T QLIB_SAMPLE_DiceAttestation(void* userData_pv, bool runFieldFlow_b)
35{
36 QLIB_CONTEXT_T qlibContext;
37 QLIB_STATUS_T status = QLIB_STATUS__OK;
38 QLIB_BUS_FORMAT_T busFormat = QLIB_BUS_FORMAT(QLIB_BUS_MODE_1_1_1, (bool)false);
39
40 /*-------------------------------------------------------------------------------------------------------
41 Init QLIB - needed when running QLIB either on a device or on a remote server.
42 -------------------------------------------------------------------------------------------------------*/
43 QLIB_STATUS_RET_CHECK(QLIB_InitLib(&qlibContext));
44
45 /*-------------------------------------------------------------------------------------------------------
46 Set the user data into Qlib's context
47 -------------------------------------------------------------------------------------------------------*/
48 QLIB_SetUserData(&qlibContext, userData_pv);
49
50 /*-------------------------------------------------------------------------------------------------------
51 Take the ownership of flash communication channel (it belongs to local MCU or to remote server, exclusively)
52 -------------------------------------------------------------------------------------------------------*/
53 QLIB_STATUS_RET_CHECK(QLIB_Connect(&qlibContext));
54
55 /*-------------------------------------------------------------------------------------------------------
56 Init Flash Device - not needed when using QLIB on a remote server
57 -------------------------------------------------------------------------------------------------------*/
58 QLIB_STATUS_RET_CHECK_GOTO(QLIB_InitDevice(&qlibContext, busFormat), status, disconnect);
59
60 if (runFieldFlow_b == (bool)false)
61 {
62 /*---------------------------------------------------------------------------------------------------
63 configuration for sample
64 ---------------------------------------------------------------------------------------------------*/
65 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_DiceAttestation_manuf_flow(&qlibContext), status, disconnect);
66 }
67 else
68 {
69 /*---------------------------------------------------------------------------------------------------
70 sample flow
71 ---------------------------------------------------------------------------------------------------*/
72 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_DiceAttestation_field_flow(&qlibContext), status, disconnect);
73 }
74disconnect:
75 (void)QLIB_Disconnect(&qlibContext);
76 return status;
77}

79QLIB_STATUS_T QLIB_SAMPLE_DiceAttestation_manuf_flow(QLIB_CONTEXT_T* qlibContext)
80{
81 QLIB_HW_VER_T hwVer;
82 QLIB_ID_T id;
83 uint32_t sect_u32 = 0u;
84 uint8_t die_u8 = 0u;
85 QLIB_FLASH_CONFIG_T flashCfg = {0u};
86 QLIB_DIE_CONFIG_T dieCfg = {0u};
87 QLIB_SECTIONS_CONF_TABLE_T cfgLine[QLIB_MAX_DIES_SUPPORTED];
88 QLIB_EXTENDED_SECTION_CONF_T* sectCfg;
89 uint8_t diesNum_u8 = QLIB_GET_NUM_OF_DIES(qlibContext);
90 uint32_t startConfSect_u32 = 0u;
91 QLIB_STATUS_T status = QLIB_STATUS__OK;
92 uint8_t writeBuf_u8[QLIB_FLASH_PAGE_SIZE];
93 uint64_t digest_u64 = 0u;
94
95#if QLIB_MAX_DIES_SUPPORTED > 1u
96 QLIB_GetActiveDie(qlibContext, &die_u8);
97#endif // QLIB_MAX_DIES_SUPPORTED > 1u
98
99 (void)memset(writeBuf_u8, 0xa5, sizeof(writeBuf_u8));
100
101 // See what chip is connected and its unique ids. We might need this info to obtain keys for this
102 // chips according to formula or via DB. see QLIB_SAMPLE_GetDeviceKey() function
103 QLIB_STATUS_RET_CHECK(QLIB_GetHWVersion(qlibContext, &hwVer));
104 QLIB_STATUS_RET_CHECK(QLIB_GetId(qlibContext, &id));
105
106 // We do read / modify /write configuration since we want to change only what is needed
107 // read configuration
108 QLIB_STATUS_RET_CHECK(QLIB_GetDeviceConfigMultiDie(qlibContext, &flashCfg, &dieCfg, cfgLine, diesNum_u8));
109
110 // modify configuration
111
112 // set pre provision master key if needed
113 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetDeviceKey(qlibContext,
114 id.std.uniqueID,
115 QLIB_KID__DEVICE_KEY_PRE_PROVISIONING,
116 dieCfg.preProvisionedMasterKey));
117
118 // set new master key
119 QLIB_STATUS_RET_CHECK(
120 QLIB_SAMPLE_GetDeviceKey(qlibContext, id.std.uniqueID, QLIB_KID__DEVICE_MASTER, dieCfg.deviceMasterKey));
121
122 // set device secret key as it participates in the CDI calculation
123 QLIB_STATUS_RET_CHECK(
124 QLIB_SAMPLE_GetDeviceKey(qlibContext, id.std.uniqueID, QLIB_KID__DEVICE_SECRET, dieCfg.deviceSecretKey));
125
126 // set Secure device ID as it participate in the CDI calculation
127 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSecureUID(qlibContext, id.std.uniqueID, dieCfg.suid));
128
129 // configure sections 0, 1, 2 with digest
130 // first configure without digest, then erase section_u32 contents, write some data, calculate digest and enable it on flash
131
132#ifdef QLIB_SUPPORT_XIP
133 // if sample is run in XIP mode, we can not change the boot section configuration.
134 // verify section 0 was already configured with digest for this sample
135 QLIB_ASSERT_RET((cfgLine[die_u8].sectionConfigTable[0].size > 0u) &&
136 (cfgLine[die_u8].sectionConfigTable[0].sectionConfig.policy.digestIntegrity == 1u),
137 QLIB_STATUS__SYSTEM_IN_INCORRECT_STATE);
138 startConfSect_u32 = 1u;
139
140#endif // QLIB_SUPPORT_XIP
141
142#ifdef QLIB_SAMPLE_CDI_PRESERVE_BOOT_SECTION
143 // When running on a board with a live boot loader in section 0, skip reconfiguring the boot section policies.
144 // Its digest will be computed from existing content below if needed.
145 // Set boot section size to 4MB (enough for ~2.2MB boot loader) instead of the full flash.
146 startConfSect_u32 = 1u;
147 cfgLine[die_u8].sectionConfigTable[QLIB_SAMPLE_CDI_BOOT_SECTION].baseAddr = 0u;
148 cfgLine[die_u8].sectionConfigTable[QLIB_SAMPLE_CDI_BOOT_SECTION].size = QLIB_SAMPLE_RESERVED_BOOT_SECTION_SIZE;
149 // Get boot section full access key (needed for digest configuration below)
150 QLIB_STATUS_RET_CHECK(
151 QLIB_SAMPLE_GetSectionAccessKey(qlibContext,
152 id.std.uniqueID,
153 QLIB_KID__FULL_ACCESS_SECTION,
154 QLIB_SAMPLE_CDI_BOOT_SECTION,
155 die_u8,
156 cfgLine[die_u8].sectionConfigTable[QLIB_SAMPLE_CDI_BOOT_SECTION].fullAccessKey));
157#endif // QLIB_SAMPLE_CDI_PRESERVE_BOOT_SECTION
158
159 sectCfg = cfgLine[die_u8].sectionConfigTable;
160 for (sect_u32 = startConfSect_u32; sect_u32 < QLIB_SAMPLE_CDI_NUM_SECTIONS; sect_u32++)
161 {
162 sectCfg[sect_u32].baseAddr = (sect_u32 == 0u ? 0u : QLIB_SAMPLE_RESERVED_BOOT_SECTION_SIZE);
163 sectCfg[sect_u32].size = SIZE_256KB_;
164
165 sectCfg[sect_u32].sectionConfig.policy.checksumIntegrity = 0u;
166 sectCfg[sect_u32].sectionConfig.policy.digestIntegrity = 0u;
167 sectCfg[sect_u32].sectionConfig.policy.digestIntegrityOnAccess = 0u;
168 sectCfg[sect_u32].sectionConfig.policy.rollbackProt = 0u;
169 sectCfg[sect_u32].sectionConfig.policy.slog = 0u;
170 sectCfg[sect_u32].sectionConfig.policy.writeProt = 0u;
171
172 // Full access key configuration.This is the key to use later for the section configuration and write
173 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSectionAccessKey(qlibContext,
174 id.std.uniqueID,
175 QLIB_KID__FULL_ACCESS_SECTION,
176 sect_u32,
177 die_u8,
178 sectCfg[sect_u32].fullAccessKey));
179 }
180
181 // write configuration
182 QLIB_STATUS_RET_CHECK(QLIB_ConfigDeviceMultiDie(qlibContext, &flashCfg, &dieCfg, cfgLine, diesNum_u8));
183
184 // write configuration sets die to 0. return to orig die
185#if QLIB_MAX_DIES_SUPPORTED > 1u
186 QLIB_STATUS_RET_CHECK(QLIB_SetActiveDie(qlibContext, die_u8));
187#endif // QLIB_MAX_DIES_SUPPORTED > 1u
188
189 // now set sections data and add Write Protected and Authenticity Protected
190 QLIB_STATUS_RET_CHECK(QLIB_UTILS_CalcDigestWithPadding(writeBuf_u8,
191 QLIB_FLASH_PAGE_SIZE,
192 0xFF,
193 SIZE_256KB_ - QLIB_FLASH_PAGE_SIZE,
194 &digest_u64,
195 NULL));
196 for (sect_u32 = QLIB_SAMPLE_CDI_BOOT_SECTION; sect_u32 < QLIB_SAMPLE_CDI_NUM_SECTIONS; sect_u32++)
197 {
198 sectCfg = cfgLine[die_u8].sectionConfigTable;
199
200 // If digestIntegrity is already enabled for this section, no need to reconfigure
201 if (sectCfg[sect_u32].sectionConfig.policy.digestIntegrity == 1u)
202 {
203 continue;
204 }
205
206 sectCfg[sect_u32].sectionConfig.policy.digestIntegrity = 1u;
207#ifdef QLIB_SAMPLE_CDI_PRESERVE_BOOT_SECTION
208 // Don't write-protect the boot section — it must remain writable for boot loader updates
209 if (sect_u32 != QLIB_SAMPLE_CDI_BOOT_SECTION)
210#endif // QLIB_SAMPLE_CDI_PRESERVE_BOOT_SECTION
211 sectCfg[sect_u32].sectionConfig.policy.writeProt = 1u;
212 QLIB_STATUS_RET_CHECK(QLIB_LoadKey(qlibContext, sect_u32, sectCfg[sect_u32].fullAccessKey, (bool)true));
213 QLIB_STATUS_RET_CHECK_GOTO(QLIB_OpenSession(qlibContext, sect_u32, QLIB_SESSION_ACCESS_FULL), status, remove_key);
214
215#ifdef QLIB_SAMPLE_CDI_PRESERVE_BOOT_SECTION
216 if (sect_u32 == QLIB_SAMPLE_CDI_BOOT_SECTION)
217 {
218 // Boot section: read existing content and compute its digest incrementally.
219 // Do NOT erase or write — preserve the boot loader.
220 // Section can be large (e.g. 16MB), so we read and hash in page-sized chunks.
221 static uint8_t readBuf_u8[QLIB_FLASH_PAGE_SIZE];
222 uint32_t sectionSize_u32 = sectCfg[sect_u32].size;
223 uint32_t offset_u32 = 0u;
224 void* hashCtx_pv = NULL;
225 TYPE_256BIT hashResult;
226
227 QLIB_SAMPLE_PRINTF("[CDI manuf] section_u32 %u: hashing existing content (%u bytes) in chunks\n",
228 sect_u32,
229 sectionSize_u32);
230 QLIB_ASSERT_RET(PLAT_HASH_Init(&hashCtx_pv, QLIB_HASH_OPT_NONE) == 0u, QLIB_STATUS__HARDWARE_FAILURE);
231 while (offset_u32 < sectionSize_u32)
232 {
233 uint32_t chunkSize_u32 = ((sectionSize_u32 - offset_u32) < QLIB_FLASH_PAGE_SIZE) ? (sectionSize_u32 - offset_u32)
234 : QLIB_FLASH_PAGE_SIZE;
235 status = QLIB_Read(qlibContext, readBuf_u8, sect_u32, offset_u32, chunkSize_u32, (bool)false, (bool)false);
236 if (QLIB_STATUS__OK != status)
237 {
238 goto close_session;
239 }
240 QLIB_ASSERT_RET(PLAT_HASH_Update(hashCtx_pv, readBuf_u8, chunkSize_u32) == 0u, QLIB_STATUS__HARDWARE_FAILURE);
241 offset_u32 += chunkSize_u32;
242 }
243 QLIB_ASSERT_RET(PLAT_HASH_Finish(hashCtx_pv, hashResult) == 0u, QLIB_STATUS__HARDWARE_FAILURE);
244 (void)memcpy((void*)&digest_u64, (void*)&hashResult[6], sizeof(uint64_t));
245 QLIB_SAMPLE_PRINTF("[CDI manuf] section_u32 %u: digest_u64 computed OK\n", sect_u32);
246 }
247 else
248#endif // QLIB_SAMPLE_CDI_PRESERVE_BOOT_SECTION
249 {
250 QLIB_STATUS_RET_CHECK_GOTO(QLIB_EraseSection(qlibContext, sect_u32, (bool)true), status, close_session);
251 QLIB_STATUS_RET_CHECK_GOTO(QLIB_Write(qlibContext, writeBuf_u8, sect_u32, 0u, QLIB_FLASH_PAGE_SIZE, (bool)true),
252 status,
253 close_session);
254 }
255
256 sectCfg[sect_u32].sectionConfig.digest = digest_u64;
257 sectCfg[sect_u32].sectionConfig.postActions.reload = 1u;
258 sectCfg[sect_u32].sectionConfig.postActions.reset = 0u;
259 sectCfg[sect_u32].sectionConfig.postActions.swap = 0u;
260
261 QLIB_STATUS_RET_CHECK_GOTO(QLIB_ConfigDeviceSection(qlibContext, sect_u32, &sectCfg[sect_u32].sectionConfig),
262 status,
263 close_session);
264
265 QLIB_STATUS_RET_CHECK_GOTO(QLIB_CloseSession(qlibContext, sect_u32), status, remove_key);
266 QLIB_STATUS_RET_CHECK(QLIB_RemoveKey(qlibContext, sect_u32, (bool)true));
267 }
268
269 return QLIB_STATUS__OK;
270
271close_session:
272 (void)QLIB_CloseSession(qlibContext, sect_u32);
273remove_key:
274 (void)QLIB_RemoveKey(qlibContext, sect_u32, (bool)true);
275 return status;
276}

278QLIB_STATUS_T QLIB_SAMPLE_DiceAttestation_field_flow(QLIB_CONTEXT_T* qlibContext)
279{
280 TYPE_256BIT nextCdi;
281 TYPE_256BIT prevCdi;
282 QLIB_HW_VER_T hwVer;
283 QLIB_ID_T id;
284 uint8_t die_u8 = 0u;
285 uint32_t section_u32;
286 CONST_KEY_P_T keyF;
287
288#if QLIB_MAX_DIES_SUPPORTED > 1u
289 QLIB_GetActiveDie(qlibContext, &die_u8);
290#endif // QLIB_MAX_DIES_SUPPORTED > 1u
291
292 // See what chip is connected and its unique ids. We might need this info to obtain keys for this
293 // chips according to formula or via DB. see QLIB_SAMPLE_GetSectionKeyPtr() function
294 QLIB_STATUS_RET_CHECK(QLIB_GetHWVersion(qlibContext, &hwVer));
295 QLIB_STATUS_RET_CHECK(QLIB_GetId(qlibContext, &id));
296
297 section_u32 = 0u;
298
299 /*-------------------------------------------------------------------------------------------------------
300 obtain full access key
301 -------------------------------------------------------------------------------------------------------*/
302 QLIB_STATUS_RET_CHECK(
303 QLIB_SAMPLE_GetSectionKeyPtr(qlibContext, id.std.uniqueID, QLIB_KID__FULL_ACCESS_SECTION, section_u32, die_u8, &keyF));
304
305 /*-------------------------------------------------------------------------------------------------------
306 to retrieve first CDI in the fw trust chain, section 0u is used
307 -------------------------------------------------------------------------------------------------------*/
308 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetCdi(qlibContext, nextCdi, NULL, section_u32, keyF));
309
310#if (QLIB_SAMPLE_CDI_NUM_SECTIONS < 3u)
311#error "CDI sample assumes sections 0,1,2 configured with digest"
312#endif // (QLIB_SAMPLE_CDI_NUM_SECTIONS < 3u)
313
314 /*-------------------------------------------------------------------------------------------------------
315 assuming the second firmware resides in section 1u and its policy has digest enabled we use previous(first)
316 cdi to obtain next(second) value
317 ------------------------------------------------------------------------------------------------------*/
318 section_u32 = 1u;
319 QLIB_STATUS_RET_CHECK(
320 QLIB_SAMPLE_GetSectionKeyPtr(qlibContext, id.std.uniqueID, QLIB_KID__FULL_ACCESS_SECTION, section_u32, die_u8, &keyF));
321 (void)memcpy(prevCdi, nextCdi, sizeof(TYPE_256BIT));
322 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetCdi(qlibContext, nextCdi, prevCdi, section_u32, keyF));
323
324 /*-------------------------------------------------------------------------------------------------------
325 assuming the third firmware resides in section 2u and its policy has digest enabled we use previous(second)
326 cdi to obtain next (third) value
327 ------------------------------------------------------------------------------------------------------*/
328 section_u32 = 2u;
329 QLIB_STATUS_RET_CHECK(
330 QLIB_SAMPLE_GetSectionKeyPtr(qlibContext, id.std.uniqueID, QLIB_KID__FULL_ACCESS_SECTION, section_u32, die_u8, &keyF));
331 (void)memcpy(prevCdi, nextCdi, sizeof(TYPE_256BIT));
332 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetCdi(qlibContext, nextCdi, prevCdi, section_u32, keyF));
333
334 /*-------------------------------------------------------------------------------------------------------
335 At this point nextCdi contains last CDI that was derived from three stage boot chain. it can be used further
336 in any attestation mechanism. (for instance server ...)
337 ------------------------------------------------------------------------------------------------------*/
338
339 return QLIB_STATUS__OK;
340}

342QLIB_STATUS_T QLIB_SAMPLE_GetCdi(QLIB_CONTEXT_T* qlibContext,
343 TYPE_256BIT nextCdi,
344 TYPE_256BIT prevCdi,
345 uint32_t sectionId_u32,
346 const KEY_T sectionFullAccessKey)
347{
348 QLIB_STATUS_T status = QLIB_STATUS__OK;
349
350 /*-------------------------------------------------------------------------------------------------------
351 Load the key for the secure write operations
352 -------------------------------------------------------------------------------------------------------*/
353 QLIB_STATUS_RET_CHECK(QLIB_LoadKey(qlibContext, sectionId_u32, sectionFullAccessKey, (bool)true));
354
355 /*-------------------------------------------------------------------------------------------------------
356 Open secure session for the secure commands
357 -------------------------------------------------------------------------------------------------------*/
358 QLIB_STATUS_RET_CHECK_GOTO(QLIB_OpenSession(qlibContext, sectionId_u32, QLIB_SESSION_ACCESS_FULL), status, remove_key);
359
360 /*-------------------------------------------------------------------------------------------------------
361 Retrieving cdi value
362 -------------------------------------------------------------------------------------------------------*/
363 QLIB_STATUS_RET_CHECK_GOTO(QLIB_CalcCDI(qlibContext, nextCdi, prevCdi, sectionId_u32), status, close_session);
364
365close_session:
366 (void)QLIB_CloseSession(qlibContext, sectionId_u32);
367
368remove_key:
369 (void)QLIB_RemoveKey(qlibContext, sectionId_u32, (bool)true);
370
371 return status;
372}
