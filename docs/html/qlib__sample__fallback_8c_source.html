50QLIB_STATUS_T QLIB_SAMPLE_Fallback(void* userData_pv)
51{
52 QLIB_CONTEXT_T qlibContext;
53 QLIB_STATUS_T status = QLIB_STATUS__OK;
54 QLIB_BUS_FORMAT_T busFormat = QLIB_BUS_FORMAT(QLIB_BUS_MODE_1_1_1, (bool)false);
55
56 /*-------------------------------------------------------------------------------------------------------
57 Init QLIB - needed when running QLIB either on a device or on a remote server.
58 -------------------------------------------------------------------------------------------------------*/
59 QLIB_STATUS_RET_CHECK(QLIB_InitLib(&qlibContext));
60
61 /*-------------------------------------------------------------------------------------------------------
62 Set the user data into Qlib's context
63 -------------------------------------------------------------------------------------------------------*/
64 QLIB_SetUserData(&qlibContext, userData_pv);
65
66 /*-------------------------------------------------------------------------------------------------------
67 Take the ownership of flash communication channel (it belongs to local MCU or to remote server, exclusively)
68 -------------------------------------------------------------------------------------------------------*/
69 QLIB_STATUS_RET_CHECK(QLIB_Connect(&qlibContext));
70
71 /*-------------------------------------------------------------------------------------------------------
72 Init Flash Device - not needed when using QLIB on a remote server
73 -------------------------------------------------------------------------------------------------------*/
74 QLIB_STATUS_RET_CHECK_GOTO(QLIB_InitDevice(&qlibContext, busFormat), status, disconnect);
75
76 /*-------------------------------------------------------------------------------------------------------
77 Run sample code
78 -------------------------------------------------------------------------------------------------------*/
79
80#ifndef QLIB_ONLY_FALLBACK
81 /*-------------------------------------------------------------------------------------------------------
82 configuration for sample
83 -------------------------------------------------------------------------------------------------------*/
84 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_Fallback_manuf_flow(&qlibContext), status, disconnect);
85#endif // QLIB_ONLY_FALLBACK
86
87 /*-------------------------------------------------------------------------------------------------------
88 sample flow
89 -------------------------------------------------------------------------------------------------------*/
90 QLIB_STATUS_RET_CHECK_GOTO(QLIB_SAMPLE_Fallback_field_flow(&qlibContext), status, disconnect);
91disconnect:
92 (void)QLIB_Disconnect(&qlibContext);
93 return status;
94}

96QLIB_STATUS_T QLIB_SAMPLE_Fallback_config_section(QLIB_CONTEXT_T* qlibContext,
97 QLIB_ID_T* id,
98 uint32_t die_u32,
99 uint32_t sect_u32,
100 uint32_t* crc_pu32,
101 uint64_t* digest_pu64)
102{
103 KEY_T keyF;
104
105 // section configuration is done via read, modify write
106 QLIB_SECTION_CONF_T cfgNew = {0};
107 QLIB_STATUS_RET_CHECK(QLIB_GetDeviceSectionConfig(qlibContext, sect_u32, NULL, NULL, &cfgNew));
108
109 // section will be write protected
110 cfgNew.policy.plainAccessReadEnable = 1;
111 cfgNew.policy.plainAccessWriteEnable = 0;
112
113 // crc integrity on reboot set
114 if (crc_pu32 != NULL)
115 {
116 cfgNew.policy.checksumIntegrity = 1;
117 cfgNew.crc = *crc_pu32;
118 }
119
120 // digest integrity during configuration set
121 if (digest_pu64 != NULL)
122 {
123 cfgNew.policy.digestIntegrity = 1;
124 cfgNew.digest = *digest_pu64;
125 }
126
127 // we want new policy to be effective immediately
128 cfgNew.postActions.reload = 1;
129
130 // obtain full, access key, load its pointer to library, open session
131 QLIB_STATUS_RET_CHECK(
132 QLIB_SAMPLE_GetSectionAccessKey(qlibContext, id->sec.wid, QLIB_KID__FULL_ACCESS_SECTION, sect_u32, die_u32, keyF));
133
134 // at this point we want to swap between sections
135 QLIB_STATUS_RET_CHECK(QLIB_LoadKey(qlibContext, sect_u32, keyF, true));
136 QLIB_STATUS_RET_CHECK(QLIB_OpenSession(qlibContext, sect_u32, QLIB_SESSION_ACCESS_FULL));
137
138 QLIB_STATUS_RET_CHECK(QLIB_ConfigDeviceSection(qlibContext, sect_u32, &cfgNew));
139
140 // note that since .reload = 1 in case of successful section configuration session is
141 // reopened, so we close it manually
142 QLIB_STATUS_RET_CHECK(QLIB_CloseSession(qlibContext, sect_u32));
143 QLIB_STATUS_RET_CHECK(QLIB_RemoveKey(qlibContext, sect_u32, true));
144
145 return QLIB_STATUS__OK;
146}

149QLIB_STATUS_T QLIB_SAMPLE_Fallback_manuf_flow(QLIB_CONTEXT_T* qlibContext)
150{
151 QLIB_HW_VER_T hwVer;
152 QLIB_ID_T id;
153 uint32_t sect_u32;
154 uint8_t die_u8 = 0u;
155 QLIB_FLASH_CONFIG_T flashCfg = {0u};
156 QLIB_DIE_CONFIG_T dieCfg = {0u};
157 QLIB_SECTIONS_CONF_TABLE_T cfgLine[QLIB_MAX_DIES_SUPPORTED];
158 QLIB_EXTENDED_SECTION_CONF_T* sectCfg;
159 uint8_t diesNum_u8 = QLIB_GET_NUM_OF_DIES(qlibContext);
160 uint32_t crc_u32 = 0u;
161 uint64_t digest_u64 = 0u;
162 uint8_t digest_bytes_u8[8] = {0x4d, 0x34, 0x89, 0xdf, 0xac, 0x39, 0x53, 0xec};
163 uint8_t buf_u8[FW_STRING_SIZE] = FW_STRING_FB;
164
165 // See what chip is connected and its unique ids. We might need this info to obtain keys for this
166 // chips according to formula or via DB see QLIB_SAMPLE_GetDeviceKey() function
167 QLIB_STATUS_RET_CHECK(QLIB_GetHWVersion(qlibContext, &hwVer));
168 QLIB_STATUS_RET_CHECK(QLIB_GetId(qlibContext, &id));
169
170 // We do read / modify /write configuration since we want to change only what is needed
171 // read configuration
172 QLIB_STATUS_RET_CHECK(QLIB_GetDeviceConfigMultiDie(qlibContext, &flashCfg, &dieCfg, cfgLine, diesNum_u8));
173
174 // modify configuration
175
176 // set pre provision master key if needed
177 QLIB_STATUS_RET_CHECK(
178 QLIB_SAMPLE_GetDeviceKey(qlibContext, id.sec.wid, QLIB_KID__DEVICE_KEY_PRE_PROVISIONING, dieCfg.preProvisionedMasterKey));
179
180 // set new master key
181 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetDeviceKey(qlibContext, id.sec.wid, QLIB_KID__DEVICE_MASTER, dieCfg.deviceMasterKey));
182
183#if QLIB_MAX_DIES_SUPPORTED > 1u
184 QLIB_STATUS_RET_CHECK(QLIB_GetActiveDie(qlibContext, &die_u8));
185#endif // QLIB_MAX_DIES_SUPPORTED > 1u
186
187 // setting layout and policy of the section 1
188 // this section plain read enabled and plain write enabled
189
190 dieCfg.safeFB = 1u; // enable safe fallback for the die
191 sect_u32 = 0u;
192 sectCfg = cfgLine[die_u8].sectionConfigTable;
193
194 sectCfg[sect_u32].baseAddr = 0u;
195 sectCfg[sect_u32].size = SIZE_1MB_;
196 sectCfg[sect_u32].sectionConfig.policy.plainAccessReadEnable = 1u;
197 sectCfg[sect_u32].sectionConfig.policy.plainAccessWriteEnable = 1u;
198 sectCfg[sect_u32].sectionConfig.policy.authPlainAccess = 0u;
199 sectCfg[sect_u32].sectionConfig.policy.checksumIntegrity = 0u;
200 sectCfg[sect_u32].sectionConfig.policy.digestIntegrity = 0u;
201 sectCfg[sect_u32].sectionConfig.policy.digestIntegrityOnAccess = 0u;
202 sectCfg[sect_u32].sectionConfig.policy.rollbackProt = 0u;
203 sectCfg[sect_u32].sectionConfig.policy.slog = 0u;
204 sectCfg[sect_u32].sectionConfig.policy.writeProt = 0u;
205
206 // Full access key configuration.This is the key to use later for the section configuration
207 // changes or write access
208 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSectionAccessKey(qlibContext,
209 id.sec.wid,
210 QLIB_KID__FULL_ACCESS_SECTION,
211 sect_u32,
212 die_u8,
213 sectCfg[sect_u32].fullAccessKey));
214 sect_u32 = 1u;
215 sectCfg[sect_u32].baseAddr = 0u;
216 sectCfg[sect_u32].size = SIZE_2MB_;
217 sectCfg[sect_u32].sectionConfig.policy.plainAccessReadEnable = 1u;
218 sectCfg[sect_u32].sectionConfig.policy.plainAccessWriteEnable = 1u;
219 sectCfg[sect_u32].sectionConfig.policy.authPlainAccess = 0u;
220 sectCfg[sect_u32].sectionConfig.policy.checksumIntegrity = 0u;
221 sectCfg[sect_u32].sectionConfig.policy.digestIntegrity = 0u;
222 sectCfg[sect_u32].sectionConfig.policy.digestIntegrityOnAccess = 0u;
223 sectCfg[sect_u32].sectionConfig.policy.rollbackProt = 0u;
224 sectCfg[sect_u32].sectionConfig.policy.slog = 0u;
225 sectCfg[sect_u32].sectionConfig.policy.writeProt = 0u;
226
227 // Full access key configuration.This is the key to use later for the section configuration
228 // changes or write access
229 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSectionAccessKey(qlibContext,
230 id.sec.wid,
231 QLIB_KID__FULL_ACCESS_SECTION,
232 sect_u32,
233 die_u8,
234 sectCfg[sect_u32].fullAccessKey));
235
236 // fall back section, we make plain access for the sake of example
237 sect_u32 = 7u;
238 sectCfg[sect_u32].baseAddr = SIZE_2MB_;
239 sectCfg[sect_u32].size = SIZE_1MB_;
240 sectCfg[sect_u32].sectionConfig.policy.plainAccessReadEnable = 1u;
241 sectCfg[sect_u32].sectionConfig.policy.plainAccessWriteEnable = 1u;
242 sectCfg[sect_u32].sectionConfig.policy.authPlainAccess = 0u;
243 sectCfg[sect_u32].sectionConfig.policy.checksumIntegrity = 0u;
244 sectCfg[sect_u32].sectionConfig.policy.digestIntegrity = 0u;
245 sectCfg[sect_u32].sectionConfig.policy.digestIntegrityOnAccess = 0u;
246 sectCfg[sect_u32].sectionConfig.policy.rollbackProt = 0u;
247 sectCfg[sect_u32].sectionConfig.policy.slog = 0u;
248 sectCfg[sect_u32].sectionConfig.policy.writeProt = 0u;
249
250 // Full access key configuration.This is the key to use later for the section configuration
251 // changes or write access
252 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSectionAccessKey(qlibContext,
253 id.sec.wid,
254 QLIB_KID__FULL_ACCESS_SECTION,
255 sect_u32,
256 die_u8,
257 sectCfg[sect_u32].fullAccessKey));
258
259 // get restricted access key to configure into chip.
260 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_GetSectionAccessKey(qlibContext,
261 id.sec.wid,
262 QLIB_KID__RESTRICTED_ACCESS_SECTION,
263 sect_u32,
264 die_u8,
265 sectCfg[sect_u32].restrictedKey));
266
267 // write configuration
268 QLIB_STATUS_RET_CHECK(QLIB_ConfigDeviceMultiDie(qlibContext, &flashCfg, &dieCfg, cfgLine, diesNum_u8));
269
270#if QLIB_MAX_DIES_SUPPORTED > 1u
271 // write configuration sets die to 0. return to orig die
272 QLIB_STATUS_RET_CHECK(QLIB_SetActiveDie(qlibContext, die_u8));
273#endif // QLIB_MAX_DIES_SUPPORTED > 1u
274
275 QLIB_STATUS_RET_CHECK(QLIB_EraseSection(qlibContext, 0, false));
276 QLIB_STATUS_RET_CHECK(QLIB_EraseSection(qlibContext, 1, false));
277 QLIB_STATUS_RET_CHECK(QLIB_EraseSection(qlibContext, 7, false));
278 // At this point section 0 is plain write. Now we can:
279 // 1. Write to it fw
280 // 2. Calculate crc of the section with fw.
281 // Section crc is calculated and compared to this value every boot. If integrity damaged will
282 // result in fallback mapping to section 7
283 // 3. calculate sha256 of the section with fw.
284 // section sha256 is calculated and compared to this value during configuration. This enables you
285 // write your firmware where/as long as you want since if integrity is damaged configuration will
286 // fail.
287 // 4. configure these values adding write protected policy.
288 // For the sake of example we use precalculated crc32/sha256 for 1M of 0xff, assuming flash is empty,
289 // so this code runs successfully on formatted flash
290
291 // crc32/sha256 for 1M of 0xff. For sha we work with bytes' sequence to avoid endianity issues
292 crc_u32 = 0x956bac74;
293
294 // sha256 for 1M of 0xff. We work with bytes' sequence to avoid endianity issues
295 memcpy(&digest_u64, digest_bytes_u8, sizeof(digest_u64));
296
297 // final security configuration of the boot section
298 QLIB_STATUS_RET_CHECK(QLIB_SAMPLE_Fallback_config_section(qlibContext, &id, die_u8, 0, &crc_u32, &digest_u64));
299
300 // write a unique string to the fallback section for later reading during test flow
301 QLIB_STATUS_RET_CHECK(QLIB_Write(qlibContext, buf_u8, 7, 0, FW_STRING_SIZE, false));
302
303 return QLIB_STATUS__OK;
304}
305#endif // QLIB_ONLY_FALLBACK

307QLIB_STATUS_T QLIB_SAMPLE_Fallback_field_flow(QLIB_CONTEXT_T* qlibContext)
308{
309 QLIB_HW_VER_T hwVer;
310 QLIB_ID_T id;
311 uint8_t boot_str_u8[FW_STRING_SIZE] = FW_STRING_BOOT;
312 uint8_t fb_str_u8[FW_STRING_SIZE] = FW_STRING_FB;
313 uint8_t on_flash_u8[FW_STRING_SIZE];
314 uint8_t hacker_data_u8[] = {0xde, 0xad, 0xbe, 0xef};
315 QLIB_REG_SSR_T ssr;
316
317 // See what chip is connected and its unique ids. We might need this info to obtain keys for this
318 // chips according to formula or via DB see QLIB_SAMPLE_GetSectionAccessKey() function
319 QLIB_STATUS_RET_CHECK(QLIB_GetHWVersion(qlibContext, &hwVer));
320 QLIB_STATUS_RET_CHECK(QLIB_GetId(qlibContext, &id));
321
322 // ensure we have expected data on boot section
323 QLIB_STATUS_RET_CHECK(QLIB_Read(qlibContext, on_flash_u8, 0, 0, FW_STRING_SIZE, false, false));
324 QLIB_ASSERT_RET(0 == memcmp(boot_str_u8, on_flash_u8, FW_STRING_SIZE), QLIB_STATUS__TEST_FAIL);
325
326 // now lets test fall back feature. To achieve this, we corrupt the boot section data
327 // integrity through section 1.
328
329 QLIB_STATUS_RET_CHECK(QLIB_Write(qlibContext, hacker_data_u8, 1, 0, sizeof(hacker_data_u8), false));
330
331 // In real life crc check will be triggered by hard reset aka reboot. To simulate a power cycle
332 // in test flow we use reset function to initiate the CRC check
333 QLIB_STATUS_RET_CHECK(QLIB_ResetFlash(qlibContext));
334
335 // now flash should map plain read to the fallback section
336 QLIB_STATUS_RET_CHECK(QLIB_Read(qlibContext, on_flash_u8, 0, 0, FW_STRING_SIZE, false, false));
337 QLIB_ASSERT_RET(0 == memcmp(fb_str_u8, on_flash_u8, FW_STRING_SIZE), QLIB_STATUS__TEST_FAIL);
338
339 // we can test chip is in fall back mode according to ssr bit
340 QLIB_STATUS_RET_CHECK(QLIB_GetStatus(qlibContext, &ssr));
341 QLIB_ASSERT_RET(ssr.asStruct.FB_REMAP == 1, QLIB_STATUS__TEST_FAIL);
342
343 return QLIB_STATUS__OK;
344}
