function(add_custom_target TARGET SOURCES SOURCES_LIST DEFINES DEFINES_LIST INCLUDES INCLUDES_LIST LIBS)
    if (NOT ${TARGET} MATCHES None)
        set(project_name ${TARGET})

        set(combined_sources)
        if (NOT ${SOURCES_LIST} MATCHES None)
            list(APPEND combined_sources ${${SOURCES_LIST}})
        endif()
        if (NOT ${SOURCES} MATCHES None)
            separate_arguments(SOURCES)
            list(APPEND combined_sources ${SOURCES})
        endif()
        if (combined_sources)
            if (${ARTIFACT_TYPE} STREQUAL "executable")
                add_executable(${project_name} ${combined_sources})
            elseif (${ARTIFACT_TYPE} STREQUAL "shared_library")
                add_library(${project_name} SHARED ${combined_sources})
            elseif (${ARTIFACT_TYPE} STREQUAL "library")
                add_library(${project_name} STATIC ${combined_sources})
            endif()
        endif()

        set(combined_defines)
        if (NOT ${DEFINES_LIST} MATCHES None)
            list(APPEND combined_defines ${${DEFINES_LIST}})
        endif()
        if (NOT ${DEFINES} MATCHES None)
            separate_arguments(DEFINES)
            list(APPEND combined_defines ${DEFINES})
        endif()
        if (combined_defines)
            target_compile_definitions(${project_name} PRIVATE ${combined_defines})
        endif()

        set(combined_includes)
        if (NOT ${INCLUDES_LIST} MATCHES None)
            list(APPEND combined_includes ${${INCLUDES_LIST}})
        endif()
        if (NOT ${INCLUDES} MATCHES None)
            separate_arguments(INCLUDES)
            list(APPEND combined_includes ${INCLUDES})
        endif()
        if (combined_includes)
            target_include_directories(${project_name} PRIVATE ${combined_includes})
        endif()
        
        if (NOT ${LIBS} MATCHES None)
            target_link_libraries(${project_name} PRIVATE ${${LIBS}})
        endif()
    
    endif()
endfunction()

add_custom_target(${TARGET} ${SOURCES} ${SOURCES_LIST} ${DEFINES} ${DEFINES_LIST} ${INCLUDES} ${INCLUDES_LIST} ${LIBS})
